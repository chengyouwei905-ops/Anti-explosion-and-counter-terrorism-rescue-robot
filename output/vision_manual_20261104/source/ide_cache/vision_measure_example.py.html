7PROJECT_ROOT = "/data/guanggongsai"
8if PROJECT_ROOT not in sys.path:
9 sys.path.append(PROJECT_ROOT)

11from calibration_params import load_calibration
12from tasks.task_qr import QRTask
13from tasks.task_bomb import BombTask
14from yolo_detector import YoloDetector
15try:
16 import ujson as json
17except ImportError:
18 import json

21def main(mode="qr", max_frames=120, target_color_id=1, detector_backend=None):
22 """Modes qr and bomb_refine. Camera is physically stationary for this demo.
23
24 Future MCU integration invokes task APIs directly. This example has no
25 automatic scan pass and never moves the robot or controls the laser.
26 """
27 if mode not in ("qr", "bomb_refine"):
28 raise ValueError("mode must be qr or bomb_refine")
29 calibration = load_calibration(PROJECT_ROOT)
30 print("HANDEYE_ACCEPTED", calibration.handeye_accepted)
31 if mode == "qr":
32 task = QRTask(calibration)
33 else:
34 with open(PROJECT_ROOT+"/models/bomb_three_colors.template.json", "r") as stream:
35 metadata = json.load(stream)
36 if metadata.get("model_path") and not metadata["model_path"].startswith("/"):
37 metadata["model_path"] = PROJECT_ROOT+"/"+metadata["model_path"]
38 detector = YoloDetector(metadata, detector_backend)
39 loaded = detector.load()
40 if not loaded["ok"]:
41 print("BOMB_NOT_READY", loaded["reason"])
42 return loaded
43 task = BombTask(calibration, detector)
44 task.start_refine(target_color_id)
45
46 sensor = raw = gray = preview = None
47 display_touched = media_touched = False
48 frame_id, last_print = 0, time.ticks_ms()
49 try:
50 from media.sensor import Sensor
51 from media.display import Display
52 from media.media import MediaManager, CAM_CHN_ID_0
53 sensor = Sensor(id=0)
54 sensor.reset()
55 sensor.set_framesize(Sensor.VGA, chn=CAM_CHN_ID_0)
56 sensor.set_pixformat(Sensor.RGB888, chn=CAM_CHN_ID_0)
57 display_touched = True
58 Display.init(Display.ST7701, width=800, height=480, to_ide=False)
59 media_touched = True
60 MediaManager.init()
61 sensor.run()
62 for _ in range(10):
63 raw = sensor.snapshot(chn=CAM_CHN_ID_0, timeout=2000, auto_reboot=False)
64 raw = None
65 while max_frames is None or frame_id < max_frames:
66 os.exitpoint()
67 raw = sensor.snapshot(chn=CAM_CHN_ID_0, timeout=2000, auto_reboot=False)
68 frame_id += 1
69 sampled_ms = time.ticks_ms() # Return time, not exposure time.
70 if mode == "qr":
71 gray = raw.to_grayscale()
72 result = task.measure(gray, frame_id, geometry_confirmed=True, stable=True)
73 preview=gray.to_rgb565()
74 Display.show_image(preview)
75 else:
76 result = task.poll(raw, frame_id, geometry_confirmed=True, stable=True)
77 preview=raw.to_rgb565()
78 Display.show_image(preview)
79 result["capture_return_ms"] = sampled_ms
80 now = time.ticks_ms()
81 if time.ticks_diff(now, last_print) >= 500:
82 # Large matrices are intentionally not printed every frame.
83 print("VISION", frame_id, result["reason"], "p_E_mm", result.get("position_tool_mm"),
84 "control_ready", result["control_ready"])
85 last_print = now
86 raw = gray = preview = None
87 gc.collect()
88 finally:
89 raw = gray = preview = None
90 try:
91 if isinstance(task, BombTask):
92 task.close()
93 finally:
94 try:
95 if sensor is not None:
96 sensor.stop()
97 finally:
98 try:
99 if display_touched:
100 Display.deinit()
101 finally:
102 gc.collect()
103 os.exitpoint(os.EXITPOINT_ENABLE_SLEEP)
104 time.sleep_ms(100)
105 if media_touched:
106 MediaManager.deinit()
107
108
109if __name__ == "__main__":
110 main()
