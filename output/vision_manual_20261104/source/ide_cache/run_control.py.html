6class RunControl:

16 @property
17 def state(self):
18 return "FAULT" if self.error else "STARTED" if self.running else "WAIT_START"

20 def _start(self):
21 # Purge before START; an immediate MCU reply after write must survive.
22 self.link.set_command_acceptance(True)
23 try: self.link.send(p.STATUS,cfg.TOTAL_START_BYTES)
24 except BaseException:
25 self.link.set_command_acceptance(False)
26 raise
27 self.manager.resume()
28 self.running=True
29 self.starts+=1
30 self.generation+=1
31 print("K230_TOTAL_START",cfg.TOTAL_START_BYTES.decode())
32
33 def _stop(self):
34 self.running=False
35 self.generation+=1
36 # Pause/discard poses before writing STOP; no binary CANCELLED packet.
37 try:
38 self.manager.pause()
39 if cfg.TOTAL_STOP_POLICY=="restart":self.manager.reset_session()
40 finally:
41 try:
42 self.link.send(p.STATUS,cfg.TOTAL_STOP_BYTES)
43 self.stops+=1
44 print("K230_TOTAL_STOP",cfg.TOTAL_STOP_BYTES.decode())
45 finally: self.link.set_command_acceptance(False)

47 def poll(self):
48 if self.error: return
49 try:
50 if not self.button.healthy(): raise OSError("USER_BUTTON_FAILED:"+str(self.button.error))
51 for _ in range(8):
52 if not self.button.pressed(): break
53 if self.running: self._stop()
54 else: self._start()
55 except Exception as exc:
56 self.error=type(exc).__name__+":"+str(exc)
57 if self.running:
58 try: self._stop()
59 except Exception: pass
60 self.running=False
61 self.manager.pause()
62 try: self.link.set_command_acceptance(False)
63 except Exception: pass
64 print("K230_CONTROL_ERROR",self.error)
65
66 def close(self):
67 if self.running:
68 try: self._stop()
69 except Exception as exc:
70 self.error=type(exc).__name__+":"+str(exc)
