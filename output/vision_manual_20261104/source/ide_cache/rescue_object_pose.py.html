12def body_surface(shape_id,angles=24):
13 height=cfg.OBJECT_HEIGHT_MM
14 if shape_id not in (1,2,3):raise ValueError("INVALID_SHAPE_ID")
15 points=[]
16 levels=2 if shape_id!=3 else 9
17 for level in range(levels):
18 z=-height/2+height*level/(levels-1)
19 if shape_id==1:radius=cfg.CYLINDER_DIAMETER_MM/2
20 elif shape_id==2:
21 ratio=(z+height/2)/height
22 radius=(cfg.FRUSTUM_BOTTOM_DIAMETER_MM*(1-ratio)+cfg.FRUSTUM_TOP_DIAMETER_MM*ratio)/2
23 else:
24 R=cfg.DRUM_SIDE_ARC_RADIUS_MM
25 radius=cfg.DRUM_END_DIAMETER_MM/2+math.sqrt(R*R-z*z)-math.sqrt(R*R-(height/2)**2)
26 for n in range(angles):
27 t=2*math.pi*n/angles
28 points.append((radius*math.cos(t),radius*math.sin(t),z))
29 return points

32def _basis(theta,phi):
33 axis=[math.sin(theta)*math.cos(phi),math.sin(theta)*math.sin(phi),math.cos(theta)]
34 # Smooth fixed angular parameterization; no cross-product branch flips.
35 first=[math.cos(theta)*math.cos(phi),math.cos(theta)*math.sin(phi),-math.sin(theta)]
36 second=[-math.sin(phi),math.cos(phi),0.0]
37 return axis,first,second

40def project_surface(camera,parameters,surface):
41 H=cfg.OBJECT_HEIGHT_MM
42 C=[v*H for v in parameters[:3]]
43 if not cfg.OBJECT_MIN_DEPTH_MM<=C[2]<=cfg.OBJECT_MAX_DEPTH_MM:raise ValueError("RESCUE_OBJECT_DEPTH_OUT_OF_RANGE")
44 if not 0.015<parameters[3]<math.pi-0.015:raise ValueError("RESCUE_OBJECT_AXIS_DEGENERATE")
45 axis,first,second=_basis(parameters[3],parameters[4]);out=[]
46 for x,y,z in surface:
47 px=C[0]+first[0]*x+second[0]*y+axis[0]*z
48 py=C[1]+first[1]*x+second[1]*y+axis[1]*z
49 pz=C[2]+first[2]*x+second[2]*y+axis[2]*z
50 if pz<=10:raise ValueError("RESCUE_OBJECT_BEHIND_CAMERA")
51 out.append((camera.fx*px/pz,camera.fy*py/pz))
52 return out

55def _supports(points,directions):
56 return [max(x*dx+y*dy for x,y in points) for dx,dy in directions]

59def _evaluate(camera,p,surface,directions,observed,jacobian=False):
60 prediction=_supports(project_surface(camera,p,surface),directions)
61 residual=[a-b for a,b in zip(prediction,observed)]
62 if not jacobian:return residual,[]
63 rows=[[0.0]*5 for _ in residual]
64 for col in range(5):
65 delta=0.002 if col<3 else 0.001
66 shifted=list(p);shifted[col]+=delta
67 values=_supports(project_surface(camera,shifted,surface),directions)
68 for i,value in enumerate(values):rows[i][col]=(value-prediction[i])/delta
69 return residual,rows

72def _equations(residual,rows):
73 H=[[sum(row[i]*row[j] for row in rows) for j in range(5)] for i in range(5)]
74 g=[sum(row[i]*r for row,r in zip(rows,residual)) for i in range(5)]
75 return H,g

78def _refine(camera,initial,surface,directions,observed,iterations=15):
79 p=list(initial);damping=0.01
80 for _ in range(iterations):
81 residual,rows=_evaluate(camera,p,surface,directions,observed,True)
82 H,g=_equations(residual,rows);cost=sum(r*r for r in residual)
83 improved=False
84 for _ in range(5):
85 A=[[H[i][j]+(damping*max(H[i][i],1e-6) if i==j else 0) for j in range(5)] for i in range(5)]
86 try:
87 step=_solve(A,[-v for v in g]);candidate=[a+b for a,b in zip(p,step)]
88 after=sum(r*r for r in _evaluate(camera,candidate,surface,directions,observed)[0])
89 except (ValueError,OverflowError,ZeroDivisionError):after=1e30
90 if after<cost:
91 p=candidate;damping=max(1e-7,damping/3);improved=True;break
92 damping*=6
93 if not improved or cost-after<1e-6 or sum(v*v for v in step)<1e-10:break
94 residual,rows=_evaluate(camera,p,surface,directions,observed,True)
95 return p,sum(r*r for r in residual),rows

98class RescueObjectEstimator:
99 def __init__(self,camera):self.camera=camera;self.previous={}
100 def reset(self):self.previous={}
101 def fit(self,pixels,shape_id,pixel_noise=0.8):
102 try:
103 if not 24<=len(pixels)<=256:raise ValueError("RESCUE_CONTOUR_POINT_COUNT")
104 rays=[self.camera.ray(p) for p in pixels]
105 points=[(self.camera.fx*r[0]/r[2],self.camera.fy*r[1]/r[2]) for r in rays]
106 directions=[(math.cos(2*math.pi*i/24),math.sin(2*math.pi*i/24)) for i in range(24)]
107 observed=_supports(points,directions)
108 xmin,xmax=min(x for x,y in points),max(x for x,y in points)
109 ymin,ymax=min(y for x,y in points),max(y for x,y in points)
110 if min(xmax-xmin,ymax-ymin)<12:raise ValueError("RESCUE_OBJECT_TOO_SMALL")
111 surface=body_surface(shape_id)
112 mean=[sum(p[i] for p in points)/len(points) for i in range(2)]
113 xx=sum((x-mean[0])**2 for x,y in points)
114 yy=sum((y-mean[1])**2 for x,y in points)
115 xy=sum((x-mean[0])*(y-mean[1]) for x,y in points)
116 phi=0.5*math.atan2(2*xy,xx-yy)
117 initials=[]
118 if shape_id in self.previous:initials.append(self.previous[shape_id])
119 else:
120 for angle in (0.55,1.15,1.8,2.5):
121 for azimuth in (phi,phi+math.pi) if shape_id==2 else (phi,):
122 trial=[0.,0.,8.,angle,azimuth]
123 projected=project_surface(self.camera,trial,surface)
124 ratio=((max(x for x,y in projected)-min(x for x,y in projected))/(xmax-xmin)+(max(y for x,y in projected)-min(y for x,y in projected))/(ymax-ymin))/2
125 z=max(cfg.OBJECT_MIN_DEPTH_MM+40,min(cfg.OBJECT_MAX_DEPTH_MM-40,8*cfg.OBJECT_HEIGHT_MM*ratio))
126 initials.append([(xmin+xmax)/2/self.camera.fx*z/cfg.OBJECT_HEIGHT_MM,(ymin+ymax)/2/self.camera.fy*z/cfg.OBJECT_HEIGHT_MM,z/cfg.OBJECT_HEIGHT_MM,angle,azimuth])
127 solutions=[]
128 for initial in initials:
129 try:solutions.append(_refine(self.camera,initial,surface,directions,observed))
130 except (ValueError,OverflowError,ZeroDivisionError):pass
131 if not solutions:raise ValueError("RESCUE_OBJECT_GEOMETRY_UNOBSERVABLE")
132 solutions.sort(key=lambda v:v[1]);p,cost,rows=solutions[0]
133 rms=math.sqrt(cost/len(observed))
134 if rms>max(cfg.OBJECT_MAX_REPROJECTION_PX,2*pixel_noise):raise ValueError("RESCUE_OBJECT_REPROJECTION_ERROR")
135 center=[v*cfg.OBJECT_HEIGHT_MM for v in p[:3]]
136 for alternate,other_cost,_ in solutions[1:]:
137 if other_cost<=cost+len(observed)*0.25 and norm([(a-b)*cfg.OBJECT_HEIGHT_MM for a,b in zip(p[:3],alternate[:3])])>cfg.OBJECT_AMBIGUITY_SEPARATION_MM:
138 raise ValueError("RESCUE_OBJECT_POSE_AMBIGUOUS")
139 H,_=_equations([0.]*len(rows),rows)
140 sigma=max(pixel_noise,rms,0.6)
141 uncertainty=0.0
142 for i in range(3):
143 rhs=[float(i==j) for j in range(5)]
144 diagonal=_solve(H,rhs)[i]
145 if diagonal<=0 or not finite(diagonal):raise ValueError("RESCUE_OBJECT_GEOMETRY_UNOBSERVABLE")
146 uncertainty=max(uncertainty,3*sigma*math.sqrt(diagonal)*cfg.OBJECT_HEIGHT_MM)
147 if uncertainty>cfg.OBJECT_MAX_POSITION_3SIGMA_MM:raise ValueError("RESCUE_OBJECT_POSITION_UNCERTAIN")
148 self.previous[shape_id]=list(p)
149 return {"ok":True,"reason":"OK","position_camera_mm":center,"axis_camera":_basis(p[3],p[4])[0],
150 "reference_point":"rescue_object_axis_midpoint","orientation_valid":False,
151 "quality":{"reprojection_rms_px":rms,"position_3sigma_mm":uncertainty,"model":"known_body_of_revolution_silhouette"}}
152 except (ValueError,OverflowError,ZeroDivisionError) as exc:
153 self.previous.pop(shape_id,None)
154 return {"ok":False,"reason":str(exc),"position_camera_mm":None}

157def extract_white_object(raw,bbox):
158 """Bounded local white component; background contact/partial objects rejected."""
159 from target_color import _component
160 x,y,w,h=bbox
161 if min(x,y)<1 or x+w>=raw.width()-1 or y+h>=raw.height()-1:raise ValueError("RESCUE_TARGET_CLIPPED")
162 margin=max(5,int(max(w,h)*0.12))
163 x0,y0=max(0,int(x)-margin),max(0,int(y)-margin)
164 x1,y1=min(raw.width(),int(x+w)+margin+1),min(raw.height(),int(y+h)+margin+1)
165 step=1
166 while ((x1-x0+step-1)//step)*((y1-y0+step-1)//step)>cfg.OBJECT_MAX_ROI_CELLS:step+=1
167 columns,rows=(x1-x0+step-1)//step,(y1-y0+step-1)//step
168 gray=raw.to_grayscale();values=bytearray(columns*rows);hist=[0]*256
169 try:
170 for gy in range(rows):
171 for gx in range(columns):
172 value=gray.get_pixel(x0+gx*step,y0+gy*step)
173 values[gy*columns+gx]=value;hist[value]+=1
174 finally:gray=None
175 total=len(values);weighted=sum(i*n for i,n in enumerate(hist));weight=partial=0;best=-1;threshold=80
176 for i in range(255):
177 weight+=hist[i];partial+=i*hist[i]
178 if not weight or weight==total:continue
179 score=(weighted*weight-partial*total)**2/(weight*(total-weight))
180 if score>best:best=score;threshold=i
181 threshold=max(55,threshold)
182 mask=bytearray(int(v>threshold) for v in values)
183 components=[]
184 for i in range(len(mask)):
185 if mask[i]==1:
186 item=_component(mask,i,columns,rows,1,2)
187 inside=sum(x<=x0+(n%columns)*step<=x+w and y<=y0+(n//columns)*step<=y+h for n in item)
188 if inside>=cfg.OBJECT_MIN_COMPONENT_SAMPLES:components.append((inside,item))
189 if not components:raise ValueError("RESCUE_WHITE_OBJECT_NOT_FOUND")
190 components.sort(key=lambda item:item[0],reverse=True);component=components[0][1]
191 if len(components)>1 and components[1][0]>components[0][0]*0.35:raise ValueError("RESCUE_WHITE_COMPONENTS_AMBIGUOUS")
192 if any(i%columns in (0,columns-1) or i//columns in (0,rows-1) for i in component):raise ValueError("RESCUE_CONTOUR_BACKGROUND_CONTACT")
193 occupied=set(component);boundary=[]
194 for i in component:
195 if any(n not in occupied for n in (i-1,i+1,i-columns,i+columns)):
196 boundary.append((x0+(i%columns)*step,y0+(i//columns)*step))
197 if len(boundary)>192:boundary=[boundary[i*len(boundary)//192] for i in range(192)]
198 return boundary,max(0.8,step*0.6)
