30 def pause(self):
31 """Freeze the active request; never transmit a cancellation/old pose here."""
32 if self.paused: return
33 self.paused=True
34 self.state=PAUSED
35 if self.request is not None:
36 self._elapsed_before_pause+=self.diff(self.clock(),self.started_ms)
37 if self.average is not None: self.average.reset()
38 if self.job is not None:
39 method=getattr(self.job,"pause_window",self.job.reset_window)
40 method()
41 # Scan results contain identities/slots, not a stale measured pose.
42 if (self.pending is not None and self.pending[0] in (p.SUCCESS,p.OBSERVATION_ONLY)
43 and self.job is not None and not self.job.is_scan):
44 self.pending=None

121 def on_command(self,frame):
122 if self.paused: return
123 if frame.cmd==p.HEARTBEAT: return
124 now=self.clock()
125 key=(frame.seq,frame.cmd,bytes(frame.payload))
126 if self._seen(key,now): return
127 self._remember(key,now)
128 try:
129 request=p.decode_command(frame)
130 except ValueError as exc:
131 invalid={"cmd":frame.cmd,"input_seq":frame.seq,"target_id":None,"slot_id":None,"key":key}
132 self._transmit_once(invalid,self._new_run_id(),p.FAILED,str(exc))
133 return
134 if request["cmd"]==p.STOP:
135 if self.request is not None: self._finish_active(p.CANCELLED,"CANCELLED")
136 return # STOP while idle is intentionally silent.
137 if self.request is not None: self._finish_active(p.CANCELLED,"PREEMPTED")
138 self.request=request
139 self.run_id=self._new_run_id()
140 self.started_ms=now
141 self._settle_started_ms=now
142 self._elapsed_before_pause=0
143 self.timeout_ms=cfg.SCAN_TIMEOUT_MS if request["cmd"] in p.SCAN_COMMANDS else cfg.MEASUREMENT_TIMEOUT_MS
144 self.average=MeasurementAverage(cfg.MEASUREMENT_SAMPLES,cfg.MAX_POSITION_SCATTER_MM,cfg.MAX_ROTATION_SPREAD_DEG)
145 self.pending=None
146 self.last_reason="STARTING"
147 try:
148 self.job=self.factory.start(request)
149 self.state=RUNNING if self.job.is_scan else SETTLING
150 except Exception as exc:
151 self.job=None
152 self.pending=(p.FAILED,str(exc),None)
153 self.state=REPORT

155 def tick(self):
156 if self.state not in (SETTLING,RUNNING): return
157 elapsed=self._elapsed()
158 if elapsed>=self.timeout_ms:
159 self.pending=(p.FAILED,"TIMEOUT",None)
160 self.state=REPORT
161 elif self.state==SETTLING and self.diff(self.clock(),self._settle_started_ms)>=cfg.SETTLE_MS:
162 self.state=RUNNING

178 def poll_frame(self,frame):
179 if self.state!=RUNNING or self.job is None: return
180 if self.last_frame_id is not None and frame["frame_id"]<=self.last_frame_id: return
181 self.last_frame_id=frame["frame_id"]
182 self.observe_camera_epoch(frame["epoch"])
183 if self.state!=RUNNING: return
184 try:
185 result=self.job.poll(frame)
186 if self._elapsed()>=self.timeout_ms:
187 self.pending=(p.FAILED,"TIMEOUT",None); self.state=REPORT; return
188 if self.job.is_scan:
189 if result.get("ok") and result.get("complete"):
190 self.pending=(p.SUCCESS,"OK",result)
191 self.state=REPORT
192 elif terminal_reason(result.get("reason","")):
193 self.pending=(p.FAILED,result["reason"],None)
194 self.state=REPORT
195 else: self.last_reason=result.get("reason","SCANNING")
196 return
197 if result.get("ok") and self.request["cmd"]==p.QR_START and result.get("T_tool_from_qr") is None:
198 raise ValueError("INVALID_QR_ROTATION")
199 averaged,reason=self.average.add(result)
200 self.last_reason=reason
201 if averaged is not None:
202 accepted=averaged.get("control_ready") is True
203 if not accepted and not cfg.ALLOW_CANDIDATE_REPORTS:
204 self.pending=(p.FAILED,"CALIBRATION_NOT_READY",None)
205 else:
206 self.pending=(p.SUCCESS if accepted else p.OBSERVATION_ONLY,
207 "OK" if accepted else "HANDEYE_CANDIDATE_ONLY",averaged)
208 self.state=REPORT
209 elif terminal_reason(reason):
210 self.pending=(p.FAILED,reason,None)
211 self.state=REPORT
212 except Exception as exc:
213 self.pending=(p.FAILED,"VISION_ERROR",None)
214 self.last_reason=type(exc).__name__+":"+str(exc)
215 self.state=REPORT

217 def flush_report(self):
218 # Main pumps RX AGAIN before calling this, including after heavy inference.
219 if self.state!=REPORT or self.pending is None: return
220 status,reason,observation=self.pending
221 if status in (p.SUCCESS,p.OBSERVATION_ONLY) and observation is not None:
222 try: self.factory.completed(self.request,observation)
223 except Exception:
224 status,reason,observation=p.FAILED,"VISION_ERROR",None
225 self._finish_active(status,reason,observation)
