93def write_all(stream, data):
94 view, offset = memoryview(data), 0
95 while offset < len(view):
96 written = stream.write(view[offset:])
97 if not written or written < 0:
98 raise OSError("SHORT_FILE_WRITE")
99 offset += written

102class CaptureButton:
103 def __init__(self, pin, clock=ticks_ms):
104 self.pin, self.clock = pin, clock
105 self.reset()
106
107 def reset(self):
108 self.raw = self.stable = self.pin.value()
109 self.changed_at = self.clock()
110 # A held key at startup/recovery must first be released.
111 self.armed = self.stable != KEY_ACTIVE
112
113 def pressed(self):
114 now, value = self.clock(), self.pin.value()
115 if value != self.raw:
116 self.raw, self.changed_at = value, now
117 if self.raw != self.stable and ticks_diff(now, self.changed_at) >= DEBOUNCE_MS:
118 self.stable = self.raw
119 if self.stable != KEY_ACTIVE:
120 self.armed = True
121 elif self.armed:
122 self.armed = False
123 return True
124 return False

127def avi_header(width, height, fps, count, movie_bytes, max_jpeg, finalized):
128 """Standard single-stream MJPEG AVI; fixed 224-byte header, no audio."""
129 rate = max(1, int(fps * 1000 + 0.5))
130 interval = max(1, int(1000000000 / rate + 0.5))
131 avih = struct.pack("<14I", interval, int(max_jpeg * fps), 0,
132 0x10 if finalized else 0, count, 0, 1, max_jpeg,
133 width, height, 0, 0, 0, 0)
134 strh = struct.pack("<4s4sIHHIIIIIIIIhhhh", b"vids", b"MJPG", 0, 0, 0,
135 0, 1000, rate, 0, count, max_jpeg, 0xFFFFFFFF, 0,
136 0, 0, width, height)
137 strf = struct.pack("<IiiHH4sIiiII", 40, width, height, 1, 24,
138 b"MJPG", width * height * 3, 0, 0, 0, 0)
139 stream = b"strl" + b"strh" + struct.pack("<I", len(strh)) + strh
140 stream += b"strf" + struct.pack("<I", len(strf)) + strf
141 hdrl = b"hdrl" + b"avih" + struct.pack("<I", len(avih)) + avih
142 hdrl += b"LIST" + struct.pack("<I", len(stream)) + stream
143 end_size = 224 + movie_bytes + (8 + count * 16 if finalized else 0)
144 header = b"RIFF" + struct.pack("<I", end_size - 8) + b"AVI "
145 header += b"LIST" + struct.pack("<I", len(hdrl)) + hdrl
146 header += b"LIST" + struct.pack("<I", 4 + movie_bytes) + b"movi"
147 if len(header) != 224:
148 raise ValueError("INVALID_AVI_HEADER")
149 return header

152class AviRecorder:
153 """Append JPEGs and bounded index entries; publish .avi only after close."""
154 def __init__(self, base, width=WIDTH, height=HEIGHT, fps=TARGET_FPS):
155 self.base, self.width, self.height = base, width, height
156 self.nominal_fps = fps
157 self.count = self.movie_bytes = self.max_jpeg = 0
158 self.index = bytearray()
159 self.first_ms = self.last_ms = None
160 self.file = self.csv = None
161 self.closed = False
162 self.path, self.pending = base + ".avi", base + ".avi.part"
163 for extension in (".avi", ".avi.part", ".frames.csv", ".json"):
164 if exists(base + extension):
165 raise OSError("CLIP_ALREADY_EXISTS")
166 try:
167 self.file = open(self.pending, "w+b")
168 write_all(self.file, self._header(False))
169 self.csv = open(base + ".frames.csv", "w")
170 self.csv.write("frame_index,capture_elapsed_ms,jpeg_bytes\n")
171 except BaseException:
172 if self.csv:
173 self.csv.close()
174 if self.file:
175 self.file.close()
176 self.closed = True
177 raise

179 def fps(self):
180 span = ticks_diff(self.last_ms, self.first_ms) if self.count > 1 else 0
181 return (self.count - 1) * 1000.0 / span if span > 0 else self.nominal_fps
182
183 def _header(self, final):
184 return avi_header(self.width, self.height, self.fps(), self.count,
185 self.movie_bytes, self.max_jpeg, final)
186
187 def append(self, jpeg, captured_ms):
188 if self.closed:
189 raise OSError("RECORDER_CLOSED")
190 size = len(jpeg)
191 if size < 4 or jpeg[:2] != b"\xff\xd8" or jpeg[-2:] != b"\xff\xd9":
192 raise ValueError("INVALID_JPEG")
193 chunk_bytes = 8 + size + (size & 1)
194 projected = 224 + self.movie_bytes + chunk_bytes + 8 + (self.count + 1) * 16
195 if projected > MAX_CLIP_BYTES:
196 return False
197 offset = self.movie_bytes + 4 # Relative to the 'movi' list type.
198 write_all(self.file, b"00dc" + struct.pack("<I", size))
199 write_all(self.file, jpeg)
200 if size & 1:
201 write_all(self.file, b"\x00")
202 self.index.extend(struct.pack("<4sIII", b"00dc", 0x10, offset, size))
203 self.movie_bytes += chunk_bytes
204 self.max_jpeg = max(self.max_jpeg, size)
205 if self.first_ms is None:
206 self.first_ms = captured_ms
207 self.last_ms = captured_ms
208 self.count += 1
209 self.csv.write("%d,%d,%d\n" % (self.count - 1, ticks_diff(captured_ms, self.first_ms), size))
210 return True
211
212 def checkpoint(self):
213 position = 224 + self.movie_bytes
214 self.file.seek(0)
215 write_all(self.file, self._header(False))
216 self.file.seek(position)
217 self.file.flush()
218 self.csv.flush()
219 sync_storage()
220
221 def close(self, reason="button", encoder_name="unknown"):
222 if self.closed:
223 return None
224 self.closed = True
225 try:
226 # Discard an incomplete last chunk by replacing it at the committed end.
227 self.file.seek(224 + self.movie_bytes)
228 write_all(self.file, b"idx1" + struct.pack("<I", len(self.index)))
229 write_all(self.file, self.index)
230 self.file.seek(0)
231 write_all(self.file, self._header(True))
232 self.file.flush()
233 self.csv.flush()
234 finally:
235 try:
236 self.file.close()
237 finally:
238 self.csv.close()
239 sync_storage()
240 if self.count:
241 os.rename(self.pending, self.path)
242 summary = {"version": APP_VERSION, "video": self.path if self.count else None,
243 "pending_video": None if self.count else self.pending,
244 "image_size": [self.width, self.height], "sensor_id": 0,
245 "sensor_pixel_format": "RGB888", "undistorted": False,
246 "cropped": False, "mirrored_by_script": False, "overlay_in_video": False,
247 "codec": "MJPG", "encoder": encoder_name, "jpeg_quality": JPEG_QUALITY,
248 "frame_count": self.count, "requested_fps": self.nominal_fps,
249 "measured_fps": self.fps(), "stop_reason": reason,
250 "timestamps": self.base + ".frames.csv"}
251 with open(self.base + ".json", "w") as stream:
252 json.dump(summary, stream)
253 sync_storage()
254 self.index = bytearray()
255 return summary

331 def start_recording(self):
332 if self.recorder is not None:
333 return

345 def finish_recording(self, reason="button"):
346 recorder, self.recorder = self.recorder, None
347 if recorder is None:
348 return
349 self.status, self.detail = "正在保存", "请勿断电或拔卡"
350 try:
351 result = recorder.close(reason, self.encoder.name)
352 except Exception as exc:
353 self.status, self.detail, self.error = "保存失败", str(exc), True
354 print("REC_SAVE_ERROR", recorder.pending, type(exc).__name__, str(exc))
355 else:
356 if result["frame_count"]:
357 self.status, self.error = "已保存", False
358 self.detail = "%d 帧，%.1f FPS" % (result["frame_count"], result["measured_fps"])
359 print("REC_SAVED", result["video"], result["frame_count"], "frames", result["measured_fps"], "fps", reason)
360 else:
361 self.status, self.detail, self.error = "未录到画面", "请查看控制台", True
362 print("REC_EMPTY", recorder.pending, reason)

365 def read_frame(self):
366 try:
367 raw = self.sensor.snapshot(chn=self.channel, timeout=SNAPSHOT_TIMEOUT_MS, auto_reboot=False)
368 self.failures = 0
369 return raw
370 except (RuntimeError, OSError) as exc:
371 self.finish_recording("camera_error")
372 self.status, self.detail, self.error = "相机重试", "恢复后请重新按键", True
373 self.button.reset()
374 self.failures += 1
375 print("CAPTURE_CAMERA_ERROR", self.failures, str(exc))
376 if self.failures >= SNAPSHOT_ATTEMPTS:
377 if self.restarts >= MAX_CAMERA_RESTARTS:
378 raise RuntimeError("CAMERA_RECOVERY_LIMIT")
379 self.restarts += 1
380 old, self.sensor = self.sensor, None
381 old.stop()
382 old = None
383 gc.collect()
384 sleep_ms(200)
385 self._new_sensor()
386 self.sensor.run()
387 sleep_ms(500)
388 self.failures = 0
389 self.button.reset()
390 return None

417 def run(self, max_frames=None, auto_record=False):
418 raw = jpeg = None
419 last_capture = last_preview = None
420 seen = 0
421 if auto_record:
422 self.start_recording()
423 while max_frames is None or seen < max_frames:
424 exitpoint()
425 if self.button.pressed():
426 if self.recorder is not None:
427 self.finish_recording()
428 else:
429 try:
430 self.start_recording()
431 last_capture = None
432 except Exception as exc:
433 self.status, self.detail, self.error = "开始失败", str(exc), True
434 print("REC_START_ERROR", type(exc).__name__, str(exc))
435 now = ticks_ms()
436 if self.recorder is not None and ticks_diff(now, self.record_started) >= MAX_RECORD_SECONDS * 1000:
437 self.finish_recording("duration_limit")
438 interval = max(1, 1000 // TARGET_FPS) if self.recorder is not None else PREVIEW_INTERVAL_MS
439 if last_capture is not None and ticks_diff(now, last_capture) < interval:
440 sleep_ms(5)
441 continue
442 last_capture = now
443 raw = self.read_frame()
444 if raw is None:
445 sleep_ms(20)
446 continue
447 seen += 1
448 captured = ticks_ms()
449 if self.recorder is not None:
450 try:
451 jpeg = self.encoder.encode(raw)
452 if not self.recorder.append(jpeg, captured):
453 self.finish_recording("size_limit")
454 elif ticks_diff(ticks_ms(), self.last_sync) >= SYNC_INTERVAL_MS:
455 self.recorder.checkpoint()
456 self.last_sync = ticks_ms()
457 free = free_megabytes(self.directory)
458 print("REC_PROGRESS", self.recorder.count, "frames", free, "MB free")
459 if free is not None and free < STOP_FREE_MB:
460 self.finish_recording("low_space")
461 except Exception as exc:
462 print("REC_FRAME_ERROR", type(exc).__name__, str(exc))
463 self.finish_recording("encode_or_write_error")
464 self.status, self.detail, self.error = "录像已停止", str(exc), True
465 finally:
466 jpeg = None
467 if last_preview is None or ticks_diff(ticks_ms(), last_preview) >= PREVIEW_INTERVAL_MS:
468 self.show(raw)
469 last_preview = ticks_ms()
470 raw = None
471 gc.collect()

473 def close(self):
474 self.finish_recording("program_exit")
475 for name in ("encoder", "sensor"):
476 resource = getattr(self, name)
477 setattr(self, name, None)
478 if resource is not None:
479 try:
480 resource.close() if name == "encoder" else resource.stop()
481 except Exception as exc:
482 print("CLEANUP_ERROR", name, str(exc))
483 self.screen = None
484 gc.collect()
485 if self.display_touched:
486 try:
487 self.Display.deinit()
488 except Exception as exc:
489 print("CLEANUP_ERROR display", str(exc))
490 self.display_touched = False
491 if self.media_touched:
492 try:
493 sleep_ms(100)
494 self.Media.deinit()
495 except Exception as exc:
496 print("CLEANUP_ERROR media", str(exc))
497 self.media_touched = False
