9class Calibration:
10 def __init__(self, K, D, T_tool_from_camera=None, handeye_accepted=False,
11 image_size=(640, 480), quality_review=None):
12 self.camera = CameraModel(K, D, image_size)
13 self.T_tool_from_camera = (validate_rigid(T_tool_from_camera)
14 if T_tool_from_camera is not None else None)
15 self.handeye_accepted = handeye_accepted is True
16 self.quality_review = quality_review or {}
17 # E is the existing calibration's commanded TCP / gripper-center frame.
18 self.tool_origin = "commanded_gripper_center_tcp"
19
20 def point_to_tool(self, point_camera_mm, allow_candidate=True):
21 if self.T_tool_from_camera is None:
22 return {"ok": False, "reason": "HANDEYE_NOT_CONFIGURED", "position_tool_mm": None,
23 "control_ready": False}
24 if not self.handeye_accepted and not allow_candidate:
25 return {"ok": False, "reason": "HANDEYE_NOT_ACCEPTED", "position_tool_mm": None,
26 "control_ready": False}
27 return {"ok": True, "reason": "OK" if self.handeye_accepted else "HANDEYE_CANDIDATE_ONLY",
28 "position_tool_mm": transform_point(self.T_tool_from_camera, point_camera_mm),
29 "handeye_accepted": self.handeye_accepted, "control_ready": self.handeye_accepted,
30 "coordinate_frame": "E_commanded_gripper_center", "units": "mm"}

33def load_calibration(project_root="/sdcard", handeye_accepted_override=False):
34 root = project_root.rstrip("/\\")
35 with open(root+"/calibration/results/camera_params.json", "r") as stream:
36 camera = json.load(stream)
37 handeye = None
38 review = {}
39 try:
40 with open(root+"/calibration/handeye/session_20261025_fixed/result/handeye_result.json", "r") as stream:
41 handeye_data = json.load(stream)
42 handeye = handeye_data["T_tool_from_camera"]
43 review = handeye_data.get("quality_review", {})
44 except OSError:
45 pass # Camera-only observations remain usable; no identity fallback.
46 return Calibration(camera["K"], camera["D"], handeye,
47 review.get("recommended_for_robot_use", False) or (handeye_accepted_override is True and handeye is not None),
48 camera["image_size"], review)
