28 def detector_result(self,result):
29 self.detections=[]
30 if not result.get("ok"):
31 self.error=result.get("reason","DETECTION_ERROR")
32 for item in result.get("detections",[]):
33 box=item.get("bbox_raw_px")
34 if box is None or len(box)!=4 or not all(finite(v) for v in box):continue
35 identity=item.get("shape_id") if self.kind=="rescue" else item.get("color_id")
36 if type(identity) is not int or not 1<=identity<=3:continue
37 selected=identity==self.target_id
38 in_roi=True
39 if self.kind in ("bomb","aim"):
40 if self.kind=="bomb":
41 from config import BOMB_DETECTION_ROI_RAW_PX
42 x,y,w,h=BOMB_DETECTION_ROI_RAW_PX
43 else:
44 from aim_config import TARGET_DETECTION_ROI_RAW_PX
45 x,y,w,h=TARGET_DETECTION_ROI_RAW_PX
46 cx,cy=box[0]+box[2]*.5,box[1]+box[3]*.5
47 in_roi=x<=cx<x+w and y<=cy<y+h
48 self.detections.append({"bbox_raw_px":tuple(box),"label":target_label(self.kind,identity),
49 "selected":selected,"score":item.get("score"),"in_roi":in_roi})
50
51 def qr_result(self,results):
52 self.detections=[]
53 for item in results:
54 corners=item.get("corners_px")
55 if not corners or len(corners)!=4:continue
56 if any(len(p)!=2 or not all(finite(v) for v in p) for p in corners):continue
57 xs,ys=[p[0] for p in corners],[p[1] for p in corners]
58 payload=str(item.get("payload") or "").replace("\n"," ")[:12]
59 self.detections.append({"bbox_raw_px":(min(xs),min(ys),max(xs)-min(xs),max(ys)-min(ys)),
60 "corners_px":tuple(tuple(p) for p in corners),
61 "label":"二维码:"+payload,"selected":True,"score":None,"in_roi":True})

63 def snapshot(self):
64 return {"frame_id":self.frame_id,"source_size":self.size,
65 "detections":list(self.detections),"error":self.error}

68class DetectorProbe:
69 def __init__(self,detector,trace):self.inner,self.trace=detector,trace
70 def __getattr__(self,name):return getattr(self.inner,name)
71 def detect(self,raw):
72 result=self.inner.detect(raw)
73 try:self.trace.detector_result(result)
74 except Exception as exc:self.trace.error="DISPLAY_METADATA_ERROR:"+str(exc)
75 return result

78class QRPoseProbe:

134def draw_detections(canvas,detections,source_size,pose_valid=None,preview_size=(640,480)):
135 sw,sh=source_size
136 width,height=preview_size
137 if sw<=0 or sh<=0:return
138 sx,sy=width/float(sw),height/float(sh)
139 for item in detections:
140 x,y,w,h=item["bbox_raw_px"]
141 if w<=0 or h<=0:continue
142 x0,y0=max(0,int(x*sx)),max(0,int(y*sy))
143 x1,y1=min(width-1,int((x+w)*sx)),min(height-1,int((y+h)*sy))
144 if x1<=x0 or y1<=y0:continue
145 selected=item.get("selected",False)
146 color=((255,80,80) if not item.get("in_roi",True) else
147 ((0,255,100) if pose_valid is True else (255,165,0))) if selected else (150,180,210)
148 canvas.draw_rectangle(x0,y0,x1-x0,y1-y0,color=color,thickness=3 if selected else 1)
149 cx,cy=(x0+x1)//2,(y0+y1)//2
150 canvas.draw_line(max(0,cx-5),cy,min(width-1,cx+5),cy,color=color,thickness=2)
151 canvas.draw_line(cx,max(0,cy-5),cx,min(height-1,cy+5),color=color,thickness=2)
152 corners=item.get("corners_px")
153 if corners:
154 points=[(max(0,min(width-1,int(p[0]*sx))),max(0,min(height-1,int(p[1]*sy)))) for p in corners]
155 for a,b in zip(points,points[1:]+points[:1]):
156 canvas.draw_line(a[0],a[1],b[0],b[1],color=color,thickness=2)
157 label=item["label"]
158 score=item.get("score")
159 if score is not None and finite(score):label+=" %d%%" % int(score*100+.5)
160 if selected:label+=" 区外" if not item.get("in_roi",True) else (" 有效" if pose_valid is True else " 已检测")
161 text_width=sum(14 if ord(c)>127 else 7 for c in label)
162 tx=max(0,min(x0,width-text_width))
163 ty=max(0,y0-18)
164 canvas.draw_string_advanced(tx,ty,14,label,color=color)
