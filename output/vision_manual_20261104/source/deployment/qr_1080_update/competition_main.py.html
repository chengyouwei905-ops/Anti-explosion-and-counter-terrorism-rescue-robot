28def pump_commands(link,manager):
29 # STOP is returned before the normal queue and clears queued old starts.
30 priority=link.pop_priority()
31 if priority is not None: manager.on_command(priority)
32 overflow=link.take_overflow()
33 if overflow is not None:
34 if manager.request is not None: manager.fail_active("COMMAND_QUEUE_OVERFLOW")
35 else: manager.reject_command(overflow,"COMMAND_QUEUE_OVERFLOW")
36 return
37 for _ in range(cfg.COMMANDS_PER_POLL):
38 frame=link.pop_command()
39 if frame is None: break
40 manager.on_command(frame)
41 if not link.healthy(): manager.fail_active("LINK_ERROR")

44def debug_preview(camera,manager,control,image=None):
45 renderer=getattr(camera,"render_debug",None)
46 if renderer is None:
47 if image is not None:camera.show(image)
48 return
49 state=control.state if control is not None and not control.running else manager.state
50 labels={"WAIT_START":"等待启动","FAULT":"故障","IDLE":"空闲",
51 "SETTLING":"等待停稳","RUNNING":"识别中","REPORT":"准备回传","PAUSED":"已停止"}
52 request=manager.request
53 cmd=request["cmd"] if request else None
54 count=len(manager.average.samples) if manager.average is not None else 0
55 error=getattr(control,"error",None) if control is not None else None
56 lines=["K230 视觉调试",labels.get(state,state),
57 "任务:"+protocol.NAMES.get(cmd,"--"),
58 "采样:%d/%d" % (count,cfg.MEASUREMENT_SAMPLES),
59 "帧:%d" % getattr(camera,"frame_id",0),"原因:",error or manager.last_reason]
60 if cmd==protocol.QR_START:
61 lines.insert(3,"QR:%dx%d" % cfg.QR_CAPTURE_SIZE if cfg.QR_HIGH_RES_ENABLED else "QR:640x480")
62 selection=getattr(manager.factory,"selection",None)
63 if selection:lines.append("任务码:"+selection.get("code","已缓存"))
64 last=manager.last_report
65 if last is not None:
66 obs=last.get("observation") or {}
67 lines.extend(["上次结果:",protocol.NAMES.get(last["request"]["cmd"],"--"),
68 "状态:%d" % last["status"],
69 "序号:%d" % (obs.get("target_slot",obs.get("slot_id")) or 0)])
70 point=obs.get("position_tool_mm")
71 if point is not None:
72 lines.extend(["%s:%+.1f mm" % (axis,value) for axis,value in zip("XYZ",point)])
73 lines.append("散布:%.2f mm" % obs.get("scatter_mm",0.0))
74 gate=None
75 if cmd==protocol.BOMB_SCAN:
76 from config import SCAN_GATE_RAW_PX
77 gate=SCAN_GATE_RAW_PX
78 elif cmd==protocol.AIM_SCAN:
79 from aim_config import SCAN_GATE_RAW_PX
80 gate=SCAN_GATE_RAW_PX
81 elif cmd==protocol.RESCUE_SCAN:
82 from rescue_config import SCAN_GATE_RAW_PX
83 gate=SCAN_GATE_RAW_PX
84 renderer(image,{"lines":lines,"scan_gate":gate})

87def run_loop(link,manager,camera,max_loops=None,exitpoint=None,sleep_ms=None,control=None):
88 loops,last_preview=0,ticks_ms()
89 frame=None
90 while max_loops is None or loops<max_loops:
91 loops+=1
92 if exitpoint is not None: exitpoint()
93 if control is not None:
94 control.poll()
95 if not control.running:
96 if ticks_diff(ticks_ms(),last_preview)>=cfg.DISPLAY_REFRESH_MS or control.error:
97 debug_preview(camera,manager,control)
98 last_preview=ticks_ms()
99 if control.error or not link.healthy(): break
100 if sleep_ms is not None: sleep_ms(cfg.LOOP_SLEEP_MS)
101 continue
102 pump_commands(link,manager)
103 manager.tick()
104 frame=None
105 capture_generation=control.generation if control is not None else None
106 capture_request_key=manager.request["key"] if manager.request is not None else None
107 try:
108 # Keep capture warmed even in IDLE; no inference or result packets in IDLE.
109 if manager.state!=REPORT and getattr(manager.factory,"camera_enabled",True):
110 select_qr=getattr(camera,"set_qr_capture",None)
111 if select_qr is not None:
112 select_qr(manager.request is not None and manager.request["cmd"]==protocol.QR_START)
113 frame=camera.get_frame()
114 manager.observe_camera_epoch(camera.epoch)
115 if camera.state=="ERROR": manager.fail_active("CAMERA_ERROR")
116 except Exception:
117 manager.fail_active("CAMERA_ERROR")
118 if control is not None:
119 control.poll()
120 if not control.running or control.generation!=capture_generation:
121 frame=None
122 continue
123 pump_commands(link,manager)
124 manager.tick()
125 current_key=manager.request["key"] if manager.request is not None else None
126 if frame is not None and capture_request_key==current_key:
127 manager.poll_frame(frame)
128 # A STOP/new command received during snapshot or inference wins before report.
129 if control is not None:
130 control.poll()
131 if not control.running or control.generation!=capture_generation:
132 frame=None
133 continue
134 pump_commands(link,manager)
135 manager.tick()
136 manager.flush_report()
137 if not link.healthy():
138 print("K230_LINK_ERROR",link.diagnostics())
139 break
140 if ticks_diff(ticks_ms(),last_preview)>=cfg.DISPLAY_REFRESH_MS:
141 debug_preview(camera,manager,control,frame["image"] if frame is not None else None)
142 last_preview=ticks_ms()
143 frame=None
144 gc.collect()
145 if sleep_ms is not None: sleep_ms(cfg.LOOP_SLEEP_MS)

148def main(max_loops=None):
149 link=factory=camera=manager=button=control=None
150 try:
151 from vision_link import VisionLink
152 link=VisionLink(uart_id=cfg.UART_ID,baudrate=cfg.UART_BAUDRATE,tx_pin=cfg.UART_TX_PIN,
153 rx_pin=cfg.UART_RX_PIN,queue_size=cfg.COMMAND_QUEUE_SIZE,
154 priority_cmds=(protocol.STOP,),read_chunk=cfg.RX_READ_BYTES)
155 try:
156 calibration=load_calibration(cfg.PROJECT_ROOT,handeye_accepted_override=cfg.HANDEYE_ACCEPTED_BY_USER)
157 factory=VisionTaskFactory(calibration,cfg.PROJECT_ROOT)
158 except (OSError,ValueError,KeyError) as exc:
159 print("CALIBRATION_NOT_READY",str(exc))
160 factory=UnavailableFactory()
161 camera=CameraService(qr_enabled=cfg.QR_HIGH_RES_ENABLED)
162 manager=TaskManager(link,factory)
163 button=UserButton()
164 control=RunControl(link,manager,button)
165 print("K230_RUNTIME","WAIT_START","GPIO",cfg.USER_KEY_GPIO,"UART2 115200",
166 "mean_samples",cfg.MEASUREMENT_SAMPLES)
167 print("QR_CAPTURE_SIZE",cfg.QR_CAPTURE_SIZE if cfg.QR_HIGH_RES_ENABLED else (640,480),
168 "RGB_PREVIEW_YOLO_SIZE",(640,480))
169 debug_preview(camera,manager,control)
170 run_loop(link,manager,camera,max_loops,os.exitpoint,time.sleep_ms,control)
171 except KeyboardInterrupt:
172 print("K230_RUNTIME_STOPPED")
173 finally:
174 try:
175 if control is not None: control.close()
176 finally:
177 try:
178 if manager is not None: manager.close()
179 finally:
180 try:
181 if button is not None and not button.close(): print("BUTTON_SHUTDOWN_TIMEOUT")
182 finally:
183 try:
184 if factory is not None: factory.close()
185 finally:
186 try:
187 if camera is not None: camera.close()
188 finally:
189 if link is not None and not link.deinit(): print("UART_SHUTDOWN_TIMEOUT")
