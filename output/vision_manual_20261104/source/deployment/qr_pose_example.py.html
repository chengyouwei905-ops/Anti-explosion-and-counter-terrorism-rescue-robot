1"""K230 / GC2093 最小相机示例；在 CanMV 中运行，导入本文件不会启相机。"""
2import gc
3import os
4import sys
5import time
6from media.sensor import Sensor
7from media.display import Display
8from media.media import MediaManager, CAM_CHN_ID_0
9# VS Code 的“运行当前文件”通过 <stdin> 执行，不保证当前目录在 SD 卡。
10if "/data/guanggongsai" not in sys.path:
11 sys.path.insert(0,"/data/guanggongsai")
12from qr_pose import QRPoseEstimator, CAMERA_MATRIX, DIST_COEFFS, IMAGE_SIZE
13
14# 必须填写实测编码正方形边长（mm），不能包含周围白色静区。
15QR_SIZE_MM = 80.0 # 用户确认的实测编码边长，不含白色静区。
16TARGET_PAYLOAD = None # 例如 "target-01"；None 表示逐个返回全部普通 QR。
17
18
19def main(max_frames=None):
20 # 未填有效尺寸会在分配任何相机/显示资源前明确报错。
21 estimator = QRPoseEstimator(QR_SIZE_MM, CAMERA_MATRIX, DIST_COEFFS,
22 IMAGE_SIZE, detector_order="canmv_quirc")
23 sensor = raw = gray = preview = None
24 display_touched = media_touched = False
25 frames = detections = valid_poses = 0
26 last_status = time.ticks_ms()
27 try:
28 # 沿用本工程标定采集的初始化：id=0、默认传感器模式、VGA、RGB888。
29 # 不额外设置 hmirror/vflip/crop/传感器输入尺寸，保持标定时的默认值。
30 # 若另有标定采集配置，必须把这里改成同一配置，尺寸相同并不足够。
31 sensor = Sensor(id=0)
32 sensor.reset()
33 sensor.set_framesize(Sensor.VGA, chn=CAM_CHN_ID_0)
34 sensor.set_pixformat(Sensor.RGB888, chn=CAM_CHN_ID_0)
35 display_touched = True
36 Display.init(Display.ST7701, width=800, height=480, to_ide=False)
37 media_touched = True
38 MediaManager.init()
39 sensor.run()
40 for _ in range(10):
41 os.exitpoint()
42 raw = sensor.snapshot(chn=CAM_CHN_ID_0)
43 print("QR_POSE_READY", "640x480 raw", "qr_size_mm", QR_SIZE_MM)
44 while True:
45 os.exitpoint()
46 raw = sensor.snapshot(chn=CAM_CHN_ID_0)
47 # 仅转换颜色；不去畸变、不缩放、不裁剪、不镜像。
48 gray = raw.to_grayscale()
49 results = estimator.detect(gray, target_payload=TARGET_PAYLOAD,
50 geometry_confirmed=True)
51 detections += len(results)
52 for result in results:
53 if result["ok"]:
54 valid_poses += 1
55 print("QR", result["payload"], "t_C_from_Q_mm", result["t_camera_from_qr_mm"])
56 print("depth_mm", result["depth_mm"], "distance_mm", result["distance_mm"],
57 "reprojection_RMS_px", result["reprojection_error_px"])
58 else:
59 print("QR invalid", result)
60 # 用四个角画边及索引，便于检查物理方向；不使用 rect()。
61 corners = result.get("corners_px")
62 if corners is not None and len(corners) == 4:
63 for i in range(4):
64 x, y = int(corners[i][0]), int(corners[i][1])
65 nx, ny = int(corners[(i+1)%4][0]), int(corners[(i+1)%4][1])
66 gray.draw_line(x, y, nx, ny, color=255, thickness=2)
67 gray.draw_string_advanced(x, y, 16, str(i), color=255)
68 preview=gray.to_rgb565()
69 Display.show_image(preview)
70 # 颜色转换产生临时图，及时回收，避免堆增长。
71 raw = gray = preview = None
72 gc.collect()
73 frames += 1
74 now = time.ticks_ms()
75 if time.ticks_diff(now, last_status) >= 2000:
76 print("QR_POSE_ALIVE", "frames", frames, "detections", detections,
77 "valid_poses", valid_poses, "heap_free", gc.mem_free())
78 last_status = now
79 if max_frames is not None and frames >= max_frames:
80 print("QR_POSE_RUN_COMPLETE", frames, detections, valid_poses)
81 break
82 except KeyboardInterrupt:
83 print("用户停止")
84 finally:
85 # 嵌套 finally 保证某个释放失败时仍尝试其余资源释放。
86 raw = gray = preview = None
87 try:
88 if sensor is not None:
89 sensor.stop()
90 finally:
91 try:
92 if display_touched:
93 Display.deinit()
94 finally:
95 os.exitpoint(os.EXITPOINT_ENABLE_SLEEP)
96 time.sleep_ms(100)
97 gc.collect()
98 if media_touched:
99 MediaManager.deinit()
100 print("QR_POSE_RESOURCES_RELEASED")
101
102
103if __name__ == "__main__":
104 main()
