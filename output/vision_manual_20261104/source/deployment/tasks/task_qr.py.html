7def parse_task(payload):
8 # Same strict semantics as the verified recognition/task01_qr.py parser.
9 if not isinstance(payload, str) or len(payload) != 3 or any(c not in "123" for c in payload):
10 raise ValueError("INVALID_TASK_CODE")
11 a, b, c = [int(v) for v in payload]
12 return {"code": payload, "bomb_color_id": a, "target_color_id": b, "rescue_shape_id": c,
13 "bomb_color": COLOR_NAMES[a], "target_color": COLOR_NAMES[b],
14 "rescue_shape": (None, "cylinder", "cone", "drum")[c]}

17class QRTask:
18 def __init__(self, calibration, high_resolution_size=None, contrast_retry=False,
19 contrast_retry_every=3):
20 if type(contrast_retry_every) is not int or contrast_retry_every<1:
21 raise ValueError("INVALID_QR_CONTRAST_RETRY_INTERVAL")
22 self.calibration = calibration
23 self.contrast_retry=bool(contrast_retry)
24 self.contrast_retry_every=contrast_retry_every
25 self.missed_detections=0
26 c = calibration.camera
27 self.estimator = QRPoseEstimator(QR_SIZE_MM, c.K, c.D, c.image_size, detector_order="canmv_quirc",
28 max_center_disagreement_mm=QR_MAX_CENTER_DISAGREEMENT_MM,
29 max_position_uncertainty_mm=QR_MAX_POSITION_3SIGMA_MM)
30 self.high_resolution_estimator = None
31 if high_resolution_size is not None:
32 width,height=high_resolution_size
33 if type(width) is not int or type(height) is not int or min(width,height)<=0:
34 raise ValueError("INVALID_QR_CAPTURE_SIZE")
35 sx,sy=width/float(c.image_size[0]),height/float(c.image_size[1])
36 # 同一完整视场、无额外裁剪/翻转：K_new=diag(sx,sy,1)*K。
37 # 640x480 -> 1920x1080为sx=3、sy=2.25，不能全部简单乘3。
38 # 仅推导二维码内参，不修改calibration.camera、D或手眼矩阵。
39 scaled_K=[[value*sx for value in c.K[0]],
40 [value*sy for value in c.K[1]],list(c.K[2])]
41 self.high_resolution_estimator=QRPoseEstimator(
42 QR_SIZE_MM,scaled_K,c.D,(width,height),detector_order="canmv_quirc",
43 max_center_disagreement_mm=QR_MAX_CENTER_DISAGREEMENT_MM,
44 max_position_uncertainty_mm=QR_MAX_POSITION_3SIGMA_MM)
45 self.last_frame = -1

47 def measure(self, raw_gray, frame_id, geometry_confirmed=False, stable=False):
48 base = {"ok": False, "frame_id": frame_id, "decoded": False,
49 "task_valid": False, "pose_valid": False, "control_ready": False,
50 "position_tool_mm": None}
51 if type(frame_id) is not int or frame_id <= self.last_frame:
52 base["reason"] = "STALE_OR_DUPLICATE_FRAME"
53 return base
54 self.last_frame = frame_id
55 if not stable:
56 base["reason"] = "OBSERVATION_NOT_STABLE"
57 return base
58 try:
59 estimator=self.estimator
60 if self.high_resolution_estimator is not None:
61 size=(raw_gray.width(),raw_gray.height())
62 if size==self.high_resolution_estimator.image_size:
63 estimator=self.high_resolution_estimator
64 results = estimator.detect(raw_gray, geometry_confirmed=geometry_confirmed)
65 except Exception as exc:
66 base["reason"] = "QR_DETECTION_ERROR:"+str(exc)
67 return base
68 detection_mode="raw"
69 if results:
70 self.missed_detections=0
71 else:
72 self.missed_detections+=1
73 if (self.contrast_retry and geometry_confirmed and
74 self.missed_detections>=self.contrast_retry_every):
75 self.missed_detections=0
76 enhanced=None
77 try:
78 # Change intensities only: full-size copy, no crop/resize or
79 # lens correction, so original calibrated pixel geometry holds.
80 enhanced=raw_gray.copy()
81 enhanced.histeq(adaptive=False)
82 results=estimator.detect(enhanced,geometry_confirmed=geometry_confirmed)
83 detection_mode="contrast"
84 except Exception as exc:
85 # Optional enhancement failure must not break raw detection.
86 base["contrast_retry_error"]=str(exc)
87 finally:
88 enhanced=None
89 observations = []
90 for result in results:
91 obs = dict(base)
92 obs.update(decoded=True, payload=result["payload"], pose_valid=result["ok"],
93 detection_image_size=estimator.image_size,detection_mode=detection_mode)
94 try:
95 obs["task"] = parse_task(result["payload"])
96 obs["task_valid"] = True
97 except ValueError:
98 obs["task"] = None
99 obs["pose_reason"] = result.get("reason", "OK")
100 obs["reason"] = "INVALID_TASK_CODE" if not obs["task_valid"] else obs["pose_reason"]
101 if result["ok"]:
102 transformed = self.calibration.point_to_tool(result["position_camera_mm"], REPORT_CANDIDATE_TRANSFORMS)
103 obs.update(position_tool_mm=transformed["position_tool_mm"],
104 position_camera_mm=result["position_camera_mm"],
105 position_3sigma_mm=result["position_3sigma_mm"],
106 center_disagreement_mm=result["center_disagreement_mm"],
107 reprojection_error_px=result["reprojection_error_px"],
108 coordinate_frame="E_commanded_gripper_center", units="mm",
109 handeye_accepted=self.calibration.handeye_accepted)
110 obs["ok"] = obs["task_valid"] and transformed["ok"]
111 obs["control_ready"] = obs["ok"] and transformed["control_ready"]
112 if obs["task_valid"]:
113 obs["reason"] = transformed["reason"]
114 observations.append(obs)
115 if len(observations) == 1:
116 return observations[0]
117 base.update(reason="QR_NOT_FOUND" if not observations else "MULTIPLE_QRS_AMBIGUOUS",
118 decoded=bool(observations), observations=observations)
119 return base
