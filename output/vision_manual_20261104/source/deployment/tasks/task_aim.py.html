18def measure_target_from_contours(calibration, outer_points_px, inner_points_px,
19 color_id, frame_id, geometry_confirmed=False,
20 stable=False, source="external_contours"):
21 if not valid_color(color_id):
22 return _error("INVALID_COLOR_ID", frame_id)
23 if not geometry_confirmed:
24 return _error("RAW_GEOMETRY_NOT_CONFIRMED", frame_id)
25 if not stable:
26 return _error("OBSERVATION_NOT_STABLE", frame_id)
27 result = fit_target_center(calibration.camera, outer_points_px, inner_points_px,
28 settings.TARGET_OUTER_DIAMETER_MM, settings.TARGET_INNER_DIAMETER_MM)
29 if not result["ok"]:
30 return _error(result["reason"], frame_id)
31 transformed = calibration.point_to_tool(result["position_camera_mm"], REPORT_CANDIDATE_TRANSFORMS)
32 result.update(frame_id=frame_id, color_id=color_id, color=COLOR_NAMES[color_id],
33 geometry_valid=True, geometry_id=GEOMETRY_ID,
34 position_tool_mm=transformed["position_tool_mm"],
35 handeye_accepted=calibration.handeye_accepted,
36 coordinate_frame="E_commanded_gripper_center", units="mm",
37 laser_offset_compensated=False, hit_confirmed=False, source=source,
38 control_ready=transformed["control_ready"] and source in ("color_target", "yolo_target"))
39 result["ok"], result["reason"] = transformed["ok"], transformed["reason"]
40 return result

52class AimTask:
53 def __init__(self, calibration, detector=None, measure_hits=3, max_scatter_mm=6.0):
54 if type(measure_hits) is not int or not 1 <= measure_hits <= 10 or not finite(max_scatter_mm) or max_scatter_mm <= 0:
55 raise ValueError("INVALID_MEASUREMENT_STABILITY")

69 def start_measure(self, color_id, slot_id=None, max_frames=180):

86 def poll(self, raw, frame_id, geometry_confirmed=False, stable=False):
87 if self.mode not in ("SCAN", "MEASURE"):
88 return _error("TASK_NOT_ACTIVE", frame_id)
89 if type(frame_id) is not int or frame_id <= self.last_frame:
90 return _error("STALE_OR_DUPLICATE_FRAME", frame_id)
91 previous = self.last_frame
92 self.last_frame, self.frames = frame_id, self.frames+1
93 if self.frames > self.max_frames:
94 self.mode, self.samples = "ERROR", []
95 return _error("TARGET_TASK_FRAME_BUDGET_EXCEEDED", frame_id)
96 if not geometry_confirmed or (raw.width(), raw.height()) != IMAGE_SIZE:
97 if self.mode == "SCAN": self.scan.drop_frame("RAW_GEOMETRY_NOT_CONFIRMED")
98 return self._invalid("RAW_GEOMETRY_NOT_CONFIRMED", frame_id)
99 if self.mode == "MEASURE" and not stable:
100 return self._invalid("OBSERVATION_NOT_STABLE", frame_id)
101 try:
102 detection = self.detector.detect(raw)
103 except Exception as exc:
104 if self.mode == "SCAN": self.scan.drop_frame("TARGET_DETECTOR_ERROR:"+str(exc))
105 return self._invalid("TARGET_DETECTOR_ERROR:"+str(exc), frame_id)
106 if not detection["ok"]:
107 if self.mode == "SCAN": self.scan.drop_frame(detection["reason"])
108 return self._invalid(detection["reason"], frame_id)
109 if self.mode == "SCAN":
110 result = self.scan.update(detection["detections"], frame_id)
111 result.update(frame_id=frame_id, task="aim", laser_offset_compensated=False)
112 if result["complete"]: self.mode = "DONE"
113 return result
114 matches = [d for d in detection["detections"] if d["color_id"] == self.target_color]
115 if not matches:
116 reasons = [r["reason"] for r in detection.get("rejected", []) if r.get("color_id") == self.target_color]
117 return self._invalid(reasons[0] if reasons else "TARGET_NOT_FOUND", frame_id)
118 x,y,w,h=settings.TARGET_DETECTION_ROI_RAW_PX
119 matches=[item for item in matches if
120 x<=item["bbox_raw_px"][0]+item["bbox_raw_px"][2]*.5<x+w and
121 y<=item["bbox_raw_px"][1]+item["bbox_raw_px"][3]*.5<y+h]
122 if not matches:
123 return self._invalid("TARGET_OUTSIDE_DETECTION_ROI",frame_id)
124 if len(matches) != 1:
125 return self._invalid("MULTIPLE_SAME_COLOR_TARGETS", frame_id)
126 target = matches[0]
127 if boundary_checks() and target.get("bbox_clipped", False):
128 return self._invalid("TARGET_CONTOUR_CLIPPED", frame_id)
129 contours = target.get("contours")
130 if contours is None:
131 rgb = raw.to_rgb565()
132 try:
133 contours = extract_target_rings(rgb, target["bbox_raw_px"], self.target_color)
134 finally:
135 rgb = None
136 if not contours["ok"]:
137 return self._invalid(contours["reason"], frame_id)
138 result = measure_target_from_contours(self.calibration, contours["outer_points_px"],
139 contours["inner_points_px"], self.target_color,
140 frame_id, True, True, detection.get("source", "unknown"))
141 if not result["ok"]:
142 self.samples = []
143 return result
144 if previous >= 0 and frame_id != previous+1:
145 self.samples = []
146 self.samples.append((frame_id, result["position_camera_mm"]))
147 self.samples = self.samples[-self.measure_hits:]
148 mean = [sum(s[1][i] for s in self.samples)/len(self.samples) for i in range(3)]
149 scatter = max(norm([s[1][i]-mean[i] for i in range(3)]) for s in self.samples)
150 if scatter > self.max_scatter_mm:
151 self.samples = self.samples[-1:]
152 return _error("TARGET_POSITION_NOT_STABLE", frame_id)
153 if len(self.samples) < self.measure_hits:
154 return _error("TARGET_MEASUREMENT_STABILIZING", frame_id)
155 result["position_camera_mm"] = mean
156 result["position_tool_mm"] = self.calibration.point_to_tool(mean, REPORT_CANDIDATE_TRANSFORMS)["position_tool_mm"]
157 result.update(slot_id=self.slot_id, sample_frame_ids=[s[0] for s in self.samples],
158 measurement_scatter_mm=scatter, measurement_complete=True)
159 self.samples, self.mode = [], "DONE"
160 return result
