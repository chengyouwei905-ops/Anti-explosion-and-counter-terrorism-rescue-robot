24class RescueScanSession:
25 """Reuse three-identity scan rules; color names never leak into rescue output."""
26 def __init__(self,target_shape_id,scan_order_confirmed=False,start_boundary_confirmed=False,gate=None):
27 if not valid_shape(target_shape_id): raise ValueError("INVALID_SHAPE_ID")
28 self.session = BombScanSession(target_shape_id,scan_order_confirmed,start_boundary_confirmed,
29 gate or settings.SCAN_GATE_RAW_PX,
30 settings.SCAN_MIN_HITS,settings.SCAN_RELEASE_FRAMES)
31
32 def _rename(self,result):
33 return {"ok":result["ok"],"reason":result["reason"],"complete":result["complete"],
34 "observed_shape_order":result["observed_order"],
35 "shape_to_slot":result["color_to_slot"],"target_slot":result["target_slot"]}
36
37 def update(self,detections,frame_id):
38 converted=[]
39 for item in detections:
40 if not valid_shape(item["shape_id"]):
41 self.session.failed=True
42 return self._rename(self.session._state("INVALID_SHAPE_ID"))
43 converted.append({"color_id":item["shape_id"],"bbox_raw_px":item["bbox_raw_px"]})
44 return self._rename(self.session.update(converted,frame_id))
45
46 def drop_frame(self,reason):
47 self.session.drop_frame(reason)
48
49 def fail(self):
50 self.session.drop_frame("RESCUE_DETECTION_FRAME_INVALID")
51
52 def finish(self):
53 return self._rename(self.session.finish())

56def measure_rescue_reference(calibration,layout,table_pose,shape_id,slot_id,bbox_raw_px,
57 frame_id,geometry_confirmed=False,stable=False,
58 source="external_detections"):
59 """A table-plane slot center is a fixed landmark, not a bbox/object center."""
60 if not valid_shape(shape_id): return _error("INVALID_SHAPE_ID",frame_id)
61 if not geometry_confirmed: return _error("RAW_GEOMETRY_NOT_CONFIRMED",frame_id)
62 if not stable: return _error("OBSERVATION_NOT_STABLE",frame_id)
63 reason=layout.ready_reason()
64 if reason != "OK": return _error(reason,frame_id)
65 if not table_pose.get("ok"): return _error(table_pose.get("reason","TABLE_REFERENCE_NOT_READY"),frame_id)
66 if table_pose.get("frame_id") != frame_id: return _error("TABLE_FRAME_MISMATCH",frame_id)
67 if any(abs(table_pose.get(key,0)-layout.data[key]) > 1e-6
68 for key in ("table_width_mm","table_depth_mm")):
69 return _error("TABLE_GEOMETRY_MISMATCH",frame_id)
70 try:
71 if len(bbox_raw_px) != 4 or not all(finite(v) for v in bbox_raw_px) or min(bbox_raw_px[2:]) <= 0:
72 raise ValueError("INVALID_DETECTION_BOX")
73 x,y,w,h=bbox_raw_px
74 if boundary_checks() and (x < 0 or y < 0 or x+w > IMAGE_SIZE[0] or y+h > IMAGE_SIZE[1]):
75 raise ValueError("RESCUE_TARGET_CLIPPED")
76 table_point=layout.slot_point(slot_id)
77 measured=project_table_point(calibration.camera,table_pose,table_point)
78 margin=settings.SLOT_ASSOCIATION_MARGIN_PX
79 matching=[]
80 for slot in (1,2,3):
81 p=transform_point(table_pose["T_camera_from_table"],layout.slot_point(slot))
82 u,v=calibration.camera.project_ray(p)
83 if x-margin <= u <= x+w+margin and y-margin <= v <= y+h+margin:
84 matching.append(slot)
85 if matching != [slot_id]:
86 raise ValueError("SLOT_TARGET_ASSOCIATION_FAILED")
87 transformed=calibration.point_to_tool(measured["position_camera_mm"],REPORT_CANDIDATE_TRANSFORMS)
88 result={"ok":transformed["ok"],"reason":transformed["reason"],"frame_id":frame_id,
89 "shape_id":shape_id,"shape":settings.SHAPE_NAMES[shape_id],"slot_id":slot_id,
90 "position_camera_mm":measured["position_camera_mm"],
91 "position_tool_mm":transformed["position_tool_mm"],
92 "reference_point":"table_slot_center_at_table_surface","reference_is_fixed":True,
93 "coordinate_frame":"E_commanded_gripper_center","units":"mm",
94 "orientation_valid":False,"handeye_accepted":calibration.handeye_accepted,
95 "geometry_valid":True,"geometry_id":GEOMETRY_ID,"source":source,
96 "control_ready":transformed["control_ready"] and source=="yolo_rescue"
97 and table_pose.get("source") in ("native_table_rectangle","verified_table_features"),
98 "T_camera_from_table":table_pose["T_camera_from_table"],
99 "table_pose_frame_id":frame_id,"quality":dict(table_pose["quality"]),
100 "grasp_position_camera_mm":None,"grasp_position_tool_mm":None,
101 "grasp_reference_point":None,"grasp_reason":"GRASP_HEIGHT_NOT_CONFIGURED"}
102 result["quality"]["reference_position_3sigma_mm"]=measured["position_3sigma_mm"]
103 try:
104 grasp_point=layout.grasp_point(slot_id)
105 if grasp_point is not None:
106 grasp=project_table_point(calibration.camera,table_pose,grasp_point)
107 converted=calibration.point_to_tool(grasp["position_camera_mm"],REPORT_CANDIDATE_TRANSFORMS)
108 result.update(grasp_position_camera_mm=grasp["position_camera_mm"],
109 grasp_position_tool_mm=converted["position_tool_mm"],
110 grasp_reference_point="configured_height_on_slot_axis",
111 grasp_reason=converted["reason"])
112 except (ValueError,KeyError) as exc:
113 result["grasp_reason"]=str(exc)
114 return result
115 except (ValueError,TypeError,IndexError,KeyError,ZeroDivisionError) as exc:
116 return _error(str(exc),frame_id)

119class RescueTask:
120 def __init__(self,calibration,detector,layout,table_reference=None,
121 measure_hits=settings.MEASURE_MIN_HITS,max_scatter_mm=settings.MAX_SCATTER_MM):
122 if type(measure_hits) is not int or not 1 <= measure_hits <= 10 or not finite(max_scatter_mm) or max_scatter_mm <= 0:
123 raise ValueError("INVALID_MEASUREMENT_STABILITY")
124 self.calibration,self.detector,self.layout=calibration,detector,layout
125 self.table_reference=table_reference
126 self.measure_hits,self.max_scatter_mm=measure_hits,max_scatter_mm
127 self.mode,self.scan,self.mapping,self.samples="IDLE",None,None,[]
128 self.last_frame,self.frames,self.max_frames=-1,0,0
129 self.target_shape,self.slot_id=None,None
130
131 def _budget(self,max_frames):
132 if type(max_frames) is not int or max_frames < 1: raise ValueError("INVALID_FRAME_BUDGET")
133 self.max_frames=max_frames
134
135 def start_scan(self,target_shape_id,scan_order_confirmed=False,start_boundary_confirmed=False,
136 gate=None,max_frames=2400):
137 self._budget(max_frames)
138 self.scan=RescueScanSession(target_shape_id,scan_order_confirmed,start_boundary_confirmed,gate)
139 self.mode,self.mapping,self.samples="SCAN",None,[]
140 self.last_frame,self.frames=-1,0
141
142 def start_measure(self,shape_id,slot_id=None,max_frames=180):
143 if not valid_shape(shape_id): raise ValueError("INVALID_SHAPE_ID")
144 if slot_id is None and self.mapping is not None: slot_id=self.mapping.get(shape_id)
145 if not valid_shape(slot_id): raise ValueError("TARGET_SLOT_NOT_KNOWN")
146 if self.mapping is not None and self.mapping.get(shape_id) != slot_id:
147 raise ValueError("SLOT_MAPPING_CONFLICT")
148 self._budget(max_frames)
149 self.target_shape,self.slot_id=shape_id,slot_id
150 self.mode,self.scan,self.samples="MEASURE",None,[]
151 self.last_frame,self.frames=-1,0
152
153 def _invalid(self,reason,frame_id):
154 self.samples=[]
155 return _error(reason,frame_id)
156
157 def poll(self,raw,frame_id,geometry_confirmed=False,stable=False):
158 if self.mode not in ("SCAN","MEASURE"): return _error("TASK_NOT_ACTIVE",frame_id)
159 if type(frame_id) is not int or frame_id <= self.last_frame:
160 return _error("STALE_OR_DUPLICATE_FRAME",frame_id)
161 previous=self.last_frame
162 self.last_frame,self.frames=frame_id,self.frames+1
163 if self.frames > self.max_frames:
164 self.mode="ERROR"
165 return self._invalid("RESCUE_TASK_FRAME_BUDGET_EXCEEDED",frame_id)
166 if not geometry_confirmed or (raw.width(),raw.height()) != IMAGE_SIZE:
167 if self.mode=="SCAN": self.scan.fail()
168 return self._invalid("RAW_GEOMETRY_NOT_CONFIRMED",frame_id)
169 if self.mode=="MEASURE" and not stable:
170 return self._invalid("OBSERVATION_NOT_STABLE",frame_id)
171 try:
172 detection=self.detector.detect(raw)
173 if not detection["ok"]:
174 if self.mode=="SCAN": self.scan.fail()
175 return self._invalid(detection["reason"],frame_id)
176 if self.mode=="SCAN":
177 result=self.scan.update(detection["detections"],frame_id)
178 result.update(frame_id=frame_id,task="rescue")
179 if result["complete"]:
180 self.mapping=dict(result["shape_to_slot"])
181 self.mode="DONE"
182 return result
183 matches=[d for d in detection["detections"] if d["shape_id"]==self.target_shape]
184 if not matches: return self._invalid("RESCUE_TARGET_NOT_FOUND",frame_id)
185 if len(matches)!=1: return self._invalid("MULTIPLE_SAME_SHAPE_TARGETS",frame_id)
186 if boundary_checks() and matches[0].get("bbox_clipped",False): return self._invalid("RESCUE_TARGET_CLIPPED",frame_id)
187 if self.table_reference is None: return self._invalid("TABLE_REFERENCE_NOT_READY",frame_id)
188 pose=self.table_reference.observe(raw,frame_id,geometry_confirmed=True)
189 result=measure_rescue_reference(self.calibration,self.layout,pose,self.target_shape,
190 self.slot_id,matches[0]["bbox_raw_px"],frame_id,True,True,detection.get("source","unknown"))
191 except Exception as exc:
192 if self.mode=="SCAN": self.scan.fail()
193 return self._invalid("RESCUE_OBSERVATION_ERROR:"+str(exc),frame_id)
194 if not result["ok"]:
195 self.samples=[]
196 return result
197 if previous >= 0 and frame_id != previous+1: self.samples=[]
198 self.samples.append((frame_id,result["position_camera_mm"],result["grasp_position_camera_mm"]))
199 self.samples=self.samples[-self.measure_hits:]
200 mean=[sum(s[1][i] for s in self.samples)/len(self.samples) for i in range(3)]
201 scatter=max(norm([s[1][i]-mean[i] for i in range(3)]) for s in self.samples)
202 if scatter > self.max_scatter_mm:
203 self.samples=self.samples[-1:]
204 return _error("RESCUE_POSITION_NOT_STABLE",frame_id)
205 if len(self.samples)<self.measure_hits:
206 return _error("RESCUE_MEASUREMENT_STABILIZING",frame_id)
207 result["position_camera_mm"]=mean
208 result["position_tool_mm"]=self.calibration.point_to_tool(mean,REPORT_CANDIDATE_TRANSFORMS)["position_tool_mm"]
209 if all(s[2] is not None for s in self.samples):
210 grasp_mean=[sum(s[2][i] for s in self.samples)/len(self.samples) for i in range(3)]
211 grasp_scatter=max(norm([s[2][i]-grasp_mean[i] for i in range(3)]) for s in self.samples)
212 if grasp_scatter <= self.max_scatter_mm:
213 result["grasp_position_camera_mm"]=grasp_mean
214 result["grasp_position_tool_mm"]=self.calibration.point_to_tool(grasp_mean,REPORT_CANDIDATE_TRANSFORMS)["position_tool_mm"]
215 result["grasp_scatter_mm"]=grasp_scatter
216 else:
217 result.update(grasp_position_camera_mm=None,grasp_position_tool_mm=None,
218 grasp_reason="GRASP_POSITION_NOT_STABLE")
219 elif any(s[2] is not None for s in self.samples):
220 result.update(grasp_position_camera_mm=None,grasp_position_tool_mm=None,
221 grasp_reason="GRASP_MEASUREMENT_NOT_STABLE")
222 result.update(sample_frame_ids=[s[0] for s in self.samples],measurement_scatter_mm=scatter,
223 measurement_complete=True)
224 self.samples,self.mode=[],"DONE"
225 return result

227 def finish_scan(self):
228 if self.scan is None or self.mode not in ("SCAN","DONE"): return _error("SCAN_NOT_ACTIVE")
229 self.mode="DONE"
230 return self.scan.finish()
