20def _is_color(rgb, color_id):
21 threshold=config.BOMB_LAB_THRESHOLDS.get(color_id)
22 if threshold is not None:
23 return matches_lab(rgb_to_lab_reference(rgb),threshold)
24 r, g, b = [v/255.0 for v in rgb]
25 maximum, minimum = max(r, g, b), min(r, g, b)
26 delta = maximum-minimum
27 if maximum < config.COLOR_MIN_VALUE or delta <= 0 or delta/maximum < config.COLOR_MIN_SATURATION:
28 return False
29 if maximum == r:
30 hue = (60.0*(g-b)/delta) % 360.0
31 elif maximum == g:
32 hue = 60.0*(b-r)/delta+120.0
33 else:
34 hue = 60.0*(r-g)/delta+240.0
35 return any(lo <= hue <= hi for lo, hi in config.COLOR_HUE_DEG[color_id])

38def _neighbors(index, columns, rows):
39 x, y = index % columns, index // columns
40 if x:
41 yield index-1
42 if x+1 < columns:
43 yield index+1
44 if y:
45 yield index-columns
46 if y+1 < rows:
47 yield index+columns

50def extract_ball_boundary(rgb_image, bbox_raw_px, color_id, _bounds=None, _expansions=0):
51 """Color-connected outer boundary on geometry-preserving RGB565 image.
52
53 YOLO box is ROI only. Interior highlights are filled by an exterior flood
54 fill; small disconnected patches are discarded. A clipped or ambiguous
55 component is rejected, not converted into a precision bbox diameter.
56 """
57 try:
58 if not _valid_color(color_id):
59 raise ValueError("INVALID_COLOR_ID")
60 width, height = rgb_image.width(), rgb_image.height()
61 if (width, height) != config.IMAGE_SIZE:
62 raise ValueError("IMAGE_SIZE_MISMATCH")
63 if len(bbox_raw_px) != 4 or not all(finite(v) for v in bbox_raw_px) or min(bbox_raw_px[2:]) <= 0:
64 raise ValueError("INVALID_ROI")
65 x, y, w, h = bbox_raw_px
66 margin = max(5, int(max(w, h)*0.15))
67 if _bounds is None:
68 x0, y0 = max(0, int(math.floor(x))-margin), max(0, int(math.floor(y))-margin)
69 x1, y1 = min(width, int(math.ceil(x+w))+margin), min(height, int(math.ceil(y+h))+margin)
70 else:
71 x0,y0,x1,y1=_bounds
72 if x1 <= x0 or y1 <= y0:
73 raise ValueError("INVALID_ROI")
74 step = 1
75 while ((x1-x0+step-1)//step)*((y1-y0+step-1)//step) > config.CONTOUR_MAX_CELLS:
76 step += 1
77 columns, rows = (x1-x0+step-1)//step, (y1-y0+step-1)//step
78 threshold=config.BOMB_LAB_THRESHOLDS.get(color_id)
79 lab_mask=native_lab_mask(rgb_image,threshold) if threshold is not None else None
80 mask = bytearray(columns*rows)
81 for gy in range(rows):
82 for gx in range(columns):
83 pixel = (lab_mask if lab_mask is not None else rgb_image).get_pixel(
84 x0+gx*step, y0+gy*step, rgbtuple=True)
85 if pixel is None or len(pixel) != 3:
86 raise ValueError("RGB_PIXEL_API_UNSUPPORTED")
87 mask[gy*columns+gx] = int(any(pixel) if lab_mask is not None else _is_color(pixel, color_id))
88 largest, second_size = [], 0
89 for index in range(len(mask)):
90 if mask[index] != 1:
91 continue
92 queue, head = [index], 0
93 mask[index] = 2
94 while head < len(queue):
95 current = queue[head]
96 head += 1
97 for adjacent in _neighbors(current, columns, rows):
98 if mask[adjacent] == 1:
99 mask[adjacent] = 2
100 queue.append(adjacent)
101 if len(queue) > len(largest):
102 second_size, largest = max(second_size, len(largest)), queue
103 else:
104 second_size = max(second_size, len(queue))
105 if len(largest) < config.CONTOUR_MIN_PIXELS:
106 raise ValueError("COLOR_COMPONENT_TOO_SMALL")
107 if second_size > len(largest)*0.35:
108 raise ValueError("MULTIPLE_COLOR_COMPONENTS")
109 if any(i % columns in (0, columns-1) or i // columns in (0, rows-1) for i in largest):
110 bounds=(x0,y0,x1,y1)
111 expanded=expand_to_frame(bounds,width,height)
112 if config.BOMB_FULL_FRAME_SEARCH and expanded!=bounds:
113 return extract_ball_boundary(rgb_image,bbox_raw_px,color_id,expanded,_expansions+1)
114 if boundary_checks():raise ValueError("CONTOUR_CLIPPED")
115 solid = bytearray(columns*rows)
116 for i in largest:
117 solid[i] = 1
118 queue, head = [], 0
119 for i in range(len(solid)):
120 if solid[i] == 0 and (i % columns in (0, columns-1) or i // columns in (0, rows-1)):
121 solid[i] = 2
122 queue.append(i)
123 while head < len(queue):
124 current = queue[head]
125 head += 1
126 for adjacent in _neighbors(current, columns, rows):
127 if solid[adjacent] == 0:
128 solid[adjacent] = 2
129 queue.append(adjacent)
130 points = []
131 for i in largest:
132 gx, gy = i % columns, i // columns
133 px, py = x0+gx*step, y0+gy*step
134 for adjacent, dx, dy in visible_edges(i,columns,rows):
135 if solid[adjacent] == 2:
136 points.append((px+dx*step, py+dy*step))
137 if len(points) > 4096:
138 raise ValueError("CONTOUR_TOO_COMPLEX")
139 if len(points) > config.CONTOUR_MAX_POINTS:
140 points = [points[i*len(points)//config.CONTOUR_MAX_POINTS]
141 for i in range(config.CONTOUR_MAX_POINTS)]
142 return {"ok": True, "reason": "OK", "boundary_points_px": points,
143 "sample_step_px": step, "component_sample_count": len(largest),
144 "search_bounds_raw_px":(x0,y0,x1,y1),"search_expansions":_expansions}
145 except (ValueError, TypeError, IndexError, AttributeError, OverflowError) as exc:
146 return {"ok": False, "reason": str(exc), "boundary_points_px": []}

149def measure_bomb_from_boundary(calibration, boundary_points_px, color_id, frame_id,
150 geometry_confirmed=False, stable=False):
151 """Pure measurement entry for real extracted raw-pixel silhouettes or offline tests."""
152 if not _valid_color(color_id):
153 return _error("INVALID_COLOR_ID", frame_id)
154 if not geometry_confirmed:
155 return _error("RAW_GEOMETRY_NOT_CONFIRMED", frame_id)
156 if not stable:
157 return _error("OBSERVATION_NOT_STABLE", frame_id)
158 result = fit_sphere_center(calibration.camera, boundary_points_px, config.BOMB_DIAMETER_MM,
159 config.SPHERE_MAX_ERROR_PX, config.SPHERE_MIN_INLIER_RATIO,
160 config.SPHERE_MIN_COVERAGE_DEG, config.SPHERE_MIN_RADIUS_PX,
161 config.SPHERE_MAX_DISTANCE_MM)
162 if not result["ok"]:
163 return _error(result["reason"], frame_id)
164 transformed = calibration.point_to_tool(result["position_camera_mm"], config.REPORT_CANDIDATE_TRANSFORMS)
165 result.update(frame_id=frame_id, color_id=color_id, color=config.COLOR_NAMES[color_id],
166 geometry_valid=True, geometry_id=config.GEOMETRY_ID,
167 position_tool_mm=transformed["position_tool_mm"],
168 handeye_accepted=calibration.handeye_accepted,
169 control_ready=transformed["control_ready"],
170 coordinate_frame="E_commanded_gripper_center", units="mm")
171 result["ok"], result["reason"] = transformed["ok"], transformed["reason"]
172 return result

175def measure_bombs(calibration, raw, detections, frame_id, geometry_confirmed=False,
176 stable=False, source="external_detections"):
177 """Measure all unambiguous colored spheres; convert image color only once.
178
179 detections must already be in full raw image coordinates. This function
180 also allows real detections from another adapter; their source is explicit.
181 """
182 if not geometry_confirmed or (raw.width(), raw.height()) != config.IMAGE_SIZE:
183 return dict(_error("RAW_GEOMETRY_NOT_CONFIRMED", frame_id), observations=[])
184 if not stable:
185 return dict(_error("OBSERVATION_NOT_STABLE", frame_id), observations=[])
186 if not detections:
187 return dict(_error("TARGET_BOMB_NOT_FOUND", frame_id), observations=[])
188 if len(detections) > config.MAX_DETECTIONS:
189 return dict(_error("TOO_MANY_DETECTIONS", frame_id), observations=[])
190 observations = []
191 rgb = raw.to_rgb565()
192 try:
193 for detection in detections:
194 color = detection["color_id"]
195 if not _valid_color(color):
196 result = _error("INVALID_COLOR_ID", frame_id)
197 elif sum(d["color_id"] == color for d in detections) != 1:
198 result = _error("MULTIPLE_SAME_COLOR_TARGETS", frame_id)
199 elif boundary_checks() and detection.get("bbox_clipped", False) and not config.BOMB_FULL_FRAME_SEARCH:
200 result = _error("DETECTION_CLIPPED", frame_id)
201 else:
202 contour = extract_ball_boundary(rgb, detection["bbox_raw_px"], color)
203 if not contour["ok"]:
204 result = _error(contour["reason"], frame_id)
205 else:
206 result = measure_bomb_from_boundary(calibration, contour["boundary_points_px"],
207 color, frame_id, geometry_confirmed, stable)
208 result["contour_sample_step_px"] = contour["sample_step_px"]
209 result.update(color_id=color, detection_score=detection.get("score"), source=source)
210 if source != "yolo":
211 result["control_ready"] = False
212 observations.append(result)
213 finally:
214 rgb = None
215 return {"ok": all(o["ok"] for o in observations),
216 "reason": "OK" if all(o["ok"] for o in observations) else "ONE_OR_MORE_MEASUREMENTS_INVALID",
217 "frame_id": frame_id, "observations": observations,
218 "control_ready": all(o["control_ready"] for o in observations)}

221class BombScanSession:
222 """A controlled pass, from before slot 1 to after slot 3.
223
224 Each color exists once. Direction and the starting boundary are caller
225 declarations, NOT inferred from arbitrary detection-list order or time.
226 The configured gate must expose at most one sphere at a time for order counting.
227 """
228 def __init__(self, target_color_id, scan_order_confirmed=False, start_boundary_confirmed=False,
229 gate=None, min_hits=None, release_frames=None):
230 if not _valid_color(target_color_id):
231 raise ValueError("INVALID_COLOR_ID")
232 self.target = target_color_id
233 self.confirmed = scan_order_confirmed is True and start_boundary_confirmed is True
234 self.gate = tuple(gate or config.SCAN_GATE_RAW_PX)
235 self.min_hits = config.SCAN_MIN_HITS if min_hits is None else min_hits
236 self.release_frames = config.SCAN_RELEASE_FRAMES if release_frames is None else release_frames
237 if len(self.gate) != 4 or not all(finite(v) for v in self.gate) or min(self.gate[2:]) <= 0:
238 raise ValueError("INVALID_SCAN_GATE")
239 if type(self.min_hits) is not int or self.min_hits < 2 or type(self.release_frames) is not int or self.release_frames < 1:
240 raise ValueError("INVALID_SCAN_STABILITY")
241 self.order, self.candidate, self.hits, self.latched = [], None, 0, None
242 self.center, self.blank_frames, self.last_frame = None, 0, -1
243 self.failed, self.closed, self.completed = False, False, False
244 self.require_release=True # Runtime chooses ordinal scanning explicitly.
245 self.ambiguous_frames=0
246 self.ambiguous_limit=3
247 self.failure_reason=None
248 self.last_detector_error=None

250 def drop_frame(self,reason):
251 """An unusable frame breaks hits, not previously confirmed identities."""
252 self.candidate,self.hits,self.center=None,0,None
253 self.blank_frames=self.ambiguous_frames=0
254 self.last_detector_error=str(reason)

256 def fail(self,reason):
257 self.failed=True
258 self.failure_reason=str(reason)
259 return self._state(reason)

261 def _state(self, reason, complete=False):
262 mapping = {color: i+1 for i, color in enumerate(self.order)} if complete else None
263 return {"ok": complete, "reason": reason, "complete": complete,
264 "observed_order": list(self.order), "color_to_slot": mapping,
265 "target_slot": mapping[self.target] if mapping is not None else None}

267 def update(self, detections, frame_id):
268 if self.completed:
269 return self._state("SCAN_ALREADY_COMPLETE", complete=True)
270 if self.closed:
271 return self._state("SCAN_CLOSED")
272 if not self.confirmed:
273 return self._state("SCAN_ORDER_NOT_CONFIRMED")
274 if self.failed:
275 return self._state(self.failure_reason or "SCAN_ORDER_AMBIGUOUS_RESTART_REQUIRED")
276 if type(frame_id) is not int or frame_id <= self.last_frame:
277 return self._state("STALE_OR_DUPLICATE_FRAME")
278 self.last_frame = frame_id
279 x, y, w, h = self.gate
280 eligible = []
281 for det in detections:
282 box = det["bbox_raw_px"]
283 cx, cy = box[0]+box[2]*0.5, box[1]+box[3]*0.5
284 if x <= cx < x+w and y <= cy < y+h:
285 if not _valid_color(det["color_id"]):
286 return self._state("INVALID_COLOR_ID")
287 eligible.append((det["color_id"], cx, cy))
288 if not eligible:
289 self.candidate, self.hits, self.center = None, 0, None
290 self.blank_frames += 1
291 if self.blank_frames >= self.release_frames:
292 self.latched = None
293 return self._state("SCAN_WAIT_TARGET")
294 self.blank_frames = 0
295 if len(eligible) != 1:
296 self.candidate, self.hits, self.center = None, 0, None
297 self.ambiguous_frames+=1
298 if self.require_release or self.ambiguous_frames>=self.ambiguous_limit:
299 return self.fail("SCAN_MULTIPLE_TARGETS_IN_GATE")
300 return self._state("SCAN_WAIT_SINGLE_TARGET")
301 self.ambiguous_frames=0
302 color, cx, cy = eligible[0]
303 if not self.require_release:
304 if color in self.order:
305 self.candidate,self.hits,self.center=None,0,None
306 return self._state("SCAN_ALREADY_COUNTED")
307 self.latched=None
308 if self.latched is not None:
309 if color != self.latched:
310 return self.fail("SCAN_TRANSITION_WITHOUT_RELEASE")
311 return self._state("SCAN_ALREADY_COUNTED")
312 if color in self.order:
313 return self.fail("SCAN_REVISIT_ORDER_AMBIGUOUS")
314 close = self.center is not None and (cx-self.center[0])**2+(cy-self.center[1])**2 <= config.SCAN_MAX_STEP_PX**2
315 self.hits = self.hits+1 if self.candidate == color and close else 1
316 self.candidate, self.center = color, (cx, cy)
317 if self.hits >= self.min_hits:
318 self.order.append(color)
319 self.latched = color
320 self.candidate, self.hits = None, 0
321 if len(self.order) == 3:
322 self.completed = True
323 return self._state("OK", complete=True)
324 return self._state("SCAN_TARGET_RECORDED")
325 return self._state("SCAN_STABILIZING")
326
327 def finish(self):
328 # Three confirmed unique colors complete automatically: no end command.
329 if self.completed:
330 return self._state("OK", complete=True)
331 if self.closed:
332 return self._state("SCAN_CLOSED")
333 self.closed = True
334 if not self.confirmed:
335 return self._state("SCAN_ORDER_NOT_CONFIRMED")
336 if self.failed:
337 return self._state(self.failure_reason or "SCAN_ORDER_AMBIGUOUS_RESTART_REQUIRED")
338 if len(self.order) != 3 or set(self.order) != {1, 2, 3}:
339 return self._state("SCAN_INCOMPLETE")
340 return self._state("OK", complete=True)

360 def start_refine(self, color_id, slot_id=None):
361 if not _valid_color(color_id) or (slot_id is not None and not _valid_color(slot_id)):
362 raise ValueError("INVALID_TARGET")
363 self.target_color, self.slot_id = color_id, slot_id
364 self.mode, self.last_frame = "REFINE", -1

366 def poll(self, raw, frame_id, geometry_confirmed=False, stable=False):
367 if self.mode not in ("SCAN", "REFINE"):
368 return _error("TASK_NOT_ACTIVE", frame_id)
369 if type(frame_id) is not int or frame_id <= self.last_frame:
370 return _error("STALE_OR_DUPLICATE_FRAME", frame_id)
371 self.last_frame = frame_id
372 if not geometry_confirmed or (raw.width(), raw.height()) != config.IMAGE_SIZE:
373 return _error("RAW_GEOMETRY_NOT_CONFIRMED", frame_id)
374 if self.mode == "REFINE" and not stable:
375 return _error("OBSERVATION_NOT_STABLE", frame_id)
376 detection = self.detector.detect(raw)
377 if not detection["ok"]:
378 # Retry a transient detector error without losing confirmed order.
379 if self.mode == "SCAN":self.scan.drop_frame(detection["reason"])
380 return _error(detection["reason"], frame_id)
381 if self.mode == "SCAN":
382 result = self.scan.update(detection["detections"], frame_id)
383 result["frame_id"] = frame_id
384 if result["complete"]:
385 self.mode = "DONE"
386 return result
387 matches = [d for d in detection["detections"] if d["color_id"] == self.target_color]
388 if not matches:
389 return _error("TARGET_BOMB_NOT_FOUND", frame_id)
390 rx,ry,rw,rh=config.BOMB_DETECTION_ROI_RAW_PX
391 matches=[d for d in matches if
392 rx<=d["bbox_raw_px"][0]+d["bbox_raw_px"][2]*.5<rx+rw and
393 ry<=d["bbox_raw_px"][1]+d["bbox_raw_px"][3]*.5<ry+rh]
394 if not matches:
395 return _error("BOMB_TARGET_OUTSIDE_DETECTION_ROI",frame_id)
396 if len(matches) != 1:
397 return _error("MULTIPLE_SAME_COLOR_TARGETS", frame_id)
398 if boundary_checks() and matches[0].get("bbox_clipped", False) and not config.BOMB_FULL_FRAME_SEARCH:
399 return _error("DETECTION_CLIPPED", frame_id)
400 measured = measure_bombs(self.calibration, raw, matches, frame_id, geometry_confirmed,
401 stable, detection.get("source", "unknown"))
402 result = measured["observations"][0]
403 result["slot_id"] = self.slot_id
404 return result
