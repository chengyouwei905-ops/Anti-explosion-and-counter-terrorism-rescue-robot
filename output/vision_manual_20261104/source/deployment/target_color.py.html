13def _color_matches(rgb, color):
14 r, g, b = [v/255.0 for v in rgb]
15 high, low = max(r, g, b), min(r, g, b)
16 delta = high-low
17 if high < settings.MIN_VALUE or delta <= 0 or delta/high < settings.MIN_SATURATION:
18 return False
19 if high == r:
20 hue = (60*(g-b)/delta) % 360.0
21 elif high == g:
22 hue = 60*(b-r)/delta+120
23 else:
24 hue = 60*(r-g)/delta+240
25 return any(lo <= hue <= hi for lo, hi in settings.HUE_DEG[color])

28def _neighbors(i, columns, rows):
29 x, y = i % columns, i // columns
30 if x: yield i-1
31 if x+1 < columns: yield i+1
32 if y: yield i-columns
33 if y+1 < rows: yield i+columns

36def _component(mask, start, columns, rows, old, new):
37 mask[start] = new
38 queue, head = [start], 0
39 while head < len(queue):
40 current = queue[head]
41 head += 1
42 for adjacent in _neighbors(current, columns, rows):
43 if mask[adjacent] == old:
44 mask[adjacent] = new
45 queue.append(adjacent)
46 return queue

49def extract_target_rings(rgb, bbox_raw_px, color_id):
50 """Extract outside and largest enclosed non-color hole of one colored annulus.
51
52 Interior black score rings remain inside the same non-color hole. Open,
53 clipped annuli, small highlights and multiple large components are refused.
54 """
55 try:
56 if not valid_color(color_id):
57 raise ValueError("INVALID_COLOR_ID")
58 if (rgb.width(), rgb.height()) != IMAGE_SIZE:
59 raise ValueError("IMAGE_SIZE_MISMATCH")
60 if len(bbox_raw_px) != 4 or not all(finite(v) for v in bbox_raw_px) or min(bbox_raw_px[2:]) <= 0:
61 raise ValueError("INVALID_TARGET_ROI")
62 x, y, width, height = bbox_raw_px
63 margin = max(5, int(max(width, height)*0.12))
64 x0, y0 = max(0, int(math.floor(x))-margin), max(0, int(math.floor(y))-margin)
65 x1, y1 = min(IMAGE_SIZE[0], int(math.ceil(x+width))+margin), min(IMAGE_SIZE[1], int(math.ceil(y+height))+margin)
66 if x1 <= x0 or y1 <= y0:
67 raise ValueError("INVALID_TARGET_ROI")
68 step = 1
69 while ((x1-x0+step-1)//step)*((y1-y0+step-1)//step) > settings.MAX_ROI_CELLS:
70 step += 1
71 columns, rows = (x1-x0+step-1)//step, (y1-y0+step-1)//step
72 mask = bytearray(columns*rows)
73 for gy in range(rows):
74 for gx in range(columns):
75 pixel = rgb.get_pixel(x0+gx*step, y0+gy*step, rgbtuple=True)
76 if pixel is None or len(pixel) != 3:
77 raise ValueError("RGB_PIXEL_API_UNSUPPORTED")
78 mask[gy*columns+gx] = int(_color_matches(pixel, color_id))
79 largest, second = [], 0
80 for i in range(len(mask)):
81 if mask[i] == 1:
82 component = _component(mask, i, columns, rows, 1, 2)
83 if len(component) > len(largest):
84 second, largest = max(second, len(largest)), component
85 else:
86 second = max(second, len(component))
87 if len(largest) < settings.MIN_COMPONENT_SAMPLES:
88 raise ValueError("TARGET_COLOR_TOO_SMALL")
89 if second > len(largest)*0.3:
90 raise ValueError("TARGET_COLOR_COMPONENTS_AMBIGUOUS")
91 if boundary_checks() and any(i % columns in (0, columns-1) or i // columns in (0, rows-1) for i in largest):
92 raise ValueError("TARGET_CONTOUR_CLIPPED")
93 topology = bytearray(columns*rows)
94 for i in largest: topology[i] = 1
95 # Mark background connected to any ROI border as exterior=2.
96 for i in range(len(topology)):
97 if topology[i] == 0 and (i % columns in (0, columns-1) or i // columns in (0, rows-1)):
98 _component(topology, i, columns, rows, 0, 2)
99 hole, other_hole = [], 0
100 for i in range(len(topology)):
101 if topology[i] == 0:
102 component = _component(topology, i, columns, rows, 0, 3)
103 if len(component) > len(hole):
104 other_hole, hole = max(other_hole, len(hole)), component
105 else:
106 other_hole = max(other_hole, len(component))
107 if boundary_checks() and len(hole) < settings.MIN_HOLE_SAMPLES:
108 raise ValueError("TARGET_RING_OPEN_OR_NO_HOLE")
109 if boundary_checks() and other_hole > len(hole)*0.2:
110 raise ValueError("TARGET_INNER_RING_AMBIGUOUS")
111 ratio = len(hole)/float(len(hole)+len(largest))
112 if boundary_checks() and not settings.HOLE_AREA_RATIO[0] <= ratio <= settings.HOLE_AREA_RATIO[1]:
113 raise ValueError("TARGET_HOLE_SIZE_INCONSISTENT")
114 # Other small holes (highlights) do not belong to the physical 55mm rim.
115 for i in range(len(topology)):
116 if topology[i] == 3: topology[i] = 0
117 for i in hole: topology[i] = 3
118 outer, inner = [], []
119 # When a ring is cut/open, its inner non-colour region is connected
120 # to the exterior. Classify observed edge normals around the coarse
121 # center instead of inventing a closed boundary along the image cut.
122 visible_mode=(not boundary_checks() and (len(hole)<settings.MIN_HOLE_SAMPLES or
123 any(i%columns in (0,columns-1) or i//columns in (0,rows-1) for i in largest)))
124 center_x,center_y=x+width*.5,y+height*.5
125 for i in largest:
126 gx, gy = i % columns, i // columns
127 px, py = x0+gx*step, y0+gy*step
128 for adjacent,dx,dy in visible_edges(i,columns,rows):
129 if topology[adjacent]==1:continue
130 edge=(px+dx*step,py+dy*step)
131 if visible_mode:
132 radial=dx*(edge[0]-center_x)+dy*(edge[1]-center_y)
133 if radial>0:outer.append(edge)
134 elif radial<0:inner.append(edge)
135 elif topology[adjacent]==2:outer.append(edge)
136 elif topology[adjacent]==3:inner.append(edge)
137 if len(outer)+len(inner) > 4096:
138 raise ValueError("TARGET_CONTOUR_TOO_COMPLEX")
139 def capped(points):
140 if len(points) > settings.MAX_POINTS_PER_RING:
141 return [points[i*len(points)//settings.MAX_POINTS_PER_RING] for i in range(settings.MAX_POINTS_PER_RING)]
142 return points
143 outer, inner = capped(outer), capped(inner)
144 if len(outer) < 24 or len(inner) < 20:
145 raise ValueError("TARGET_RINGS_TOO_SMALL")
146 return {"ok": True, "reason": "OK", "outer_points_px": outer,
147 "inner_points_px": inner, "sample_step_px": step, "hole_area_ratio": ratio,
148 "extraction_mode":"visible_edges" if visible_mode else "closed_ring",
149 "boundary_checks_enabled":boundary_checks()}
150 except (ValueError, TypeError, IndexError, AttributeError, OverflowError) as exc:
151 return {"ok": False, "reason": str(exc), "outer_points_px": [], "inner_points_px": []}

154class ColorTargetDetector:
155 """Native CanMV find_blobs for coarse color, explicit annulus topology check."""
156 def __init__(self):
157 self.ready = True

160 self.ready = True
161 return {"ok": True, "reason": "OK"}

163 def detect(self, raw):
164 if not self.ready:
165 return {"ok": False, "reason": "TARGET_DETECTOR_CLOSED", "detections": []}
166 rgb = None
167 try:
168 if (raw.width(), raw.height()) != IMAGE_SIZE:
169 raise ValueError("IMAGE_SIZE_MISMATCH")
170 rgb = raw.to_rgb565()
171 blobs = rgb.find_blobs(settings.LAB_THRESHOLDS, pixels_threshold=40,
172 area_threshold=120, merge=False, x_stride=2, y_stride=2)
173 if len(blobs) > settings.MAX_CANDIDATES:
174 raise ValueError("TOO_MANY_TARGET_CANDIDATES")
175 detections, rejected = [], []
176 for blob in blobs:
177 code = blob.code()
178 if code not in (1, 2, 4):
179 rejected.append({"color_id": None, "reason": "TARGET_COLOR_AMBIGUOUS"})
180 continue
181 color = {1: 1, 2: 2, 4: 3}[code]
182 box = tuple(blob.rect())
183 contour = extract_target_rings(rgb, box, color)
184 if not contour["ok"]:
185 rejected.append({"color_id": color, "reason": contour["reason"]})
186 continue
187 detections.append({"color_id": color, "label": COLOR_NAMES[color]+"_target",
188 "bbox_raw_px": box, "score": 1.0,
189 "score_kind": "annulus_topology_pass", "contours": contour})
190 return {"ok": True, "reason": "OK", "source": "color_target",
191 "detections": detections, "rejected": rejected}
192 except Exception as exc:
193 return {"ok": False, "reason": "TARGET_DETECTOR_ERROR:"+str(exc), "detections": []}
194 finally:
195 rgb = None
