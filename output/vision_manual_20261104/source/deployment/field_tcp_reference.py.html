25def _vector(values,length):
26 if len(values)!=length or any(v!=v or abs(v)>=1e30 for v in values):
27 raise ValueError("INVALID_REFERENCE_VECTOR")
28 return list(values)

31def body_from_tcp(arm_pose):
32 """Mirror MCU alg_vision_transform_from_arm_pose; mm and radians."""
33 x,y,z,pitch=_vector(arm_pose,4)
34 if math.sqrt(x*x+y*y)<1e-6 or abs(pitch)>math.pi:raise ValueError("INVALID_REFERENCE_POSE")
35 yaw=math.atan2(x,y)
36 c,s,cp,sp=math.cos(yaw),math.sin(yaw),math.cos(pitch),math.sin(pitch)
37 return [[c,-s*cp,s*sp],[s,c*cp,-c*sp],[0.,sp,cp]],[-x,y,z]

40def tcp_to_body(point_tcp,arm_pose):
41 point=_vector(point_tcp,3)
42 rotation,translation=body_from_tcp(arm_pose)
43 return [translation[i]+sum(rotation[i][j]*point[j] for j in range(3)) for i in range(3)]

46def body_to_tcp(point_body,arm_pose):
47 point=_vector(point_body,3)
48 rotation,translation=body_from_tcp(arm_pose)
49 return [sum(rotation[j][i]*(point[j]-translation[j]) for j in range(3)) for i in range(3)]

52def expected_tcp(kind,arm_pose=None,body_reference=None):
53 profile=PROFILES[kind]
54 return body_to_tcp(profile["body_reference"] if body_reference is None else body_reference,
55 profile["preset_pose"] if arm_pose is None else arm_pose)

58def expected_camera(kind,calibration,arm_pose=None,body_reference=None):
59 """Predicted C coordinates for a configured target; not a measurement."""
60 point=expected_tcp(kind,arm_pose,body_reference)
61 transform=calibration.T_tool_from_camera
62 if transform is None:raise ValueError("HANDEYE_NOT_CONFIGURED")
63 return [sum(transform[j][i]*(point[j]-transform[j][3]) for j in range(3)) for i in range(3)]

66def print_reference(kind,arm_pose=None,body_reference=None,calibration=None):
67 profile=PROFILES[kind]
68 pose=profile["preset_pose"] if arm_pose is None else arm_pose
69 reference=profile["body_reference"] if body_reference is None else body_reference
70 print("FIELD_REFERENCE",kind,"configured reference, NOT measured ground truth")
71 print("ARM_POSE_FOR_REFERENCE",tuple(pose),"mm/rad; must match MCU vision_arm_pose")
72 print("EXPECTED_B_MM",tuple(reference))
73 print("EXPECTED_ARM_ORIGIN_RELATIVE_TARGET_B_MM",tuple(-v for v in reference))
74 print("EXPECTED_TCP_MM",tuple(expected_tcp(kind,pose,reference)))
75 if calibration is not None:
76 camera=expected_camera(kind,calibration,pose,reference)
77 print("PREDICTED_CAMERA_MM",tuple(camera))
78 if camera[2]<=0:
79 print("REFERENCE_VIEW_WARNING: configured target is behind the camera")
80 else:
81 width,height=calibration.camera.image_size
82 ideal=(calibration.camera.fx*camera[0]/camera[2]+calibration.camera.cx,
83 calibration.camera.fy*camera[1]/camera[2]+calibration.camera.cy)
84 print("PREDICTED_UNDISTORTED_CENTER_VGA_PX",ideal)
85 if not (0<=ideal[0]<width and 0<=ideal[1]<height):
86 # Brown distortion polynomials can fold at extreme rays. A
87 # seemingly in-image extrapolated pixel is not evidence of view.
88 print("REFERENCE_VIEW_WARNING: outside nominal calibrated view; distortion extrapolation is unreliable")
89 else:
90 pixel=calibration.camera.project_ray(camera)
91 print("PREDICTED_CENTER_VGA_PX",tuple(pixel))
92 if kind in ("aim","rescue") and camera[2]<100:
93 print("REFERENCE_VIEW_WARNING: below current 100-mm minimum measurement depth")
94 return expected_tcp(kind,pose,reference)
