8class DebouncedKey:
9 def __init__(self,pin,active=1,debounce_ms=40,clock=ticks_ms,diff=ticks_diff):
10 self.pin,self.active,self.debounce_ms=pin,active,debounce_ms
11 self.clock,self.diff=clock,diff
12 self.raw=self.stable=pin.value()
13 self.changed_at=clock()
14 self.armed=self.stable!=active
15
16 def pressed(self):
17 now,value=self.clock(),self.pin.value()
18 if value!=self.raw: self.raw,self.changed_at=value,now
19 if self.raw!=self.stable and self.diff(now,self.changed_at)>=self.debounce_ms:
20 self.stable=self.raw
21 if self.stable!=self.active: self.armed=True
22 elif self.armed:
23 self.armed=False
24 return True
25 return False

48 def sample(self):
49 if not self.key.pressed(): return
50 self._lock.acquire()
51 try:
52 if self._pending>=8:
53 self.error="BUTTON_EVENT_OVERFLOW"

58 def _worker(self):
59 try:
60 while self._running:
61 self.sample()
62 _sleep_ms(cfg.USER_KEY_POLL_MS)
63 except BaseException as exc:
64 self.error=type(exc).__name__+":"+str(exc)
65 finally:
66 self._running=False
67 self._exited=True

69 def pressed(self):
70 self._lock.acquire()
71 try:
72 if not self._pending: return False
73 self._pending-=1
74 return True
