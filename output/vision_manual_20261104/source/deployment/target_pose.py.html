12def _solve(A, b):
13 count = len(b)
14 rows = []
15 for i in range(count):
16 scale = max(abs(v) for v in A[i])
17 if not finite(scale) or scale < 1e-16:
18 raise ValueError("TARGET_GEOMETRY_UNOBSERVABLE")
19 rows.append([v/scale for v in A[i]]+[b[i]/scale])
20 for i in range(count):
21 pivot = max(range(i, count), key=lambda j: abs(rows[j][i]))
22 if abs(rows[pivot][i]) < 1e-9:
23 raise ValueError("TARGET_GEOMETRY_UNOBSERVABLE")
24 rows[i], rows[pivot] = rows[pivot], rows[i]
25 for j in range(i+1, count):
26 factor = rows[j][i]/rows[i][i]
27 for k in range(i, count+1):
28 rows[j][k] -= factor*rows[i][k]
29 result = [0.0]*count
30 for i in range(count-1, -1, -1):
31 result[i] = (rows[i][-1]-sum(rows[i][j]*result[j] for j in range(i+1, count)))/rows[i][i]
32 if not all(finite(v) for v in result):
33 raise ValueError("NONFINITE_TARGET_SOLUTION")
34 return result

37def _evaluate(parameters, samples, radius, focal, jacobian=False):
38 C = [radius*p for p in parameters[:3]]
39 a, b = parameters[3:]
40 if not settings.MIN_DEPTH_MM <= C[2] <= settings.MAX_DEPTH_MM:
41 raise ValueError("TARGET_DEPTH_OUT_OF_RANGE")
42 N = [a, b, -1.0]
43 plane_offset = dot(N, C)
44 scale = focal/C[2]
45 residuals, rows = [], []
46 for ray, known_radius, _pixel in samples:
47 denominator = dot(N, ray)
48 if denominator >= -1e-6:
49 raise ValueError("TARGET_PLANE_BEHIND_RAY")
50 distance = plane_offset/denominator
51 if distance <= 0:
52 raise ValueError("TARGET_PLANE_BEHIND_CAMERA")
53 q = [distance*ray[i]-C[i] for i in range(3)]
54 length = norm(q)
55 if length < 1e-6:
56 raise ValueError("DEGENERATE_TARGET_POINT")
57 error = (length-known_radius)*scale
58 residuals.append(error)
59 if jacobian:
60 qray = dot(q, ray)
61 row = [scale*radius*(qray*N[j]/denominator-q[j])/length for j in range(3)]
62 row[2] -= error*radius/C[2]
63 row += [scale*qray*(C[j]*denominator-plane_offset*ray[j])/(length*denominator*denominator)
64 for j in range(2)]
65 rows.append(row)
66 return residuals, rows

69def _normal_equations(residuals, rows, robust=True):
70 H = [[0.0]*5 for _ in range(5)]
71 g = [0.0]*5
72 cost = 0.0
73 for error, row in zip(residuals, rows):
74 limit = settings.MAX_REPROJECTION_PX
75 weight = min(1.0, limit/max(abs(error), 1e-9)) if robust else 1.0
76 cost += weight*error*error
77 for i in range(5):
78 g[i] += weight*row[i]*error
79 for j in range(5):
80 H[i][j] += weight*row[i]*row[j]
81 return H, g, cost

84def _cost(residuals):
85 limit = settings.MAX_REPROJECTION_PX
86 return sum(e*e if abs(e) <= limit else 2*limit*abs(e)-limit*limit for e in residuals)

89def _refine(initial, samples, radius, focal):
90 p = list(initial)
91 damping = 0.001
92 for _ in range(32):
93 residuals, rows = _evaluate(p, samples, radius, focal, True)
94 H, gradient, _ = _normal_equations(residuals, rows)
95 old_cost = _cost(residuals)
96 improved = False
97 for _ in range(7):
98 A = [[H[i][j]+(damping*max(H[i][i], 1e-6) if i == j else 0.0)
99 for j in range(5)] for i in range(5)]
100 try:
101 step = _solve(A, [-v for v in gradient])
102 candidate = [p[i]+step[i] for i in range(5)]
103 new_cost = _cost(_evaluate(candidate, samples, radius, focal)[0])
104 except (ValueError, OverflowError, ZeroDivisionError):
105 new_cost = 1e30
106 if new_cost < old_cost:
107 p, damping, improved = candidate, max(1e-8, damping/3), True
108 break
109 damping *= 8
110 if not improved or sum(v*v for v in step) < 1e-12 or old_cost-new_cost < 1e-8:
111 break
112 return p, _cost(_evaluate(p, samples, radius, focal)[0])

115def _reproject(camera, p, samples, radius):
116 C = [radius*v for v in p[:3]]
117 N = [p[3], p[4], -1.0]
118 d = dot(N, C)
119 errors, vectors = [], []
120 for ray, known_radius, pixel in samples:
121 s = d/dot(N, ray)
122 q = [s*ray[i]-C[i] for i in range(3)]
123 q = unit(q)
124 predicted = [C[i]+known_radius*q[i] for i in range(3)]
125 u, v = camera.project_ray(predicted)
126 errors.append(math.sqrt((u-pixel[0])**2+(v-pixel[1])**2))
127 vectors.append(q)
128 return errors, vectors

131def _coverage(vectors, normal):
132 ex = unit(cross(normal, [0, 1, 0] if abs(normal[1]) < 0.9 else [1, 0, 0]))
133 ey = cross(normal, ex)
134 angles = sorted(math.atan2(dot(v, ey), dot(v, ex)) for v in vectors)
135 if len(angles) < 16:
136 return 0.0
137 gaps = [angles[i+1]-angles[i] for i in range(len(angles)-1)]
138 gaps.append(angles[0]+2*math.pi-angles[-1])
139 return 360-max(gaps)*180/math.pi

142def fit_target_center(camera, outer_points_px, inner_points_px,
143 outer_diameter_mm=100.0, inner_diameter_mm=55.0):
144 """Fit common physical center + normal to both known circles.
145
146 Several normal seeds are evaluated; competing low-error centers and poor
147 local position observability cause rejection. In-plane roll is never made up.
148 """
149 try:
150 if not all(finite(v) for v in (outer_diameter_mm, inner_diameter_mm)) or not 0 < inner_diameter_mm < outer_diameter_mm:
151 raise ValueError("INVALID_TARGET_DIMENSIONS")
152 outer, inner = list(outer_points_px), list(inner_points_px)
153 if not 24 <= len(outer) <= 192 or not 20 <= len(inner) <= 192:
154 raise ValueError("TWO_COMPLETE_RINGS_REQUIRED")
155 radius, inner_radius = outer_diameter_mm/2.0, inner_diameter_mm/2.0
156 outer_rays, inner_rays = [camera.ray(p) for p in outer], [camera.ray(p) for p in inner]
157 samples = [(ray, radius, pixel) for ray, pixel in zip(outer_rays, outer)]
158 samples += [(ray, inner_radius, pixel) for ray, pixel in zip(inner_rays, inner)]
159 xs, ys = [v[0]/v[2] for v in outer_rays], [v[1]/v[2] for v in outer_rays]
160 mx, my = sum(xs)/len(xs), sum(ys)/len(ys)
161 xx = sum((x-mx)**2 for x in xs)/len(xs)
162 yy = sum((y-my)**2 for y in ys)/len(ys)
163 xy = sum((x-mx)*(y-my) for x, y in zip(xs, ys))/len(xs)
164 largest = 0.5*(xx+yy+math.sqrt((xx-yy)**2+4*xy*xy))
165 angular_radius = math.sqrt(2*largest)
166 if boundary_checks() and angular_radius*min(camera.fx, camera.fy) < settings.MIN_OUTER_RADIUS_PX:
167 raise ValueError("TARGET_TOO_SMALL")
168 z = 1/angular_radius
169 z = min(settings.MAX_DEPTH_MM/radius, max(settings.MIN_DEPTH_MM/radius, z))
170 focal = math.sqrt(camera.fx*camera.fy)
171 candidates = []
172 for a, b in ((0, 0), (1, 0), (-1, 0), (0, 1), (0, -1), (0.8, 0.8),
173 (0.8, -0.8), (-0.8, 0.8), (-0.8, -0.8)):
174 try:
175 p, cost = _refine([mx*z, my*z, z, a, b], samples, radius, focal)
176 if not any(norm([radius*(p[i]-q[0][i]) for i in range(3)]) < 0.05
177 and norm([p[i]-q[0][i] for i in (3, 4)]+[0]) < 0.001 for q in candidates):
178 candidates.append((p, cost))
179 except (ValueError, ZeroDivisionError, OverflowError):
180 pass
181 if not candidates:
182 raise ValueError("TARGET_POSE_NOT_CONVERGED")
183 candidates.sort(key=lambda item: item[1])
184 p, cost = candidates[0]
185 C = [radius*v for v in p[:3]]
186 normal = unit([p[3], p[4], -1.0])
187 competing = []
188 for other, other_cost in candidates[1:]:
189 if other_cost-cost > 9*settings.CORNER_NOISE_PX**2:
190 continue
191 delta = norm([radius*(other[i]-p[i]) for i in range(3)])
192 competing.append({"center_difference_mm": delta})
193 if delta > 3.0:
194 raise ValueError("TARGET_CENTER_AMBIGUOUS")
195 errors, vectors = _reproject(camera, p, samples, radius)
196 limit = settings.MAX_REPROJECTION_PX
197 indices = [i for i, e in enumerate(errors) if e <= limit]
198 strict=boundary_checks()
199 inlier_ratio=len(indices)/float(len(samples))
200 if strict and len(indices) < int(math.ceil(settings.MIN_INLIER_RATIO*len(samples))):
201 raise ValueError("TARGET_RINGS_INCONSISTENT")
202 if not strict:indices=list(range(len(samples)))
203 outer_inliers = [i for i in indices if i < len(outer)]
204 inner_inliers = [i for i in indices if i >= len(outer)]
205 coverage = [_coverage([vectors[i] for i in group], normal) for group in (outer_inliers, inner_inliers)]
206 if strict and min(coverage) < settings.MIN_COVERAGE_DEG:
207 raise ValueError("TARGET_RING_INCOMPLETE")
208 rms = math.sqrt(sum(errors[i]**2 for i in indices)/len(indices))
209 _, rows = _evaluate(p, [samples[i] for i in indices], radius, focal, True)
210 H, _, _ = _normal_equations([0.0]*len(rows), rows, False)
211 inverse = [_solve(H, [float(i == j) for i in range(5)]) for j in range(3)]
212 sigma = max(settings.CORNER_NOISE_PX, rms)
213 diagonal = [inverse[i][i] for i in range(3)]
214 if min(diagonal) < 0:
215 raise ValueError("TARGET_GEOMETRY_UNOBSERVABLE")
216 position_3sigma = 3*sigma*radius*math.sqrt(sum(diagonal[:3]))
217 if position_3sigma > settings.MAX_POSITION_3SIGMA_MM:
218 raise ValueError("TARGET_POSITION_POORLY_OBSERVABLE")
219 return {"ok": True, "reason": "OK", "position_camera_mm": C,
220 "orientation_valid": False,
221 "in_plane_rotation_observable": False, "reference_point": "target_physical_center",
222 "quality": {"reprojection_rms_px": rms, "inlier_ratio": inlier_ratio,
223 "boundary_checks_enabled":strict,
224 "warnings":(["TARGET_RINGS_INCONSISTENT"] if inlier_ratio<settings.MIN_INLIER_RATIO else [])+(["TARGET_RING_INCOMPLETE"] if min(coverage)<settings.MIN_COVERAGE_DEG else []),
225 "coverage_deg": coverage, "position_3sigma_mm": position_3sigma,
226 "candidate_count": len(candidates),
227 "competing": competing, "uniqueness_guaranteed": False,
228 "outer_diameter_mm": outer_diameter_mm, "inner_diameter_mm": inner_diameter_mm}}
229 except (ValueError, TypeError, IndexError, ZeroDivisionError, OverflowError) as exc:
230 return {"ok": False, "reason": str(exc), "position_camera_mm": None,
231 "orientation_valid": False}
