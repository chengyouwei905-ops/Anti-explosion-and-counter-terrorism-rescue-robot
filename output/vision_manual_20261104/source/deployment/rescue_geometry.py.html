1"""Known rectangular table -> real slot reference, reusing verified QR pose math.
2
3Only object coordinates and their initial homography/evaluation are specialized.
4Model tilt is only used to locate physical table points. No heading is exported.
5"""
6import math
7import rescue_config as settings
8from geometry import finite, transform_point
9from qr_pose import QRPoseEstimator, _solve, _orthogonal, _dot, _cross

12class RescueTablePoseEstimator(QRPoseEstimator):
13 def __init__(self, camera, width_mm, depth_mm):
14 if not finite(width_mm) or not finite(depth_mm) or not 0 < depth_mm < width_mm:
15 raise ValueError("INVALID_TABLE_DIMENSIONS")
16 self.width_mm, self.depth_mm = width_mm, depth_mm
17 half_depth = depth_mm/(2.0*width_mm)
18 # Far-left, far-right, near-right, near-left in the physical table frame.
19 self.object_corners = ((-.5, half_depth), (.5, half_depth),
20 (.5, -half_depth), (-.5, -half_depth))
21 super().__init__(width_mm, camera.K, camera.D, camera.image_size,
22 max_reprojection_error_px=settings.TABLE_REPROJECTION_PX,
23 corner_noise_px=settings.TABLE_CORNER_NOISE_PX,
24 max_position_uncertainty_mm=settings.MAX_REFERENCE_3SIGMA_MM)

26 def _homography_pose(self, uv):
27 mx, my = sum(p[0] for p in uv)/4.0, sum(p[1] for p in uv)/4.0
28 scale = math.sqrt(sum((p[0]-mx)**2+(p[1]-my)**2 for p in uv)/4.0)
29 if scale < 1e-8:
30 raise ValueError("degenerate_table_corners")
31 A, b = [], []
32 for (x, y), (u, v) in zip(self.object_corners, uv):
33 u, v = (u-mx)/scale, (v-my)/scale
34 A.append([x,y,1.0,0.0,0.0,0.0,-u*x,-u*y]); b.append(u)
35 A.append([0.0,0.0,0.0,x,y,1.0,-v*x,-v*y]); b.append(v)
36 h = _solve(A,b)+[1.0]
37 h = [scale*h[j]+mx*h[6+j] for j in range(3)]+[
38 scale*h[3+j]+my*h[6+j] for j in range(3)]+h[6:9]
39 c1,c2 = [h[0],h[3],h[6]], [h[1],h[4],h[7]]
40 lam = 2.0/(math.sqrt(_dot(c1,c1))+math.sqrt(_dot(c2,c2)))
41 R = _orthogonal([[c1[i]*lam,c2[i]*lam,0.0] for i in range(3)])
42 return R,[h[2]*lam,h[5]*lam,lam]

44 def _evaluate(self,R,t,corners,jacobian=False):
45 residual,rows = [],[]
46 for (xo,yo),(u,v) in zip(self.object_corners,corners):
47 q = [R[i][0]*xo+R[i][1]*yo for i in range(3)]
48 p = [q[i]+t[i] for i in range(3)]
49 if not all(finite(x) for x in p) or p[2] <= 1e-6:
50 raise ValueError("nonpositive_table_depth")
51 x,y = p[0]/p[2],p[1]/p[2]
52 if max(abs(x),abs(y)) > 3:
53 raise ValueError("table_projection_out_of_range")
54 xd,yd,a,b,c,d = self._distort(x,y)
55 residual.extend((u-(self.fx*xd+self.cx),v-(self.fy*yd+self.cy)))
56 if jacobian:
57 for focal,dx,dy in ((self.fx,a,b),(self.fy,c,d)):
58 g = [focal*dx/p[2],focal*dy/p[2],-focal*(dx*x+dy*y)/p[2]]
59 rows.append(_cross(q,g)+g)
60 if not all(finite(x) for x in residual):
61 raise ValueError("nonfinite_table_projection")
62 return sum(x*x for x in residual),residual,rows

64 def estimate_table(self,corners_px,geometry_confirmed=False,physical_order_confirmed=False):
65 if not geometry_confirmed:
66 return {"ok":False,"reason":"RAW_GEOMETRY_NOT_CONFIRMED","T_camera_from_table":None}
67 if not physical_order_confirmed:
68 return {"ok":False,"reason":"TABLE_CORNER_ORDER_NOT_CONFIRMED","T_camera_from_table":None}
69 pose = self.estimate_from_corners(corners_px,actual_image_size=self.image_size)
70 if not pose["ok"]:
71 return {"ok":False,"reason":"TABLE_"+pose["reason"].upper(),"T_camera_from_table":None}
72 try:
73 R,t = pose["R_camera_from_qr"],[v/self.size for v in pose["t_camera_from_qr_mm"]]
74 _,residual,rows = self._evaluate(R,t,corners_px,True)
75 A,_,scales = self._normal(rows,residual)
76 inverse = [_solve(A,[float(i==j) for i in range(6)]) for j in range(6)]
77 covariance = [[inverse[j][i]*self.noise*self.noise/(scales[i]*scales[j])
78 for j in range(6)] for i in range(6)]
79 return {"ok":True,"reason":"OK","T_camera_from_table":pose["T_camera_from_qr"],
80 "corners_px":pose["corners_px"],"table_width_mm":self.width_mm,
81 "table_depth_mm":self.depth_mm,"_pose_covariance":covariance,
82 "_candidate_models":pose["_candidate_models"],
83 "quality":{"reprojection_rms_px":pose["reprojection_error_px"],
84 "position_3sigma_mm":pose["position_3sigma_mm"],
85 "candidate_count":pose["candidate_count"],"uniqueness_guaranteed":False}}
86 except (ValueError,ZeroDivisionError,OverflowError) as exc:
87 return {"ok":False,"reason":"TABLE_"+str(exc).upper(),"T_camera_from_table":None}

90def project_table_point(camera,table_pose,point_table_mm):
91 """Point location and local assumed-noise uncertainty; no bbox triangulation."""
92 if not table_pose.get("ok"):
93 raise ValueError("TABLE_POSE_INVALID")
94 T = table_pose["T_camera_from_table"]
95 point = transform_point(T,point_table_mm)
96 for model_r,model_t in table_pose.get("_candidate_models",[]):
97 alternate=[sum(model_r[i][j]*point_table_mm[j] for j in range(3))+model_t[i]*table_pose["table_width_mm"] for i in range(3)]
98 if math.sqrt(sum((alternate[i]-point[i])**2 for i in range(3))) > settings.MAX_REFERENCE_3SIGMA_MM:
99 raise ValueError("TABLE_POINT_AMBIGUOUS")
100 if point[2] <= 0:
101 raise ValueError("TABLE_POINT_BEHIND_CAMERA")
102 scale = table_pose["table_width_mm"]
103 q = [sum(T[i][j]*point_table_mm[j] for j in range(3)) for i in range(3)]
104 x,y,z = q
105 J = [[0,z,-y,scale,0,0],[-z,0,x,0,scale,0],[y,-x,0,0,0,scale]]
106 covariance = table_pose["_pose_covariance"]
107 variance = sum(sum(J[k][i]*covariance[i][j]*J[k][j] for i in range(6) for j in range(6)) for k in range(3))
108 if not finite(variance) or variance < -1e-6:
109 raise ValueError("TABLE_POINT_UNOBSERVABLE")
110 uncertainty = 3*math.sqrt(max(0.0,variance))
111 if uncertainty > settings.MAX_REFERENCE_3SIGMA_MM:
112 raise ValueError("TABLE_POINT_POORLY_OBSERVABLE")
113 return {"position_camera_mm":point,"pixel_raw":camera.project_ray(point),
114 "position_3sigma_mm":uncertainty}
