18def terminal_reason(reason):
19 # Preserve the original message for display, but inspect nested detector
20 # failures so an actual model/shape contract error is reported immediately.
21 for _ in range(3):
22 prefix=next((item for item in ("DETECTION_ERROR:","TARGET_DETECTOR_ERROR:","QR_DETECTION_ERROR:")
23 if reason.startswith(item)),None)
24 if prefix is None:break
25 reason=reason[len(prefix):]
26 if reason in ("IMAGE_SIZE_MISMATCH","PREPROCESS_RAW_SIZE_MISMATCH","RAW_GEOMETRY_NOT_CONFIRMED"):
27 return True
28 prefixes=("MODEL_","TEST_BACKEND","HANDEYE_NOT_","CALIBRATION_","INVALID_",
29 "TABLE_DIMENSIONS_NOT_","SLOT_LAYOUT_NOT_","SLOT_AXIS_","TABLE_TOP_ROI_NOT_",
30 "TABLE_CORNER_ORDER_NOT_","INVALID_TABLE_","INVALID_SLOT_","SLOT_CENTER_OUTSIDE_",
31 "TARGET_SLOT_NOT_KNOWN","SLOT_MAPPING_CONFLICT","SCAN_ORDER_NOT_","SCAN_BOUNDARIES_NOT_",
32 "SCAN_MULTIPLE_","SCAN_TRANSITION_","SCAN_REVISIT_","SCAN_ORDER_AMBIGUOUS_",
33 "TOO_MANY_TABLE_","RESCUE_TASK_FRAME_BUDGET_","TARGET_TASK_FRAME_BUDGET_")
34 return any(reason.startswith(prefix) for prefix in prefixes)

45 def enable_debug(self):
46 if self.debug_trace is not None:return
47 from vision_debug import DetectionTrace,DetectorProbe,QRPoseProbe
48 kind={p.QR_START:"qr",p.BOMB_MEASURE:"bomb",p.AIM_MEASURE:"aim",p.RESCUE_MEASURE:"rescue"}[self.cmd]

55 self.task.detector=DetectorProbe(self.task.detector,self.debug_trace)

67 def preview(self,frame):
68 """Detection-only feedback while waiting between reports; no pose samples."""
69 if self.debug_trace is None:return
70 self._debug_begin(frame)
71 gray=enhanced=None
72 try:
73 if self.cmd==p.QR_START:
74 gray=frame.get("qr_image")
75 if gray is None:gray=frame["image"].to_grayscale()
76 codes=gray.find_qrcodes()
77 if codes:self.debug_misses=0
78 else:self.debug_misses+=1
79 if (not codes and getattr(self.task,"contrast_retry",False) and
80 self.debug_misses>=self.task.contrast_retry_every):
81 self.debug_misses=0
82 enhanced=gray.copy();enhanced.histeq(adaptive=False)
83 codes=enhanced.find_qrcodes()
84 self.debug_trace.qr_result([{"corners_px":code.corners(),"payload":code.payload()} for code in codes])
85 else:
86 self.task.detector.detect(frame["image"])
87 except Exception as exc:
88 self.debug_trace.error="PREVIEW_ERROR:"+str(exc)
89 finally:gray=enhanced=None

91 def poll(self,frame):
92 self._debug_begin(frame)
93 raw,frame_id=frame["image"],frame["frame_id"]
94 geometry=frame.get("geometry_confirmed",False)
95 if self.cmd==p.QR_START:
96 gray=frame.get("qr_image")
97 if gray is None:gray=raw.to_grayscale()
98 else:geometry=frame.get("qr_geometry_confirmed",False)
99 try: result=self.task.measure(gray,frame_id,geometry_confirmed=geometry,stable=True)
100 finally: gray=None
101 else:
102 if not self.is_scan and self.task.mode=="DONE":
103 self.owner._start_measure(self.task,self.request)
104 result=self.task.poll(raw,frame_id,geometry_confirmed=geometry,stable=not self.is_scan)
105 result=dict(result)
106 result["target_id"]=self.request.get("target_id")
107 result["slot_id"]=result.get("slot_id",self.request.get("slot_id"))
108 if self.cmd==p.QR_START: result.setdefault("reference_point","qr_center")
109 elif self.cmd==p.BOMB_MEASURE: result.setdefault("reference_point","sphere_center")
110 if self.is_scan and result.get("complete"):
111 result["slot_mapping"]=result.get("color_to_slot",result.get("shape_to_slot"))
112 result["sample_count"]=0
113 return result

134class VisionTaskFactory:
135 def __init__(self,calibration,project_root,backend_factory=None):
136 self.calibration,self.root=calibration,project_root.rstrip("/\\")
137 self.backend_factory=backend_factory or cfg.create_model_backend
138 self.detector,self.detector_kind=None,None
139 self.selection=None
140 self.maps={"bomb":None,"aim":None,"rescue":None}
141 self.reference,self.layout=None,None

143 def _json(self,relative):
144 try:
145 with open(self.root+"/"+relative,"r") as stream: return json.load(stream)
146 except (OSError,ValueError): raise ValueError("MODEL_CONFIGURATION_ERROR" if "three_" in relative else "TABLE_CONFIGURATION_ERROR")

148 def _model(self,kind):
149 if self.detector_kind==kind and self.detector is not None and self.detector.ready: return self.detector
150 if self.detector is not None: self.detector.close()
151 self.detector,self.detector_kind=None,None
152 filename="models/bomb_three_colors.template.json" if kind=="bomb" else "models/rescue_three_shapes.template.json"
153 metadata=self._json(filename)
154 if metadata.get("model_path") and not metadata["model_path"].startswith("/"):
155 metadata["model_path"]=self.root+"/"+metadata["model_path"]
156 backend=self.backend_factory(kind,metadata) if metadata.get("model_path") else None
157 detector=(YoloDetector(metadata,backend) if kind=="bomb" else RescueYoloDetector(metadata,backend))
158 loaded=detector.load()
159 if not loaded["ok"]: raise ValueError(loaded["reason"])
160 self.detector,self.detector_kind=detector,kind
161 return detector

164 if self.reference is None:

169 def _kind(self,cmd):
170 if cmd in (p.BOMB_SCAN,p.BOMB_MEASURE): return "bomb"
171 if cmd in (p.AIM_SCAN,p.AIM_MEASURE): return "aim"
172 if cmd in (p.RESCUE_SCAN,p.RESCUE_MEASURE): return "rescue"
173 return None

175 def _start_measure(self,task,request):
176 cmd,target,slot=request["cmd"],request.get("target_id"),request.get("slot_id")
177 if cmd==p.BOMB_MEASURE: task.start_refine(target,slot)
178 else: task.start_measure(target,slot,max_frames=1000000)

180 def start(self,request):
181 cmd=request["cmd"]
182 if cmd==p.QR_START:
183 self.selection=None
184 self.maps={"bomb":None,"aim":None,"rescue":None}
185 task=QRTask(self.calibration,high_resolution_size=cfg.QR_CAPTURE_SIZE if cfg.QR_HIGH_RES_ENABLED else None,
186 contrast_retry=cfg.QR_CONTRAST_RETRY_ENABLED,
187 contrast_retry_every=cfg.QR_CONTRAST_RETRY_EVERY_N_FRAMES)
188 return VisionJob(self,request,task)
189 kind=self._kind(cmd)
190 if request.get("target_id") is None:
191 key={"bomb":"bomb_color_id","aim":"target_color_id","rescue":"rescue_shape_id"}[kind]
192 if self.selection is None: raise ValueError("TARGET_NOT_SELECTED")
193 request["target_id"]=self.selection[key]
194 mapping=self.maps[kind]
195 if cmd in p.SCAN_COMMANDS:
196 self.maps[kind]=None
197 if not cfg.SCAN_ORDER_CONFIRMED or not cfg.SCAN_START_BOUNDARY_CONFIRMED:
198 raise ValueError("SCAN_NOT_CONFIGURED")
199 else:
200 selected=mapping.get(request["target_id"]) if mapping is not None else None
201 if request.get("slot_id") is None: request["slot_id"]=selected
202 elif selected is not None and selected!=request["slot_id"]: raise ValueError("SLOT_MAPPING_CONFLICT")
203 if cmd==p.RESCUE_MEASURE and cfg.RESCUE_REFERENCE_MODE=="table" and request.get("slot_id") is None: raise ValueError("TARGET_SLOT_NOT_KNOWN")
204 if kind=="bomb": task=BombTask(self.calibration,self._model("bomb"))
205 elif kind=="aim": task=AimTask(self.calibration,measure_hits=1)
206 else:
207 detector=self._model("rescue")
208 if cfg.RESCUE_REFERENCE_MODE=="object":
209 task=RescueObjectTask(self.calibration,detector)
210 else:
211 reference=self._rescue_reference()
212 if cmd==p.RESCUE_MEASURE and reference.readiness()!="OK": raise ValueError(reference.readiness())
213 task=RescueTask(self.calibration,detector,self.layout,reference,measure_hits=1)
214 if cmd in p.SCAN_COMMANDS:
215 if kind=="bomb": task.start_scan(request["target_id"],True,True)
216 else: task.start_scan(request["target_id"],True,True,max_frames=1000000)
217 tracker=getattr(task.scan,"session",task.scan)
218 tracker.require_release=cfg.SCAN_REQUIRE_BLANK_GAP
219 tracker.ambiguous_limit=cfg.SCAN_MAX_AMBIGUOUS_FRAMES
220 else: self._start_measure(task,request)
221 return VisionJob(self,request,task)

223 def completed(self,request,observation):
224 cmd=request["cmd"]
225 if cmd==p.QR_START: self.selection=observation["task"]
226 elif cmd in p.SCAN_COMMANDS: self.maps[self._kind(cmd)]=dict(observation["slot_mapping"])
