12class RescueLayout:
13 def __init__(self, data):
14 self.data = dict(data)

16 def ready_reason(self):
17 d = self.data
18 w,h = d.get("table_width_mm"),d.get("table_depth_mm")
19 if not finite(w) or not finite(h) or not 0 < h < w:
20 return "INVALID_TABLE_DIMENSIONS"
21 if d.get("dimensions_confirmed") is not True:
22 return "TABLE_DIMENSIONS_NOT_CONFIRMED"
23 if d.get("slot_layout_confirmed") is not True or d.get("slot_centers_table_mm") is None:
24 return "SLOT_LAYOUT_NOT_CONFIRMED"
25 points = d["slot_centers_table_mm"]
26 if not isinstance(points,(list,tuple)) or len(points) != 3:
27 return "INVALID_SLOT_LAYOUT"
28 for p in points:
29 if not isinstance(p,(list,tuple)) or len(p) != 3 or not all(finite(v) for v in p):
30 return "INVALID_SLOT_LAYOUT"
31 if abs(p[2]) > 1e-5 or abs(p[0]) > w/2-25 or abs(p[1]) > h/2-25:
32 return "SLOT_CENTER_OUTSIDE_TABLE"
33 if not points[0][0] < points[1][0] < points[2][0]:
34 return "SLOT_AXIS_ORDER_INVALID"
35 if d.get("slot_axis_matches_scan_order_confirmed") is not True:
36 return "SLOT_AXIS_NOT_CONFIRMED"
37 return "OK"

39 def native_ready_reason(self,has_corner_adapter=False):
40 reason = self.ready_reason()
41 if reason != "OK": return reason
42 roi = (0,0,IMAGE_SIZE[0],IMAGE_SIZE[1]) if settings.TABLE_FULL_FRAME_SEARCH else self.data.get("table_top_roi_raw_px")
43 if not isinstance(roi,(list,tuple)) or len(roi) != 4 or not all(type(v) is int for v in roi):
44 return "TABLE_TOP_ROI_NOT_CONFIGURED"
45 x,y,w,h = roi
46 if min(x,y) < 0 or min(w,h) <= 0 or x+w > IMAGE_SIZE[0] or y+h > IMAGE_SIZE[1]:
47 return "INVALID_TABLE_TOP_ROI"
48 if not has_corner_adapter and self.data.get("top_view_image_corner_order_confirmed") is not True:
49 return "TABLE_CORNER_ORDER_NOT_CONFIRMED"
50 return "OK"

52 def slot_point(self,slot_id):
53 reason = self.ready_reason()
54 if reason != "OK": raise ValueError(reason)
55 if type(slot_id) is not int or not 1 <= slot_id <= 3:
56 raise ValueError("INVALID_SLOT_ID")
57 return list(self.data["slot_centers_table_mm"][slot_id-1])

59 def grasp_point(self,slot_id):
60 point = self.slot_point(slot_id)
61 height = self.data.get("grasp_height_above_table_mm")
62 if self.data.get("grasp_height_confirmed") is not True or height is None:
63 return None
64 if not finite(height) or not 0 <= height <= 200:
65 raise ValueError("INVALID_GRASP_HEIGHT")
66 point[2] = height
67 return point

70def order_restricted_top_view(corners):
71 """Only for a physically confirmed view: far edge upper, +P.X to image right.
72
73 Image sorting alone cannot label an unmarked table. The caller must confirm
74 this restricted-view convention. Strongly rotated views are rejected.
75 """
76 if len(corners) != 4 or any(len(p) != 2 or not all(finite(v) for v in p) for p in corners):
77 raise ValueError("INVALID_TABLE_CORNERS")
78 ordered = sorted(corners,key=lambda p:p[1])
79 top,bottom = sorted(ordered[:2],key=lambda p:p[0]),sorted(ordered[2:],key=lambda p:p[0])
80 tl,tr,bl,br = top[0],top[1],bottom[0],bottom[1]
81 for left,right in ((tl,tr),(bl,br)):
82 if right[0]-left[0] <= 2*abs(right[1]-left[1]):
83 raise ValueError("TABLE_VIEW_ORIENTATION_UNSUPPORTED")
84 if min(bl[1]-tl[1],br[1]-tr[1]) <= 5:
85 raise ValueError("TABLE_VIEW_ORIENTATION_UNSUPPORTED")
86 return [tl,tr,br,bl]

89class NativeRescueTableReference:
90 """find_rects over the whole original frame; ROI is a candidate filter only.
91
92 Configure a ROI identifying the top face, not a side face or arbitrary
93 black rectangle. A custom corner_adapter(gray, rect) may supply physical
94 corner order instead of the explicitly confirmed restricted-view convention.
95 """
96 def __init__(self,calibration,layout,corner_adapter=None):
97 if corner_adapter is not None and not callable(corner_adapter):
98 raise ValueError("INVALID_TABLE_CORNER_ADAPTER")
99 self.calibration,self.layout,self.corner_adapter = calibration,layout,corner_adapter
100 self.estimator = None
101 self.closed,self.last_frame = False,-1
102
103 def readiness(self):
104 if self.closed: return "TABLE_REFERENCE_CLOSED"
105 return self.layout.native_ready_reason(self.corner_adapter is not None)
106
107 def _failure(self,reason,frame_id):
108 return {"ok":False,"reason":reason,"frame_id":frame_id,"T_camera_from_table":None}
109
110 def _black_fraction(self,gray,pose):
111 w,h = self.layout.data["table_width_mm"],self.layout.data["table_depth_mm"]
112 black,total = 0,0
113 for x in (-.4,-.2,0,.2,.4):
114 for y in (-.3,.3):
115 point = transform_point(pose["T_camera_from_table"],[x*w,y*h,0])
116 u,v = self.calibration.camera.project_ray(point)
117 u,v = int(round(u)),int(round(v))
118 if not 0 <= u < IMAGE_SIZE[0] or not 0 <= v < IMAGE_SIZE[1]:
119 if boundary_checks():raise ValueError("TABLE_SAMPLE_OUTSIDE_IMAGE")
120 continue
121 value = gray.get_pixel(u,v)
122 if not finite(value): raise ValueError("GRAYSCALE_PIXEL_API_UNSUPPORTED")
123 black += int(value <= settings.TABLE_BLACK_MAX_GRAY)
124 total += 1
125 if not total:raise ValueError("TABLE_NO_VISIBLE_SAMPLES")
126 return black/float(total)
127
128 def observe(self,raw,frame_id,geometry_confirmed=False):
129 reason = self.readiness()
130 if reason != "OK": return self._failure(reason,frame_id)
131 if type(frame_id) is not int or frame_id <= self.last_frame:
132 return self._failure("STALE_OR_DUPLICATE_TABLE_FRAME",frame_id)
133 self.last_frame = frame_id
134 if not geometry_confirmed or (raw.width(),raw.height()) != IMAGE_SIZE:
135 return self._failure("RAW_GEOMETRY_NOT_CONFIRMED",frame_id)
136 gray = None
137 try:
138 if self.estimator is None:
139 self.estimator = RescueTablePoseEstimator(self.calibration.camera,
140 self.layout.data["table_width_mm"],self.layout.data["table_depth_mm"])
141 gray = raw.to_grayscale()
142 # Do not crop or pass ROI: keep the corner coordinates unambiguously raw.
143 rectangles = gray.find_rects(threshold=settings.RECTANGLE_THRESHOLD)
144 if len(rectangles) > settings.MAX_TABLE_CANDIDATES:
145 return self._failure("TOO_MANY_TABLE_CANDIDATES",frame_id)
146 x,y,w,h = ((0,0,IMAGE_SIZE[0],IMAGE_SIZE[1]) if settings.TABLE_FULL_FRAME_SEARCH
147 else self.layout.data["table_top_roi_raw_px"])
148 accepted,reasons = [],[]
149 for rectangle in rectangles:
150 corners = rectangle.corners()
151 if len(corners) != 4 or not all(x <= p[0] < x+w and y <= p[1] < y+h for p in corners):
152 continue
153 try:
154 if self.corner_adapter is None:
155 corners = order_restricted_top_view(corners)
156 else:
157 corners = self.corner_adapter(gray,rectangle)
158 if corners is None: continue
159 pose = self.estimator.estimate_table(corners,True,True)
160 if not pose["ok"]:
161 reasons.append(pose["reason"]); continue
162 fraction = self._black_fraction(gray,pose)
163 if fraction < settings.TABLE_MIN_BLACK_FRACTION:
164 reasons.append("TABLE_TOP_COLOR_INCONSISTENT"); continue
165 pose.update(frame_id=frame_id,source="native_table_rectangle")
166 pose["quality"]["black_sample_fraction"] = fraction
167 accepted.append(pose)
168 except (ValueError,TypeError,IndexError) as exc:
169 reasons.append(str(exc))
170 if len(accepted) > 1:
171 return self._failure("TABLE_REFERENCE_AMBIGUOUS",frame_id)
172 if not accepted:
173 return self._failure(reasons[0] if reasons else "TABLE_TOP_NOT_FOUND",frame_id)
174 return accepted[0]
175 except Exception as exc:
176 return self._failure("TABLE_REFERENCE_ERROR:"+str(exc),frame_id)
177 finally:
178 gray = None
179
180 def close(self):
181 self.closed,self.estimator = True,None
