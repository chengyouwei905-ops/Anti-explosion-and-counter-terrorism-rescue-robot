6def finite(x):

14def cross(a, b):
15 return [a[1]*b[2]-a[2]*b[1], a[2]*b[0]-a[0]*b[2], a[0]*b[1]-a[1]*b[0]]

22def unit(a):
23 length = norm(a)
24 if length < 1e-12 or not finite(length):
25 raise ValueError("DEGENERATE_VECTOR")
26 return [x / length for x in a]

29def validate_rigid(T):
30 if len(T) != 4 or any(len(row) != 4 for row in T):
31 raise ValueError("BAD_TRANSFORM_SHAPE")
32 if not all(finite(x) for row in T for x in row):
33 raise ValueError("NONFINITE_TRANSFORM")
34 if max(abs(T[3][i] - (1.0 if i == 3 else 0.0)) for i in range(4)) > 1e-5:
35 raise ValueError("BAD_HOMOGENEOUS_ROW")
36 cols = [[T[i][j] for i in range(3)] for j in range(3)]
37 if max(abs(dot(cols[i], cols[j]) - (1.0 if i == j else 0.0))
38 for i in range(3) for j in range(3)) > 0.002 or dot(cols[0], cross(cols[1], cols[2])) < 0.998:
39 raise ValueError("NOT_A_PROPER_ROTATION")
40 return T

43def compose(A, B):
44 validate_rigid(A)
45 validate_rigid(B)
46 return [[sum(A[i][k]*B[k][j] for k in range(4)) for j in range(4)] for i in range(4)]

49def transform_point(T, point):
50 validate_rigid(T)
51 if len(point) != 3 or not all(finite(x) for x in point):
52 raise ValueError("BAD_POINT")
53 return [sum(T[i][j]*point[j] for j in range(3)) + T[i][3] for i in range(3)]

56class CameraModel:
57 def __init__(self, K, D, image_size=(640, 480)):
58 if len(K) != 3 or any(len(row) != 3 for row in K) or len(D) != 5:
59 raise ValueError("BAD_CAMERA_PARAMETERS")
60 if not all(finite(v) for row in K for v in row) or not all(finite(v) for v in D):
61 raise ValueError("BAD_CAMERA_PARAMETERS")
62 if K[0][0] <= 0 or K[1][1] <= 0 or any(abs(v) > 1e-8 for v in
63 (K[0][1], K[1][0], K[2][0], K[2][1], K[2][2]-1)):
64 raise ValueError("UNSUPPORTED_CAMERA_MATRIX")
65 self.K, self.D, self.image_size = K, D, tuple(image_size)
66 self.fx, self.fy, self.cx, self.cy = K[0][0], K[1][1], K[0][2], K[1][2]
67
68 def distort(self, x, y):
69 k1, k2, p1, p2, k3 = self.D
70 r2 = x*x+y*y
71 radial = 1+k1*r2+k2*r2*r2+k3*r2*r2*r2
72 return (x*radial+2*p1*x*y+p2*(r2+2*x*x),
73 y*radial+p1*(r2+2*y*y)+2*p2*x*y)
74
75 def ray(self, pixel):
76 u, v = pixel
77 if not finite(u) or not finite(v) or not (0 <= u < self.image_size[0] and 0 <= v < self.image_size[1]):
78 raise ValueError("PIXEL_OUTSIDE_RAW_IMAGE")
79 tx, ty = (u-self.cx)/self.fx, (v-self.cy)/self.fy
80 x, y = tx, ty
81 # Fixed-point Brown inversion with a verified forward residual.
82 for _ in range(20):
83 dx, dy = self.distort(x, y)
84 ex, ey = tx-dx, ty-dy
85 x, y = x+ex, y+ey
86 if ex*ex+ey*ey < 1e-18:
87 break
88 dx, dy = self.distort(x, y)
89 if (dx-tx)**2+(dy-ty)**2 > 1e-12:
90 raise ValueError("UNDISTORT_DID_NOT_CONVERGE")
91 return unit([x, y, 1.0])
92
93 def project_ray(self, ray):
94 if ray[2] <= 1e-9:
95 raise ValueError("NONPOSITIVE_DEPTH")
96 x, y = self.distort(ray[0]/ray[2], ray[1]/ray[2])
97 return (self.fx*x+self.cx, self.fy*y+self.cy)

100def _plane_from_three(a, b, c):
101 n = unit(cross([b[i]-a[i] for i in range(3)], [c[i]-a[i] for i in range(3)]))
102 k = dot(n, a)
103 if k < 0:
104 n, k = [-v for v in n], -k
105 if not (0.01 < k < 1.0):
106 raise ValueError("DEGENERATE_TANGENT_CONE")
107 return n, math.sqrt(max(0.0, 1-k*k)), k

110def _plane_fit(rays):
111 # Smallest covariance eigenvector; Jacobi rotations need only standard math.
112 mean = [sum(r[i] for r in rays)/len(rays) for i in range(3)]
113 A = [[sum((r[i]-mean[i])*(r[j]-mean[j]) for r in rays) for j in range(3)] for i in range(3)]
114 V = [[float(i == j) for j in range(3)] for i in range(3)]
115 scale = max(abs(x) for row in A for x in row)
116 if scale < 1e-14:
117 raise ValueError("DEGENERATE_CONTOUR")
118 A = [[x/scale for x in row] for row in A]
119 for _ in range(24):
120 p, q = max(((0, 1), (0, 2), (1, 2)), key=lambda ij: abs(A[ij[0]][ij[1]]))
121 if abs(A[p][q]) < 1e-10:
122 break
123 angle = 0.5*math.atan2(2*A[p][q], A[q][q]-A[p][p])
124 c, s = math.cos(angle), math.sin(angle)
125 app, aqq, apq = A[p][p], A[q][q], A[p][q]
126 A[p][p] = c*c*app-2*s*c*apq+s*s*aqq
127 A[q][q] = s*s*app+2*s*c*apq+c*c*aqq
128 A[p][q] = A[q][p] = 0.0
129 for j in range(3):
130 if j != p and j != q:
131 ajp, ajq = A[j][p], A[j][q]
132 A[j][p] = A[p][j] = c*ajp-s*ajq
133 A[j][q] = A[q][j] = s*ajp+c*ajq
134 vjp, vjq = V[j][p], V[j][q]
135 V[j][p], V[j][q] = c*vjp-s*vjq, s*vjp+c*vjq
136 idx = min(range(3), key=lambda i: A[i][i])
137 n = unit([V[i][idx] for i in range(3)])
138 if dot(n, mean) < 0:
139 n = [-x for x in n]
140 # Use cross products rather than 1-k^2 to retain float32 precision.
141 sine = sum(norm(cross(n, r)) for r in rays)/len(rays)
142 cosine = sum(dot(n, r) for r in rays)/len(rays)
143 return n, sine, cosine

146def _pixel_errors(camera, pixels, rays, axis, sine, cosine):
147 errors = []
148 for pixel, ray in zip(pixels, rays):
149 parallel = dot(axis, ray)
150 tangent = [ray[j]-parallel*axis[j] for j in range(3)]
151 try:
152 tangent = unit(tangent)
153 ideal = [cosine*axis[j]+sine*tangent[j] for j in range(3)]
154 u, v = camera.project_ray(ideal)
155 errors.append(math.sqrt((u-pixel[0])**2+(v-pixel[1])**2))
156 except ValueError:
157 errors.append(1e9)
158 return errors

161def fit_sphere_center(camera, boundary_points_px, diameter_mm=40.0,
162 max_error_px=2.0, min_inlier_ratio=0.80,
163 min_coverage_deg=270.0, min_radius_px=8.0,
164 max_distance_mm=2000.0):
165 """Fit the tangent cone of a known sphere, NOT a bbox diameter or ellipse center.
166
167 Raw silhouette rays lie on a small circle on the unit sphere. Its axis
168 points to the physical sphere center; distance = radius/sin(half angle).
169 Orientation is unobservable. No frame history or hardware is accessed.
170 """
171 try:
172 if not finite(diameter_mm) or diameter_mm <= 0:
173 raise ValueError("INVALID_SPHERE_SIZE")
174 points = list(boundary_points_px)
175 if not 16 <= len(points) <= 512:
176 raise ValueError("CONTOUR_POINT_COUNT")
177 rays = [camera.ray(p) for p in points]
178 count = len(rays)
179 # Deterministic, bounded RANSAC. Points need not be angularly ordered.
180 candidates = [_plane_fit(rays)]
181 seed = 71237
182 for _ in range(64):
183 indices = []
184 for _ in range(3):
185 seed = (1664525*seed+1013904223) & 0xffffffff
186 indices.append(seed % count)
187 if len(set(indices)) != 3:
188 continue
189 try:
190 candidates.append(_plane_from_three(*(rays[i] for i in indices)))
191 except ValueError:
192 pass
193 best, best_score = None, None
194 for axis, sine, cosine in candidates:
195 if sine <= 1e-5 or axis[2] <= 0:
196 continue
197 errors = _pixel_errors(camera, points, rays, axis, sine, cosine)
198 selected = [i for i, e in enumerate(errors) if e <= max_error_px]
199 score = (len(selected), -sum(errors[i]**2 for i in selected))
200 if best_score is None or score > best_score:
201 best, best_score = selected, score
202 strict=boundary_checks()
203 required=max(16,int(math.ceil(count*min_inlier_ratio)))
204 if strict and (best is None or len(best)<required):
205 raise ValueError("SPHERE_CONTOUR_INCONSISTENT")
206 if best is None or len(best)<3:
207 best=list(range(count))
208 for _ in range(2):
209 axis, sine, cosine = _plane_fit([rays[i] for i in best])
210 scale = math.sqrt(sine*sine+cosine*cosine)
211 sine, cosine = sine/scale, cosine/scale
212 errors = _pixel_errors(camera, points, rays, axis, sine, cosine)
213 selected = [i for i, e in enumerate(errors) if e <= max_error_px]
214 if strict and len(selected)<required:
215 raise ValueError("SPHERE_CONTOUR_INCONSISTENT")
216 if len(selected)>=3:best=selected
217 if strict and sine*min(camera.fx, camera.fy) < min_radius_px:
218 raise ValueError("SPHERE_TOO_SMALL")
219 distance = diameter_mm*0.5/sine
220 if not finite(distance) or not (diameter_mm*0.5 < distance <= max_distance_mm):
221 raise ValueError("SPHERE_RANGE_INVALID")
222 ex = unit(cross(axis, [0, 1, 0] if abs(axis[1]) < 0.9 else [1, 0, 0]))
223 ey = cross(axis, ex)
224 angles = sorted(math.atan2(dot(rays[i], ey), dot(rays[i], ex)) for i in best)
225 gaps = [angles[i+1]-angles[i] for i in range(len(angles)-1)]
226 gaps.append(angles[0]+2*math.pi-angles[-1])
227 coverage = 360.0-max(gaps)*180.0/math.pi
228 if strict and coverage < min_coverage_deg:
229 raise ValueError("SPHERE_CONTOUR_INCOMPLETE")
230 actual_inliers=sum(e<=max_error_px for e in errors)
231 warnings=[]
232 if actual_inliers<required:warnings.append("SPHERE_CONTOUR_INCONSISTENT")
233 if coverage<min_coverage_deg:warnings.append("SPHERE_CONTOUR_INCOMPLETE")
234 if sine*min(camera.fx,camera.fy)<min_radius_px:warnings.append("SPHERE_TOO_SMALL")
235 center = [distance*v for v in axis]
236 if center[2] <= diameter_mm*0.5:
237 raise ValueError("NONPOSITIVE_DEPTH")
238 return {"ok": True, "reason": "OK", "position_camera_mm": center,
239 "reference_point": "sphere_center", "orientation_valid": False,
240 "quality": {"contour_rms_px": math.sqrt(sum(errors[i]**2 for i in best)/len(best)),
241 "inlier_ratio": actual_inliers/float(count), "coverage_deg": coverage,
242 "boundary_checks_enabled":strict,"warnings":warnings,
243 "point_count": count, "diameter_mm": diameter_mm}}
244 except (ValueError, TypeError, IndexError, ZeroDivisionError, OverflowError) as exc:
245 return {"ok": False, "reason": str(exc), "position_camera_mm": None,
246 "orientation_valid": False}
