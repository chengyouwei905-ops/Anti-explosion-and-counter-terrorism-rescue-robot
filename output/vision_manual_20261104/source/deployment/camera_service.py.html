7def ticks_ms():
8 method=getattr(time,"ticks_ms",None)
9 return method() if method is not None else int(time.monotonic()*1000)

12def ticks_diff(a,b):
13 method=getattr(time,"ticks_diff",None)
14 return method(a,b) if method is not None else a-b

17def settle_outputs():
18 """Allow display/DMA teardown to settle before releasing image/VB memory."""
19 method=getattr(time,"sleep_ms",None)
20 if method is not None:method(100)
21 else:time.sleep(0.1)

24def lcd_wrap(text,columns=18):
25 """Bound panel width, treating CJK characters as two Latin columns."""
26 lines=[];line="";width=0
27 for char in str(text):
28 size=2 if ord(char)>127 else 1
29 if char=="\n" or width+size>columns:
30 lines.append(line);line="";width=0
31 if char=="\n":continue
32 line+=char;width+=size
33 if line or not lines:lines.append(line)
34 return lines

68 def _create_sensor(self):
69 self.sensor=self.sensor_type(id=0)
70 self.sensor.reset()
71 self.sensor.set_framesize(self.sensor_type.VGA,chn=self.channel)
72 self.sensor.set_pixformat(self.sensor_type.RGB888,chn=self.channel)
73 if self.qr_enabled:
74 if cfg.QR_CAPTURE_CHANNEL==self.channel:
75 raise ValueError("QR_CHANNEL_MUST_DIFFER_FROM_RGB_CHANNEL")
76 width,height=cfg.QR_CAPTURE_SIZE
77 # 同一传感器/acq_win。原VGA默认crop=None，此路显式不裁剪。
78 self.sensor.set_framesize(width=width,height=height,
79 chn=cfg.QR_CAPTURE_CHANNEL,crop=False)
80 self.sensor.set_pixformat(self.sensor_type.GRAYSCALE,chn=cfg.QR_CAPTURE_CHANNEL)

82 def _ensure_display(self):
83 if not cfg.DISPLAY_ENABLED:return False
84 self._load_types()
85 if not self.display_touched:
86 # The official display-and-touch example supports LCD-only output.
87 # No Sensor instance or snapshots are needed before the user key.
88 self.display_touched=True
89 try:
90 self.display.init(self.display.ST7701,width=cfg.DISPLAY_WIDTH,
91 height=cfg.DISPLAY_HEIGHT,
92 to_ide=cfg.DISPLAY_TO_IDE if self.display_to_ide is None else bool(self.display_to_ide))
93 except BaseException:
94 try:self.display.deinit()
95 finally:
96 self.display_touched=False
97 settle_outputs()
98 raise
99 return True

101 def open(self):
102 if self.state!="CLOSED": return
103 self._load_types()
104 try:
105 self._create_sensor()
106 self._ensure_display()
107 self.media_touched=True
108 self.media.init()
109 self.sensor.run()
110 self.epoch+=1
111 self.warmup=cfg.CAMERA_WARMUP_FRAMES
112 self.state="WARMUP" if self.warmup else "READY"
113 except BaseException:
114 self.close()
115 raise

125 def get_frame(self):
126 if self.state=="ERROR": return None
127 operation=self.state
128 try:
129 if self.state=="CLOSED":
130 self.open()
131 return None
132 if self.state=="RECOVER_WAIT":
133 if self.diff(self.clock(),self.restart_at)<cfg.CAMERA_RESTART_WAIT_MS: return None
134 self._create_sensor()
135 self.sensor.run()
136 self.warmup=cfg.CAMERA_WARMUP_FRAMES
137 self.state="WARMUP" if self.warmup else "READY"
138 return None
139 qr_raw=None
140 if self.qr_requested:
141 qr_raw=self.sensor.snapshot(chn=cfg.QR_CAPTURE_CHANNEL,
142 timeout=cfg.SNAPSHOT_TIMEOUT_MS,auto_reboot=False)
143 if (qr_raw.width(),qr_raw.height())!=tuple(cfg.QR_CAPTURE_SIZE):
144 raise RuntimeError("QR_CAPTURE_SIZE_MISMATCH")
145 raw=self.sensor.snapshot(chn=self.channel,timeout=cfg.SNAPSHOT_TIMEOUT_MS,auto_reboot=False)
146 returned=self.clock()
147 self.frame_id+=1
148 self.failures=0
149 if self.state=="WARMUP":
150 self.warmup-=1
151 if self.warmup<=0: self.state="READY"
152 return None
153 frame={"image":raw,"frame_id":self.frame_id,"capture_return_ms":returned,
154 "epoch":self.epoch,"geometry_confirmed":cfg.RAW_GEOMETRY_CONFIRMED}
155 if qr_raw is not None:
156 frame["qr_image"]=qr_raw
157 frame["qr_geometry_confirmed"]=cfg.RAW_GEOMETRY_CONFIRMED
158 return frame
159 except (RuntimeError,OSError) as exc:
160 self.error="CAMERA_ERROR:"+str(exc)
161 self.epoch+=1
162 if operation in ("CLOSED","RECOVER_WAIT"):
163 try: self._stop_sensor()
164 except Exception: pass
165 self.state="ERROR"
166 return None
167 self.failures+=1
168 if self.failures>=cfg.SNAPSHOT_ATTEMPTS:
169 if self.restarts>=cfg.MAX_CAMERA_RESTARTS:
170 self.state="ERROR"
171 return None
172 try:
173 self._stop_sensor()
174 except Exception as stop_error:
175 self.error="CAMERA_ERROR:"+str(stop_error)
176 self.state="ERROR"
177 return None
178 self.restarts+=1
179 self.failures=0
180 self.restart_at=self.clock()
181 self.state="RECOVER_WAIT"
182 return None

184 def render_debug(self,image=None,status=None):
185 preview=None

188 if self.screen is None:

230 def close(self):
231 errors=[]
232 had_output=self.display_touched or self.media_touched
233 try:
234 self._stop_sensor()
235 except Exception as exc: errors.append(str(exc))
236 try:
237 if self.display_touched: self.display.deinit()
238 except Exception as exc: errors.append(str(exc))
239 finally: self.display_touched=False
240 # Keep the rendered canvas alive until display/DMA shutdown has settled,
241 # then release its references before destroying the media buffers.
242 if had_output:settle_outputs()
243 self.screen=None
244 gc.collect()
245 try:
246 if self.media_touched: self.media.deinit()
247 except Exception as exc: errors.append(str(exc))
248 finally:
249 self.media_touched=False
250 self.state="CLOSED"
251 if errors: self.error="CAMERA_CLOSE_ERROR:"+";".join(errors)
