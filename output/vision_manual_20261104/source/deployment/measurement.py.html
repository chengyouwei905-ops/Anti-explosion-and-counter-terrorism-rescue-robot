14class MeasurementAverage:
15 def __init__(self,count=3,max_scatter_mm=8.0):
16 if type(count) is not int or not 1<=count<=32: raise ValueError("INVALID_SAMPLE_COUNT")
17 if not finite(max_scatter_mm) or max_scatter_mm<=0: raise ValueError("INVALID_SCATTER_LIMIT")
18 self.count,self.max_scatter=count,max_scatter_mm
19 self.samples=[]
20 self.last_frame=-1
21 self.identity=None
22
23 def reset(self):
24 self.samples=[]
25 self.identity=None
26
27 def add(self,result):
28 frame=result.get("frame_id")
29 if type(frame) is not int or frame<0:
30 self.reset(); return None,"INVALID_MEASUREMENT_FRAME"
31 if frame<=self.last_frame: return None,"STALE_OR_DUPLICATE_FRAME"
32 self.last_frame=frame
33 point=result.get("position_tool_mm")
34 if not result.get("ok") or point is None:
35 self.reset(); return None,result.get("reason","INVALID_MEASUREMENT")
36 if len(point)!=3 or not all(finite(v) for v in point):
37 self.reset(); return None,"NONFINITE_MEASUREMENT"
38 if result.get("coordinate_frame")!="E_commanded_gripper_center" or result.get("units")!="mm":
39 self.reset(); return None,"INVALID_MEASUREMENT_COORDINATES"
40 task=result.get("task")
41 identity=(result.get("target_id"),result.get("slot_id"),result.get("reference_point"),
42 result.get("coordinate_frame"),task.get("code") if task else None)
43 if identity!=self.identity: self.reset(); self.identity=identity
44 sample=position_fields(result)
45 sample["position_tool_mm"]=list(point)
46 self.samples.append(sample)
47 mean=[sum(s["position_tool_mm"][i] for s in self.samples)/len(self.samples) for i in range(3)]
48 scatter=max(norm([s["position_tool_mm"][i]-mean[i] for i in range(3)]) for s in self.samples)
49 if scatter>self.max_scatter:
50 self.samples=[sample]; return None,"POSITION_NOT_STABLE"
51 if len(self.samples)<self.count: return None,"MEASUREMENT_STABILIZING"
52 output=dict(sample)
53 output.update(position_tool_mm=mean,scatter_mm=scatter,sample_count=len(self.samples),
54 sample_frame_ids=[s["frame_id"] for s in self.samples],
55 control_ready=all(s.get("control_ready") is True for s in self.samples),
56 handeye_accepted=all(s.get("handeye_accepted") is True for s in self.samples))
57 vectors=[s.get("grasp_position_tool_mm") for s in self.samples]
58 output["grasp_position_tool_mm"]=None
59 if all(v is not None and len(v)==3 and all(finite(x) for x in v) for v in vectors):
60 average=[sum(v[i] for v in vectors)/len(vectors) for i in range(3)]
61 if max(norm([v[i]-average[i] for i in range(3)]) for v in vectors)<=self.max_scatter:
62 output["grasp_position_tool_mm"]=average
63 return output,"OK"

66class QRMeasurementAverage(MeasurementAverage):
67 """Average only valid QR poses in a bounded window, tolerating brief misses."""
68
69 def __init__(self,count,max_scatter_mm,window_ms,clock,diff):
70 if type(window_ms) is not int or window_ms<=0:
71 raise ValueError("INVALID_QR_SAMPLE_WINDOW")
72 super().__init__(count,max_scatter_mm)
73 self.window_ms,self.clock,self.diff=window_ms,clock,diff
74
75 def add(self,result,captured_ms=None):
76 now=self.clock()
77 self.samples=[sample for sample in self.samples
78 if 0<=self.diff(now,sample["_qr_capture_ms"])<=self.window_ms]
79 if not self.samples:self.identity=None
80 frame=result.get("frame_id")
81 if type(frame) is not int or frame<0 or frame<=self.last_frame:
82 return super().add(result)
83 reason=result.get("reason")
84 # These observations contribute no pose. Only an identified SAME code
85 # may bridge an uncertain position; multiple/changed/invalid codes reset.
86 same_code=(self.samples and result.get("task") is not None and
87 result["task"].get("code")==self.samples[-1].get("task",{}).get("code"))
88 if (not result.get("ok") and
89 (reason=="QR_NOT_FOUND" or
90 (same_code and reason in ("qr_center_ambiguous","qr_position_uncertain")))):
91 self.last_frame=frame
92 return None,reason
93 capture=now if captured_ms is None else captured_ms
94 if type(capture) is not int or not 0<=self.diff(now,capture)<=self.window_ms:
95 self.last_frame=frame
96 self.reset()
97 return None,"QR_SAMPLE_EXPIRED"
98 sample=dict(result)
99 sample["_qr_capture_ms"]=capture
100 output,reason=super().add(sample)
101 if output is not None:output.pop("_qr_capture_ms",None)
102 return output,reason
