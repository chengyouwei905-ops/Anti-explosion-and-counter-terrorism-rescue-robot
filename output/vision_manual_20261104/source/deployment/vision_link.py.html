1"""
2vision_link.py —— K230 端串口传输与命令接收（后台线程版）
3
4协议格式（与 STM32 工程 Protocol/Inc/proto_vision.h 对齐）:
5
6 +------+------+-----+-----+-----+---------------+------+
7 | 0xAA | 0x55 | LEN | SEQ | CMD | PAYLOAD | CRC8 |
8 +------+------+-----+-----+-----+---------------+------+
9
10 LEN : PAYLOAD 字节数 (0 ~ 64)
11 SEQ : 帧序号（STM32 端自增，用于丢帧检测）
12 CMD : 命令字（原接口仍可用，新增任务业务分配见 protocol.py）
13 CRC8 : 对 LEN/SEQ/CMD/PAYLOAD 计算
14 CRC-8/MAXIM（poly 0x8C 反射、init 0x00），与 STM32 端 proto_vision_crc8 一致
15
16用法（主线程只需要一个接口 vision.get()）:
17
18 from vision_link import VisionLink
19 vision = VisionLink() # 后台线程自动开始收数
20 ...
21 frame = vision.get() # 最近一帧：Frame(seq, cmd, payload, ts)，无数据返回 None
22 if frame:
23 # frame.seq / frame.cmd / frame.payload(bytes) / frame.ts
24 ...
25 vision.deinit() # 退出前停止线程并释放 UART
26
27说明:
28- 后台线程负责接收、解析、CRC 和重同步；主线程取命令并发送结果。
29- 默认保留最新帧接口；get_new() 按本地接收编号判断，不受 SEQ 回绕影响。
30- queue_size>0 时另启有界命令队列；STOP 等 priority_cmds 优先取出。
31- send(cmd, payload) 按同一格式回发一帧，并检查 UART 写入长度。
32"""

62def crc8(data, crc=0):
63 """CRC-8/MAXIM（反射多项式 0x8C，初值 0x00），与 STM32 端 proto_vision_crc8 完全一致。"""
64 for b in data:
65 crc ^= b
66 for _ in range(8):
67 if crc & 0x01:
68 crc = (crc >> 1) ^ 0x8C
69 else:
70 crc >>= 1
71 return crc & 0xFF

144 def get_new(self):
145 """返回"新"一帧：与上次取到的是同一帧时返回 None。"""
146 self._lock.acquire()
147 try:
148 f = self._latest
149 if f is None or f.rx_id == self._last_rx_id:
150 return None
151 self._last_rx_id = f.rx_id
152 return f
153 finally:
154 self._lock.release()

205 def pack(self, cmd, payload=b""):
206 """按协议组一帧（含 CRC），返回 bytes，不发送。payload 必须为 bytes。"""
207 if len(payload) > MAX_PAYLOAD:
208 raise ValueError("payload too long")
209 if type(cmd) is not int or not 0 <= cmd <= 255:
210 raise ValueError("invalid cmd")
211 head = bytes((len(payload), self.tx_seq & 0xFF, cmd))
212 self.tx_seq = (self.tx_seq + 1) & 0xFF
213 body = head + bytes(payload)
214 return _HEADER + body + bytes((crc8(body),))

216 def send(self, cmd, payload=b""):
217 """按同一格式回发一帧给 STM32。"""
218 if self._closed:
219 raise OSError("UART_CLOSED")
220 frame = self.pack(cmd, payload)
221 written = self.uart.write(frame)
222 if written != len(frame):
223 raise OSError("UART_SHORT_WRITE")
224 return frame

226 def send_raw(self, data):
227 """Main-thread lifecycle token: no AA55 wrapper, CRC or newline."""
228 if self._closed: raise OSError("UART_CLOSED")
229 data=bytes(data)
230 if self.uart.write(data)!=len(data): raise OSError("UART_SHORT_WRITE")

233 def set_command_acceptance(self, enabled):
234 """Atomically purge stale/partial RX at a user-key lifecycle boundary."""
235 self._rx_io_lock.acquire()
236 try:
237 self._lock.acquire()
238 try:
239 self._accept_commands=False
240 self.commands_discarded+=len(self._queue)+(1 if self._priority is not None else 0)
241 self._queue,self._priority,self._overflow=[],None,None
242 self._buf=bytearray()
243 self._latest=None
244 # Reader and this purge cannot consume the UART simultaneously.
245 # Bounded drain: never hang on a continuously transmitting peer.
246 for _ in range(32):
247 available=self.uart.any()
248 if not available: break
249 self.uart.read(min(available,512))
250 else:
251 if self.uart.any(): raise OSError("UART_RX_NOT_QUIET")
252 self._accept_commands=bool(enabled)
253 finally: self._lock.release()
254 finally: self._rx_io_lock.release()

256 def deinit(self, wait_ms=300):
257 """停止接收线程并释放串口，退出前调用。"""
258 if self._closed:
259 return True
260 self._running = False
261 started = _ticks_ms()
262 while not self._rx_exited and _ticks_diff(_ticks_ms(), started) < wait_ms:
263 _sleep_ms(5)
264 if not self._rx_exited:
265 self._rx_error = "UART_SHUTDOWN_TIMEOUT"
266 return False
267 self.uart.deinit()
268 self._closed = True
269 return True

274 def _rx_task(self):
275 try:
276 while self._running:
277 data=None
278 self._rx_io_lock.acquire()
279 try:
280 available = self.uart.any()
281 if available:
282 data = self.uart.read(min(available, self._read_chunk))
283 if data: self._feed(data)
284 finally: self._rx_io_lock.release()
285 if not data: _sleep_ms(5)
286 except SystemExit:
287 if self._running:
288 self._rx_error = "RX_THREAD_EXITED"
289 except BaseException as e:
290 self._rx_error = type(e).__name__ + ":" + str(e)
291 finally:
292 self._running = False
293 self._rx_exited = True

295 def _publish(self, seq, cmd, payload):
296 self._lock.acquire()
297 try:
298 if self._accept_commands: self._publish_locked(seq,cmd,payload)
299 finally:
300 self._lock.release()

302 def _publish_locked(self,seq,cmd,payload):
303 self._rx_id += 1
304 frame = Frame(seq, cmd, payload, _ticks_ms(), self._rx_id)
305 self._latest = frame
306 if self._queue_size:
307 if cmd in self._priority_cmds: self._priority = frame
308 elif len(self._queue) < self._queue_size: self._queue.append(frame)
309 else:
310 self.queue_overflows += 1
311 self._overflow = frame

313 def _feed(self, data):
314 """解析字节流（支持半帧/粘包/垃圾数据），线程内调用。"""
315 self._lock.acquire()
316 try:
317 if not self._accept_commands:
318 self.rx_bytes+=len(data)
319 self._buf=bytearray()
320 return
321 self._feed_locked(data)
322 finally: self._lock.release()
323
324 def _feed_locked(self,data):
325 self.rx_bytes += len(data)
326 buf = self._buf + data
327 while True:
328 i = buf.find(_HEADER)
329 if i < 0:
330 if len(buf) > 1:
331 buf = buf[-1:] # 只保留最后一个字节（可能是半个帧头）
332 break
333 if i:
334 buf = buf[i:] # 丢掉帧头之前的垃圾
335 if len(buf) < OVERHEAD:
336 break # 帧还没收全
337 n = buf[2]
338 if n > MAX_PAYLOAD:
339 self.len_errors += 1
340 buf = buf[1:] # 长度非法，跳过 1 字节重新同步
341 continue
342 total = OVERHEAD + n
343 if len(buf) < total:
344 break # 帧还没收全
345 if crc8(buf[2:total - 1]) == buf[total - 1]:
346 self.ok_frames += 1
347 self._publish_locked(buf[3], buf[4], bytes(buf[5:total - 1]))
348 buf = buf[total:]
349 else:
350 self.crc_errors += 1
351 buf = buf[1:] # CRC 错，跳过 1 字节重新同步
352 self._buf = buf
