21def main(mode="inspect",shape_id=1,slot_id=None,max_frames=120,detector_backend=None,
22 scan_order_confirmed=False,start_boundary_confirmed=False):
23 """inspect works without a model; it shows rectangle candidates, no 3D result."""
24 if mode not in ("inspect","table","scan","measure"):
25 raise ValueError("mode must be inspect, table, scan or measure")
26 if mode=="scan" and not (scan_order_confirmed is True and start_boundary_confirmed is True):
27 return {"ok":False,"reason":"SCAN_ORDER_NOT_CONFIRMED"}
28 if type(max_frames) is not int or max_frames<1: raise ValueError("INVALID_FRAME_BUDGET")
29 cal=load_calibration(PROJECT_ROOT,handeye_accepted_override=runtime_cfg.HANDEYE_ACCEPTED_BY_USER)
30 with open(PROJECT_ROOT+"/"+LAYOUT_TEMPLATE,"r") as stream: layout=RescueLayout(json.load(stream))
31 reference=NativeRescueTableReference(cal,layout)
32 detector=task=sensor=raw=preview=gray=None
33 display_touched=media_touched=False
34 frame_id,last_print=0,0
35 result={"ok":False,"reason":"NO_OBSERVATION","position_tool_mm":None}
36 try:
37 if mode in ("scan","measure"):
38 with open(PROJECT_ROOT+"/"+MODEL_TEMPLATE,"r") as stream: metadata=json.load(stream)
39 if metadata.get("model_path") and not metadata["model_path"].startswith("/"):
40 metadata["model_path"]=PROJECT_ROOT+"/"+metadata["model_path"]
41 if detector_backend is None and metadata.get("model_path"):
42 detector_backend=runtime_cfg.create_model_backend("rescue",metadata)
43 detector=RescueYoloDetector(metadata,detector_backend)
44 loaded=detector.load()
45 if not loaded["ok"]: return loaded
46 object_mode=runtime_cfg.RESCUE_REFERENCE_MODE=="object"
47 task=RescueObjectTask(cal,detector) if object_mode else RescueTask(cal,detector,layout,reference)
48 if mode=="scan": task.start_scan(shape_id,True,True,max_frames=max_frames)
49 else:
50 if not object_mode:
51 if slot_id is None: return {"ok":False,"reason":"TARGET_SLOT_NOT_KNOWN"}
52 if reference.readiness()!="OK": return {"ok":False,"reason":reference.readiness()}
53 task.start_measure(shape_id,slot_id,max_frames=max_frames)
54 elif mode=="table" and reference.readiness()!="OK":
55 return {"ok":False,"reason":reference.readiness()}
56 last_print=time.ticks_ms()
57 from media.sensor import Sensor
58 from media.display import Display
59 from media.media import MediaManager,CAM_CHN_ID_0
60 sensor=Sensor(id=0)
61 sensor.reset()
62 sensor.set_framesize(Sensor.VGA,chn=CAM_CHN_ID_0)
63 sensor.set_pixformat(Sensor.RGB888,chn=CAM_CHN_ID_0)
64 display_touched=True
65 Display.init(Display.ST7701,width=800,height=480,to_ide=False)
66 media_touched=True
67 MediaManager.init()
68 sensor.run()
69 for _ in range(10):
70 os.exitpoint()
71 raw=sensor.snapshot(chn=CAM_CHN_ID_0,timeout=2000,auto_reboot=False)
72 raw=None
73 while frame_id<max_frames:
74 os.exitpoint()
75 raw=sensor.snapshot(chn=CAM_CHN_ID_0,timeout=2000,auto_reboot=False)
76 sampled_ms=time.ticks_ms()
77 frame_id+=1
78 preview=raw.to_rgb565()
79 if mode=="inspect":
80 gray=raw.to_grayscale()
81 rects=gray.find_rects(threshold=RECTANGLE_THRESHOLD)
82 candidates=[list(r.corners()) for r in rects[:MAX_TABLE_CANDIDATES]]
83 result={"ok":False,"reason":"TABLE_CANDIDATES_ONLY","frame_id":frame_id,
84 "corners_candidates_raw_px":candidates,"candidate_count":len(rects),
85 "position_tool_mm":None,"control_ready":False}
86 for index,corners in enumerate(candidates):
87 for i,p in enumerate(corners):
88 next_point=corners[(i+1)%4]
89 preview.draw_line(int(p[0]),int(p[1]),int(next_point[0]),int(next_point[1]),color=(0,255,0),thickness=2)
90 preview.draw_string_advanced(int(p[0]),int(p[1]),14,str(index)+":"+str(i),color=(255,255,0))
91 elif mode=="table":
92 result=reference.observe(raw,frame_id,True)
93 result["control_ready"]=False
94 else:
95 result=task.poll(raw,frame_id,geometry_confirmed=True,stable=(mode=="measure"))
96 if mode=="scan": preview.draw_rectangle(SCAN_GATE_RAW_PX,color=(255,255,0),thickness=2)
97 result["capture_return_ms"]=sampled_ms
98 if mode in ("scan","measure"):
99 lines=["RESCUE "+mode,"shape: %d"%shape_id,result["reason"]]
100 if result.get("position_tool_mm") is not None:
101 lines.extend(["%s: %+.1f mm"%(axis,value) for axis,value in zip("XYZ",result["position_tool_mm"])])
102 for index,text in enumerate(lines):
103 preview.draw_string_advanced(5,5+index*20,16,text,color=(255,255,0))
104 Display.show_image(preview)
105 now=time.ticks_ms()
106 if time.ticks_diff(now,last_print)>=500:
107 print("RESCUE",frame_id,result["reason"],"slot",result.get("target_slot"),
108 "p_E_mm",result.get("position_tool_mm"))
109 if mode=="inspect": print("TABLE_CORNERS",result["corners_candidates_raw_px"])
110 last_print=now
111 if result.get("complete") or result.get("measurement_complete") or (mode=="table" and result["ok"]):
112 break
113 raw=preview=gray=None
114 gc.collect()
115 else:
116 if mode!="inspect":
117 result={"ok":False,"reason":"RESCUE_TASK_FRAME_BUDGET_EXCEEDED",
118 "position_tool_mm":None,"control_ready":False}
119 return result
120 finally:
121 raw=preview=gray=None
122 try:
123 try:
124 if task is not None:
125 closer=getattr(task,"close",None)
126 if closer is not None:closer()
127 else:task.cancel()
128 finally:
129 try:
130 if detector is not None:detector.close()
131 finally:reference.close()
132 finally:
133 try:
134 if sensor is not None: sensor.stop()
135 finally:
136 try:
137 if display_touched: Display.deinit()
138 finally:
139 gc.collect()
140 # No media module was imported for preflight failures.
141 if display_touched or media_touched or sensor is not None:
142 os.exitpoint(os.EXITPOINT_ENABLE_SLEEP)
143 time.sleep_ms(100)
144 if media_touched: MediaManager.deinit()
145
146
147if __name__=="__main__":
148 # Change shape_id to 1=cylinder, 2=frustum, 3=drum for a stationary check.
149 main(mode="measure",shape_id=1)
