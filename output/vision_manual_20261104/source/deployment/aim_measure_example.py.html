1"""Standalone K230 target observation. No UART, laser, chassis or arm commands."""
2import gc
3import os
4import sys
5import time
6PROJECT_ROOT = "/data/guanggongsai"
7if PROJECT_ROOT not in sys.path:
8 sys.path.append(PROJECT_ROOT)
9from calibration_params import load_calibration
10from tasks.task_aim import AimTask
11from aim_config import SCAN_GATE_RAW_PX
12
13
14def main(mode="measure", color_id=1, max_frames=180,
15 scan_order_confirmed=False, start_boundary_confirmed=False):
16 if mode not in ("scan", "measure"):
17 raise ValueError("mode must be scan or measure")
18 if mode == "scan" and not (scan_order_confirmed is True and start_boundary_confirmed is True):
19 return {"ok": False, "reason": "SCAN_ORDER_NOT_CONFIRMED"}
20 task = AimTask(load_calibration(PROJECT_ROOT))
21 if mode == "scan":
22 task.start_scan(color_id, scan_order_confirmed, start_boundary_confirmed, max_frames=max_frames)
23 else:
24 task.start_measure(color_id, max_frames=max_frames)
25 sensor = raw = preview = None
26 display_touched = media_touched = False
27 frame_id, last_print = 0, time.ticks_ms()
28 result = {"ok": False, "reason": "TARGET_NOT_FOUND"}
29 try:
30 from media.sensor import Sensor
31 from media.display import Display
32 from media.media import MediaManager, CAM_CHN_ID_0
33 sensor = Sensor(id=0)
34 sensor.reset()
35 sensor.set_framesize(Sensor.VGA, chn=CAM_CHN_ID_0)
36 sensor.set_pixformat(Sensor.RGB888, chn=CAM_CHN_ID_0)
37 display_touched = True
38 Display.init(Display.ST7701, width=800, height=480, to_ide=False)
39 media_touched = True
40 MediaManager.init()
41 sensor.run()
42 for _ in range(10):
43 os.exitpoint()
44 raw = sensor.snapshot(chn=CAM_CHN_ID_0, timeout=2000, auto_reboot=False)
45 raw = None
46 while frame_id < max_frames:
47 os.exitpoint()
48 raw = sensor.snapshot(chn=CAM_CHN_ID_0, timeout=2000, auto_reboot=False)
49 sampled_ms = time.ticks_ms()
50 frame_id += 1
51 # In measure mode the camera MUST already be physically stationary.
52 result = task.poll(raw, frame_id, geometry_confirmed=True, stable=(mode == "measure"))
53 result["capture_return_ms"] = sampled_ms
54 preview = raw.to_rgb565()
55 if mode == "scan":
56 preview.draw_rectangle(SCAN_GATE_RAW_PX, color=(255,255,0), thickness=2)
57 elif result.get("position_camera_mm") is not None:
58 u, v = task.calibration.camera.project_ray(result["position_camera_mm"])
59 preview.draw_cross(int(u),int(v),color=(0,255,0),size=12,thickness=2)
60 Display.show_image(preview)
61 now = time.ticks_ms()
62 if time.ticks_diff(now,last_print) >= 500:
63 print("AIM",frame_id,result["reason"],"p_E_mm",result.get("position_tool_mm"),
64 "slot",result.get("target_slot"))
65 last_print = now
66 if result.get("complete") or result.get("measurement_complete"):
67 print("AIM_RESULT",result)
68 break
69 raw = preview = None
70 gc.collect()
71 else:
72 result = {"ok":False,"reason":"TARGET_TASK_FRAME_BUDGET_EXCEEDED",
73 "position_tool_mm":None,"control_ready":False}
74 return result
75 finally:
76 raw = preview = None
77 try:
78 task.close()
79 finally:
80 try:
81 if sensor is not None: sensor.stop()
82 finally:
83 try:
84 if display_touched: Display.deinit()
85 finally:
86 gc.collect()
87 os.exitpoint(os.EXITPOINT_ENABLE_SLEEP)
88 time.sleep_ms(100)
89 if media_touched: MediaManager.deinit()
90
91
92if __name__ == "__main__":
93 main()
