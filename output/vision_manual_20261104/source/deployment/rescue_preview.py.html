1"""Manual LCD check for the rescue model. Run this separately from competition_main."""
2import os,sys,time,gc
3PROJECT_ROOT='/data/guanggongsai'
4if PROJECT_ROOT not in sys.path:sys.path.insert(0,PROJECT_ROOT)
5import runtime_config as cfg
6from calibration_params import load_calibration
7from task_runner import VisionTaskFactory
8from camera_service import CameraService
9from rescue_config import SHAPE_NAMES,SCAN_GATE_RAW_PX
10
11def main(max_frames=None):
12 os.exitpoint(os.EXITPOINT_ENABLE)
13 camera=factory=frame=preview=None
14 count=0
15 try:
16 cal=load_calibration(PROJECT_ROOT,handeye_accepted_override=cfg.HANDEYE_ACCEPTED_BY_USER)
17 factory=VisionTaskFactory(cal,PROJECT_ROOT)
18 detector=factory._model('rescue')
19 camera=CameraService()
20 while max_frames is None or count<max_frames:
21 os.exitpoint()
22 frame=camera.get_frame()
23 if frame is None:
24 if camera.state=='ERROR':raise RuntimeError(camera.error)
25 time.sleep_ms(5)
26 continue
27 started=time.ticks_ms()
28 detection=detector.detect(frame['image'])
29 elapsed=time.ticks_diff(time.ticks_ms(),started)
30 if not detection['ok']:raise RuntimeError(detection['reason'])
31 preview=frame['image'].to_rgb565()
32 labels=[]
33 for item in detection['detections']:
34 x,y,w,h=[int(v) for v in item['bbox_raw_px']]
35 shape=item['shape_id']
36 color=((255,80,80),(80,255,80),(80,160,255))[shape-1]
37 preview.draw_rectangle(x,y,w,h,color=color,thickness=2)
38 text='%d %s %.2f'%(shape,SHAPE_NAMES[shape],item['score'])
39 preview.draw_string_advanced(max(0,x),max(0,y-20),18,text,color=color)
40 labels.append(text)
41 camera.render_debug(preview,{'lines':['救援模型检查','帧:%d'%frame['frame_id'],
42 '推理:%d ms'%elapsed,'目标:%d'%len(labels),'1:圆柱','2:圆台','3:腰鼓']+labels,
43 'scan_gate':SCAN_GATE_RAW_PX})
44 if camera.display_error:raise RuntimeError(camera.display_error)
45 if count%10==0:print('RESCUE_PREVIEW',frame['frame_id'],elapsed,labels)
46 count+=1
47 frame=preview=None
48 gc.collect()
49 except KeyboardInterrupt:
50 print('RESCUE_PREVIEW_STOPPED')
51 finally:
52 frame=preview=None
53 try:
54 if factory is not None:factory.close()
55 finally:
56 if camera is not None:camera.close()
57 os.exitpoint(os.EXITPOINT_ENABLE_SLEEP)
58 time.sleep_ms(100)
59 gc.collect()
60
61if __name__=='__main__':main()
