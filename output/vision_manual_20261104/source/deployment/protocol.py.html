13QR_START = 0x10
14BOMB_SCAN = 0x20
15BOMB_MEASURE = 0x21
16AIM_SCAN = 0x30
17AIM_MEASURE = 0x31
18RESCUE_SCAN = 0x40
19RESCUE_MEASURE = 0x41
20STOP = 0x7F
21RESULT = 0x81
22HEARTBEAT = 0x02
23STATUS = 0x02 # K230 -> MCU: framed TOTALSTART!/TOTALSTOP! payloads.
24START_COMMANDS = (QR_START, BOMB_SCAN, BOMB_MEASURE, AIM_SCAN, AIM_MEASURE, RESCUE_SCAN, RESCUE_MEASURE)
25SCAN_COMMANDS = (BOMB_SCAN, AIM_SCAN, RESCUE_SCAN)
26MEASURE_COMMANDS = (QR_START, BOMB_MEASURE, AIM_MEASURE, RESCUE_MEASURE)
27NAMES = {QR_START:"QR",BOMB_SCAN:"BOMB_SCAN",BOMB_MEASURE:"BOMB_MEASURE",
28 AIM_SCAN:"AIM_SCAN",AIM_MEASURE:"AIM_MEASURE",RESCUE_SCAN:"RESCUE_SCAN",
29 RESCUE_MEASURE:"RESCUE_MEASURE",STOP:"STOP"}

31SUCCESS, FAILED, CANCELLED, OBSERVATION_ONLY = 0, 1, 2, 3
32POSITION, CONTROL_READY, QUATERNION, NORMAL = 1, 2, 4, 8
33GRASP, TASK_DIGITS, SLOT_MAPPING, CANDIDATE_CALIBRATION = 16, 32, 64, 128

47def reason_code(reason):
48 if reason in REASONS: return REASONS[reason]
49 if reason.startswith("MODEL_") or reason.startswith("TEST_BACKEND"): return REASONS["MODEL_CONFIGURATION_ERROR"]
50 if reason.startswith("HANDEYE_") or reason.startswith("CALIBRATION_"): return REASONS["CALIBRATION_NOT_READY"]
51 if reason.startswith("TABLE_") or reason.startswith("SLOT_"): return REASONS["TABLE_CONFIGURATION_ERROR"]
52 if reason.startswith("SCAN_"): return REASONS["SCAN_AMBIGUOUS"]
53 if reason.startswith("CAMERA_"): return REASONS["CAMERA_ERROR"]
54 return REASONS["VISION_ERROR"]

57def decode_command(frame):
58 cmd, payload = frame.cmd, frame.payload
59 if cmd == HEARTBEAT: return None
60 if cmd not in START_COMMANDS+(STOP,): raise ValueError("UNKNOWN_COMMAND")
61 allowed = (0,) if cmd in (QR_START,STOP) else ((0,1) if cmd in SCAN_COMMANDS else (0,1,2))
62 if len(payload) not in allowed: raise ValueError("BAD_COMMAND_PAYLOAD")
63 target = payload[0] if payload else 0
64 slot = payload[1] if len(payload)==2 else 0
65 if not 0 <= target <= 3 or not 0 <= slot <= 3: raise ValueError("BAD_COMMAND_PAYLOAD")
66 return {"cmd":cmd,"input_seq":frame.seq,"target_id":target or None,"slot_id":slot or None,
67 "key":(frame.seq,cmd,bytes(payload))}

70def _pack_vector(values, size):
71 if len(values)!=size or not all(finite(v) for v in values): raise ValueError("NONFINITE_RESULT")
72 return struct.pack("<"+"f"*size,*values)

75def encode_result(request, run_id, status, reason, observation=None):
76 obs = observation or {}
77 flags = 0
78 position = obs.get("position_tool_mm")
79 carries_data = status in (SUCCESS, OBSERVATION_ONLY)
80 if carries_data and position is not None:
81 flags |= POSITION
82 if obs.get("control_ready") is True: flags |= CONTROL_READY
83 if obs.get("handeye_accepted") is False: flags |= CANDIDATE_CALIBRATION
84 if obs.get("grasp_position_tool_mm") is not None: flags |= GRASP
85 # The MCU only needs the chosen slot and the measured pose. The task's
86 # color/shape identity stays in factory.selection and factory.maps.
87 target = 0
88 slot = obs.get("target_slot",obs.get("slot_id",request.get("slot_id"))) or 0
89 count = obs.get("sample_count",0)
90 if any(type(v) is not int or not 0 <= v <= 3 for v in (target,slot)): raise ValueError("INVALID_RESULT_ID")
91 if type(count) is not int or not 0 <= count <= 255: raise ValueError("INVALID_RESULT_SAMPLE_COUNT")
92 frame_id = obs.get("frame_id",0) or 0
93 header = struct.pack(HEADER_FORMAT,VERSION,request["cmd"],request["input_seq"],status,
94 flags,target,slot,count,run_id & 0xffff,frame_id & 0xffffffff,reason_code(reason))
95 body = b""
96 if flags & POSITION:
97 body += _pack_vector(position,3)
98 if flags & GRASP: body += _pack_vector(obs["grasp_position_tool_mm"],3)
99 body += _pack_vector([obs.get("scatter_mm",0.0)],1)
100 if len(header+body)>64: raise ValueError("RESULT_PAYLOAD_TOO_LONG")
101 return header+body

104def decode_result(payload):
105 """Host/debug decoder using the exact documented byte layout."""
106 if len(payload)<HEADER_SIZE: raise ValueError("SHORT_RESULT")
107 values=struct.unpack(HEADER_FORMAT,payload[:HEADER_SIZE])
108 version,cmd,seq,status,flags,target,slot,count,run_id,frame_id,reason=values
109 if version!=VERSION: raise ValueError("RESULT_VERSION_MISMATCH")
110 result={"task_cmd":cmd,"input_seq":seq,"status":status,"flags":flags,"target_id":target,
111 "slot_id":slot,"sample_count":count,"run_id":run_id,"frame_id":frame_id,"reason_code":reason}
112 offset=HEADER_SIZE
113 def read_floats(size):
114 nonlocal offset
115 end=offset+size*4
116 if end>len(payload): raise ValueError("SHORT_RESULT")
117 data=struct.unpack("<"+"f"*size,payload[offset:end]); offset=end
118 return list(data)
119 if flags & POSITION:
120 result["position_tool_mm"]=read_floats(3)
121 if flags & QUATERNION: result["quaternion_tool_from_qr"]=read_floats(4)
122 if flags & NORMAL: result["normal_tool"]=read_floats(3)
123 if flags & GRASP: result["grasp_position_tool_mm"]=read_floats(3)
124 if flags & TASK_DIGITS:
125 if offset+3>len(payload): raise ValueError("SHORT_RESULT")
126 result["task_digits"]=list(payload[offset:offset+3]); offset+=3
127 result["scatter_mm"]=read_floats(1)[0]
128 if flags & SLOT_MAPPING:
129 if offset+3>len(payload): raise ValueError("SHORT_RESULT")
130 result["slot_mapping"]=list(payload[offset:offset+3]); offset+=3
131 if offset!=len(payload): raise ValueError("RESULT_LENGTH_MISMATCH")
132 return result
