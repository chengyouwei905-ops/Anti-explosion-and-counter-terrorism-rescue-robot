41 def _measurement_snapshot(self):
42 if self.request is None or self.request["cmd"] in p.SCAN_COMMANDS:return None
43 return {"cmd":self.request["cmd"],"input_seq":self.request["input_seq"],
44 "target_id":self.request.get("target_id"),"slot_id":self.request.get("slot_id"),
45 "elapsed_ms":self._elapsed(),"frames":self.measurement_frames,
46 "samples":len(self.average.samples) if self.average is not None else 0,
47 "peak_samples":self.measurement_peak_samples,"last_state":self.last_reason,
48 "last_error":self.last_measurement_error,
49 "rejections":dict(self.measurement_rejections)}

51 def _record_measurement_reason(self,reason):
52 count=len(self.average.samples) if self.average is not None else 0
53 self.measurement_peak_samples=max(self.measurement_peak_samples,count)
54 if reason not in ("OK","MEASUREMENT_STABILIZING"):

57 key=reason if reason in self.measurement_rejections or len(self.measurement_rejections)<8 else "OTHER"

74 def pause(self):
75 """Freeze the active request; never transmit a cancellation/old pose here."""
76 if self.paused: return
77 self.paused=True
78 self.state=PAUSED
79 if self.request is not None:
80 self._elapsed_before_pause+=self.diff(self.clock(),self.started_ms)
81 if self.average is not None: self.average.reset()
82 if self.job is not None:
83 method=getattr(self.job,"pause_window",self.job.reset_window)
84 method()
85 # Scan results contain identities/slots, not a stale measured pose.
86 if (self.pending is not None and self.pending[0] in (p.SUCCESS,p.OBSERVATION_ONLY)
87 and self.job is not None and not self.job.is_scan):
88 self.pending=None

122 def _seen(self,key,now):
123 self.history=[(k,t) for k,t in self.history if self.diff(now,t)<cfg.REQUEST_DEDUP_MS]
124 if self.request is not None and self.request["key"]==key: return True
125 return any(k==key for k,_ in self.history)

127 def _remember(self,key,now):

131 def _transmit_once(self,request,run_id,status,reason,observation=None):
132 self.report_attempts+=1
133 self.last_report=None
134 try:
135 try:
136 payload=p.encode_result(request,run_id,status,reason,observation)
137 except Exception:
138 status,reason,observation=p.FAILED,"VISION_ERROR",None
139 payload=p.encode_result(request,run_id,status,reason)
140 print("K230_RESULT",p.NAMES.get(request["cmd"],request["cmd"]),"seq",request["input_seq"],"status",status,"reason",reason)
141 self.link.send(p.RESULT,payload)
142 self.last_report={"run_id":run_id,"request":dict(request),"status":status,"reason":reason,

178 def on_command(self,frame):
179 if self.paused: return
180 if frame.cmd==p.HEARTBEAT: return
181 now=self.clock()
182 key=(frame.seq,frame.cmd,bytes(frame.payload))
183 if self._seen(key,now): return
184 self._remember(key,now)
185 try:
186 request=p.decode_command(frame)
187 except ValueError as exc:
188 invalid={"cmd":frame.cmd,"input_seq":frame.seq,"target_id":None,"slot_id":None,"key":key}
189 self._transmit_once(invalid,self._new_run_id(),p.FAILED,str(exc))
190 return
191 if request["cmd"]==p.STOP:
192 if self.request is not None: self._finish_active(p.CANCELLED,"CANCELLED")
193 return # STOP while idle is intentionally silent.
194 if self.request is not None: self._finish_active(p.CANCELLED,"PREEMPTED")
195 self.request=request
196 self._reset_measurement_diagnostics()
197 self.last_measurement_diagnostic=None
198 self.run_id=self._new_run_id()
199 self.started_ms=now
200 self._settle_started_ms=now
201 self._elapsed_before_pause=0
202 self.timeout_ms=cfg.SCAN_TIMEOUT_MS if request["cmd"] in p.SCAN_COMMANDS else cfg.MEASUREMENT_TIMEOUT_MS
203 self.average=MeasurementAverage(cfg.MEASUREMENT_SAMPLES,cfg.MAX_POSITION_SCATTER_MM)
204 if request["cmd"]==p.QR_START and cfg.QR_SAMPLE_WINDOW_MS>0:
205 self.average=QRMeasurementAverage(cfg.MEASUREMENT_SAMPLES,cfg.MAX_POSITION_SCATTER_MM,
206 cfg.QR_SAMPLE_WINDOW_MS,
207 self.clock,self.diff)
208 self.pending=None
209 self.last_scan_error=None
210 self.last_scan_order=[]
211 self.last_reason="STARTING"
212 try:
213 self.job=self.factory.start(request)
214 self.state=RUNNING if self.job.is_scan else SETTLING
215 except Exception as exc:
216 self.job=None
217 self.pending=(p.FAILED,str(exc),None)
218 self.state=REPORT

220 def tick(self):
221 if self.state not in (SETTLING,RUNNING): return
222 elapsed=self._elapsed()
223 if elapsed>=self.timeout_ms:
224 self.pending=(p.FAILED,"TIMEOUT",None)
225 self.state=REPORT
226 elif self.state==SETTLING and self.diff(self.clock(),self._settle_started_ms)>=cfg.SETTLE_MS:
227 self.state=RUNNING
228 self.log_measurement_state()

244 def poll_frame(self,frame):
245 if self.state!=RUNNING or self.job is None: return
246 if self.last_frame_id is not None and frame["frame_id"]<=self.last_frame_id: return
247 self.last_frame_id=frame["frame_id"]
248 self.observe_camera_epoch(frame["epoch"])
249 if self.state!=RUNNING: return
250 if not self.job.is_scan:self.measurement_frames+=1
251 try:
252 result=self.job.poll(frame)
253 if self._elapsed()>=self.timeout_ms:
254 if not self.job.is_scan:
255 self.last_reason=result.get("reason","UNKNOWN_MEASUREMENT") if not result.get("ok") else "MEASUREMENT_AFTER_TIMEOUT"
256 self._record_measurement_reason(self.last_reason)
257 self.pending=(p.FAILED,"TIMEOUT",None); self.state=REPORT; return
258 if self.job.is_scan:
259 reason=result.get("reason","SCANNING")
260 order=result.get("observed_order",result.get("observed_shape_order"))
261 if order is not None and list(order)!=self.last_scan_order:
262 self.last_scan_order=list(order)
263 print("K230_SCAN_PROGRESS",p.NAMES[self.request["cmd"]],"seq",self.request["input_seq"],"order",self.last_scan_order)
264 ordinary=("OK","SCANNING","SCAN_WAIT_TARGET","SCAN_STABILIZING","SCAN_TARGET_RECORDED","SCAN_ALREADY_COUNTED")
265 if reason not in ordinary:
266 if self.last_scan_error!=reason:
267 print("K230_SCAN_DIAGNOSTIC",p.NAMES[self.request["cmd"]],"seq",self.request["input_seq"],"reason",reason,"order",self.last_scan_order)
268 self.last_scan_error=reason
269 if result.get("ok") and result.get("complete"):
270 self.pending=(p.SUCCESS,"OK",result)
271 self.state=REPORT
272 elif terminal_reason(result.get("reason","")):
273 self.pending=(p.FAILED,result["reason"],None)
274 self.state=REPORT
275 else: self.last_reason=result.get("reason","SCANNING")
276 return
277 if isinstance(self.average,QRMeasurementAverage):
278 averaged,reason=self.average.add(result,frame.get("capture_return_ms"))
279 else:
280 averaged,reason=self.average.add(result)
281 self.last_reason=reason
282 self._record_measurement_reason(reason)
283 if averaged is not None:
284 accepted=averaged.get("control_ready") is True
285 if not accepted and not cfg.ALLOW_CANDIDATE_REPORTS:
286 self.pending=(p.FAILED,"CALIBRATION_NOT_READY",None)
287 else:
288 self.pending=(p.SUCCESS if accepted else p.OBSERVATION_ONLY,
289 "OK" if accepted else "HANDEYE_CANDIDATE_ONLY",averaged)
290 self.state=REPORT
291 elif terminal_reason(reason):
292 self.pending=(p.FAILED,reason,None)
293 self.state=REPORT
294 except Exception as exc:
295 self.pending=(p.FAILED,"VISION_ERROR",None)
296 self.last_reason=type(exc).__name__+":"+str(exc)
297 if not self.job.is_scan:self._record_measurement_reason(self.last_reason)
298 self.state=REPORT

300 def flush_report(self):
301 # Main pumps RX AGAIN before calling this, including after heavy inference.
302 if self.state!=REPORT or self.pending is None: return
303 status,reason,observation=self.pending
304 if status in (p.SUCCESS,p.OBSERVATION_ONLY) and observation is not None:
305 try: self.factory.completed(self.request,observation)
306 except Exception:
307 status,reason,observation=p.FAILED,"VISION_ERROR",None
308 self._finish_active(status,reason,observation)
