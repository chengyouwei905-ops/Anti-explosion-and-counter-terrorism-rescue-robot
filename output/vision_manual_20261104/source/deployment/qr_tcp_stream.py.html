1"""K230 独立调试：持续回传二维码中心在夹爪 TCP/E 坐标系中的位置。
2
3部署：将本文件放到 /data/guanggongsai/qr_tcp_stream.py，单独运行它。
4也可在板上 REPL 执行：from qr_tcp_stream import main; main()
5不要与 competition_main.py 同时运行；本文件依赖原目录中的模块和标定文件。
6二维码仍使用原 QRTask：边长由 config.QR_SIZE_MM 指定（当前80mm），内容
7必须是三个1/2/3数字，例如123；画面有多个二维码时不发送有效坐标。
8
9坐标：p_E = T_E_from_C * p_C，单位mm；位置是“二维码中心在夹爪夹取中心
10坐标系下的XYZ”，不是夹爪在二维码系的位置，也不是底盘/机械臂基座坐标。
11不读取、改变机械臂位姿，不重新标定，不改正式任务入口，不发TOTALSTART!。
12高分辨率开关在runtime_config.QR_HIGH_RES_ENABLED：启用时仅二维码读取
131920x1080灰度通道，VGA彩色预览与YOLO保持原样。QRTask按同一完整视场
14推导高分辨率内参，需上板复核距离/多姿态一致性；不对1080图直接套VGA内参。
15
16默认启动即连续测量。用户键GPIO53仅在本脚本内切换暂停/继续，不发送总启停。
17换姿态时：暂停 -> 改机械臂姿态 -> 到位停稳 -> 继续 -> 记录多组数据。
18K230不知道机械臂是否真正停稳，3帧散布检查也不能代替物理停稳。
19每次启动/继续、相机重启、识别无效及每次完成一批测量，均清空采样窗口。
20仅发送新采集的有效批次；失去二维码时不重复发送上次坐标，不发送假零坐标。
21
22串口复用runtime_config：UART2，TX=11，RX=12，115200，8N1。
23复用VisionLink.pack/send和protocol.encode_result，完整帧为：
24 AA 55 LEN SEQ 81 [原15字节结果头][XYZ][scatter] CRC8
25正常接受标定时：LEN=31，完整37字节，task_cmd=0x10，status=0，flags=0x03，
26slot_id=0，sample_count=3；浮点小端float32。scatter位于payload[27..30]。
27手眼接受状态沿用runtime_config.HANDEYE_ACCEPTED_BY_USER；候选标定只会发
28原协议OBSERVATION_ONLY状态，不擅自提升为可控制结果。
29
30这是“连续观察”调试模式，数据布局与正式协议一致，连续发送语义不同：
31不必先收到MCU请求；初始input_seq=0，收到空载荷0x10后回显其SEQ并继续发送。
32收到空载荷0x7F暂停发送；再次0x10或按用户键继续。外层SEQ由VisionLink自增。
33默认两次发送至少间隔1秒；SEND_INTERVAL_MS仅限制最短间隔，真实频率还受
343帧定位耗时限制。若定位较慢则延后发送，不重发旧坐标来凑固定频率。
35
36MCU配合要点（本文件不会修改MCU）：
371. 当前task.c空闲时忽略0x81，作业层也只接收一次匹配请求。请在独立调试
38 接收分支调用proto_vision_decode_result，不用正式任务状态机驱动底盘。
39 如调用proto_vision_result_control_ready，应在该调试分支使用3U、8.0f；
40 正式流程和本脚本均采3帧，MCU的验收也应同步改为3U。
412. 对每个已停稳机械臂姿态，用对应的当前关节角/末端位姿生成T_B_from_E：
42 alg_vision_transform_from_arm_joints 或 alg_vision_transform_from_arm_pose。
43 再算 p_B[i] = t[i] + sum(R[i][j] * result.xyz_mm[j])。
44 不要一直用task_work_load_config内固定二维码观察姿态的body_from_tcp。
453. 现有B系是X右/Y前/Z上，而机械臂位置指令+X向左；现有transform函数已
46 处理此差异。比较p_B时统一坐标系，不要再手工翻转一次X。
474. 底盘与二维码固定，机械臂每次到位后保持不动，比较各姿态下p_B均值和
48 波动。舵机指令值不等于位置反馈，应先确认实际到位。不要边动臂边比较。

80class QRStream:
81 """无硬件初始化的流会话；测量、暂停和回包可离线验证。"""
82
83 def __init__(self, task, link, clock=ticks_ms, diff=ticks_diff):
84 self.task, self.link = task, link
85 self.clock, self.diff = clock, diff
86 self.average = MeasurementAverage(SAMPLE_COUNT, MAX_SCATTER_MM)
87 self.active = False
88 self.generation = 0
89 self.run_id = 0
90 self.input_seq = INITIAL_INPUT_SEQ
91 self.started_ms = clock()
92 self.last_send_ms = None
93 self.epoch = None
94 self.last_reason = "PAUSED"
95 self.last_observation = None
96 self.sent_count = 0
97
98 def resume(self, input_seq=None):
99 if input_seq is not None:
100 self.input_seq = input_seq
101 self.active = True
102 self.generation += 1
103 self.run_id = (self.run_id + 1) & 0xffff
104 self.average.reset()
105 self.started_ms = self.clock()
106 self.last_reason = "SETTLING"
107
108 def pause(self):
109 self.active = False
110 self.generation += 1
111 self.average.reset()
112 self.last_reason = "PAUSED"
113
114 def process_frame(self, frame):
115 if frame["epoch"] != self.epoch:
116 self.epoch = frame["epoch"]
117 self.average.reset()
118 self.started_ms = self.clock()
119 if not self.active:
120 return None
121 now = self.clock()
122 if self.diff(now, self.started_ms) < SETTLE_MS:
123 self.last_reason = "SETTLING"
124 return None
125 if self.last_send_ms is not None and self.diff(now, self.last_send_ms) < SEND_INTERVAL_MS:
126 self.last_reason = "WAIT_NEXT_BATCH"
127 return None
128 gray = frame.get("qr_image")
129 geometry = frame["geometry_confirmed"]
130 if gray is None:
131 gray = frame["image"].to_grayscale()
132 else:
133 geometry = frame.get("qr_geometry_confirmed", False)
134 try:
135 # 操作者须保持机械臂不动；此声明不是机械运动反馈。
136 observed = self.task.measure(gray, frame["frame_id"],
137 geometry_confirmed=geometry, stable=True)
138 finally:
139 gray = None
140 observed["reference_point"] = "qr_center"
141 observed["slot_id"] = 0
142 observation, self.last_reason = self.average.add(observed)
143 if observation is not None:
144 # 批次互不重叠，不把不同机械臂姿态的旧样本带到下一批。
145 self.average.reset()
146 return observation
147
148 def send_observation(self, observation):
149 if not self.active:
150 return False
151 ready = observation.get("control_ready") is True
152 status = protocol.SUCCESS if ready else protocol.OBSERVATION_ONLY
153 reason = "OK" if ready else observation.get("reason", "HANDEYE_CANDIDATE_ONLY")
154 request = {"cmd": protocol.QR_START, "input_seq": self.input_seq, "slot_id": 0}
155 payload = protocol.encode_result(request, self.run_id, status, reason, observation)
156 self.link.send(protocol.RESULT, payload)
157 self.last_send_ms = self.clock()
158 self.last_observation = observation
159 self.sent_count += 1
160 self.last_reason = "SENT" if ready else "OBSERVATION_ONLY"
161 point = observation["position_tool_mm"]
162 print("QR_TCP_TX", self.sent_count, "frame", observation["frame_id"],
163 "input_seq", self.input_seq, "status", status,
164 "p_E_mm=(%.3f, %.3f, %.3f)" % (point[0], point[1], point[2]),
165 "samples", observation["sample_count"],
166 "scatter_mm=%.3f" % observation["scatter_mm"])
167 return True

170def pump_controls(link, stream, button=None):
171 if not link.healthy():
172 raise OSError("QR_STREAM_LINK_ERROR:" + str(link.diagnostics()))
173 if button is not None:
174 if not button.healthy():
175 raise OSError("QR_STREAM_BUTTON_ERROR:" + str(button.error))
176 while button.pressed():
177 if stream.active:
178 stream.pause()
179 else:
180 stream.resume()
181 print("QR_STREAM", "RUNNING" if stream.active else "PAUSED")
182 stop = link.pop_priority()
183 if stop is not None:
184 if not stop.payload:
185 stream.pause()
186 print("QR_STREAM_PAUSED_BY_MCU")
187 else:
188 print("QR_STREAM_BAD_STOP_PAYLOAD")
189 return
190 if link.take_overflow() is not None:
191 raise OSError("QR_STREAM_COMMAND_QUEUE_OVERFLOW")
192 for _ in range(cfg.COMMANDS_PER_POLL):
193 frame = link.pop_command()
194 if frame is None:
195 break
196 if frame.cmd == protocol.QR_START and not frame.payload:
197 stream.resume(frame.seq)
198 print("QR_STREAM_REQUEST", frame.seq)
199 else:
200 print("QR_STREAM_IGNORED_CMD", frame.cmd)

203def render_preview(camera, stream, image=None):
204 lines = ["QR坐标持续回传", "运行中" if stream.active else "已暂停",
205 "坐标系:夹爪E", "单位:mm", "已发送:%d" % stream.sent_count,
206 "采样:%d/%d" % (len(stream.average.samples), SAMPLE_COUNT), stream.last_reason]
207 lines.insert(2,"QR:%dx%d" % cfg.QR_CAPTURE_SIZE if cfg.QR_HIGH_RES_ENABLED else "QR:640x480")
208 observation = stream.last_observation
209 if observation is not None:
210 age = stream.diff(stream.clock(), stream.last_send_ms)
211 lines.append("上次发送:%dms前" % age)
212 lines.extend(["%s:%+.1f" % (axis, value)
213 for axis, value in zip("XYZ", observation["position_tool_mm"])])
214 lines.append("散布:%.2f" % observation["scatter_mm"])
215 camera.render_debug(image, {"lines": lines})

218def run_loop(link, stream, camera, button=None, max_loops=None,
219 exitpoint=None, sleep_ms=_sleep_ms):
220 loops = 0
221 last_preview = last_log = stream.clock()
222 frame = None
223 while max_loops is None or loops < max_loops:
224 loops += 1
225 if exitpoint is not None:
226 exitpoint()
227 pump_controls(link, stream, button)
228 select_qr = getattr(camera, "set_qr_capture", None)
229 if select_qr is not None:
230 select_qr(stream.active)
231 generation = stream.generation
232 frame = camera.get_frame()
233 if camera.state == "ERROR":
234 raise OSError(camera.error or "CAMERA_ERROR")
235 pump_controls(link, stream, button)
236 if frame is None:
237 # 相机预热/恢复期间也不能保留此前的半批坐标。
238 stream.average.reset()
239 stream.last_reason = "CAMERA_" + camera.state if stream.active else "PAUSED"
240 elif generation == stream.generation:
241 observation = stream.process_frame(frame)
242 # 检测可能耗时；其间收到暂停/新请求时，丢弃这一旧批次。
243 pump_controls(link, stream, button)
244 if observation is not None and generation == stream.generation and stream.active:
245 stream.send_observation(observation)
246 now = stream.clock()
247 if stream.diff(now, last_preview) >= PREVIEW_INTERVAL_MS:
248 render_preview(camera, stream, frame["image"] if frame is not None else None)
249 last_preview = now
250 if stream.diff(now, last_log) >= LOG_INTERVAL_MS:
251 print("QR_STREAM_STATE", stream.last_reason,
252 "samples", len(stream.average.samples), "sent", stream.sent_count)
253 last_log = now
254 frame = None
255 gc.collect()
256 sleep_ms(cfg.LOOP_SLEEP_MS)

259def main(max_loops=None):
260 link = camera = button = stream = None
261 try:
262 calibration = load_calibration(cfg.PROJECT_ROOT,
263 handeye_accepted_override=cfg.HANDEYE_ACCEPTED_BY_USER)
264 if calibration.T_tool_from_camera is None:
265 raise ValueError("HANDEYE_NOT_CONFIGURED")
266 link = VisionLink(uart_id=cfg.UART_ID, baudrate=cfg.UART_BAUDRATE,
267 tx_pin=cfg.UART_TX_PIN, rx_pin=cfg.UART_RX_PIN,
268 queue_size=cfg.COMMAND_QUEUE_SIZE, priority_cmds=(protocol.STOP,),
269 read_chunk=cfg.RX_READ_BYTES)
270 camera = CameraService(qr_enabled=cfg.QR_HIGH_RES_ENABLED)
271 if ENABLE_USER_BUTTON:
272 from user_button import UserButton
273 button = UserButton()
274 task = QRTask(calibration, high_resolution_size=cfg.QR_CAPTURE_SIZE if cfg.QR_HIGH_RES_ENABLED else None)
275 stream = QRStream(task, link)
276 if AUTO_START:
277 stream.resume()
278 print("QR_TCP_STREAM", "UART", cfg.UART_ID, cfg.UART_BAUDRATE,
279 "TX", cfg.UART_TX_PIN, "RX", cfg.UART_RX_PIN,
280 "handeye_accepted", calibration.handeye_accepted)
281 print("QR_STREAM_SAMPLING", SAMPLE_COUNT, "SEND_INTERVAL_MS", SEND_INTERVAL_MS,
282 "MAX_SCATTER_MM", MAX_SCATTER_MM)
283 print("QR_TCP_STREAM: fixed base/QR; pause, move arm, settle, resume.")
284 print("QR_TCP_STREAM: MCU must use CURRENT arm pose; no TOTALSTART is sent.")
285 print("QR_CAPTURE_SIZE", cfg.QR_CAPTURE_SIZE if cfg.QR_HIGH_RES_ENABLED else (640,480),
286 "RGB_PREVIEW_YOLO_SIZE", (640,480))
287 render_preview(camera, stream)
288 run_loop(link, stream, camera, button, max_loops, getattr(os, "exitpoint", None))
289 except KeyboardInterrupt:
290 print("QR_TCP_STREAM_STOPPED")
291 finally:
292 if stream is not None:
293 stream.pause()
294 try:
295 if button is not None and not button.close():
296 print("BUTTON_SHUTDOWN_TIMEOUT")
297 finally:
298 try:
299 if camera is not None:
300 camera.close()
301 finally:
302 if link is not None and not link.deinit():
303 print("UART_SHUTDOWN_TIMEOUT")
304
305
306if __name__ == "__main__":
307 main()
