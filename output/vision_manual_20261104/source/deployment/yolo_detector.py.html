11def restore_bbox_xywh(box, transform, image_size):
12 """Inverse of actual raw -> network resize and padding, including rounding."""
13 if len(box) != 4 or not all(finite(v) for v in box) or min(box[2:]) <= 0:
14 raise ValueError("BAD_DETECTION_BOX")
15 if tuple(transform["raw_size"]) != tuple(image_size):
16 raise ValueError("PREPROCESS_RAW_SIZE_MISMATCH")
17 sx, sy = transform["scale_x"], transform["scale_y"]
18 px, py = transform["pad_x"], transform["pad_y"]
19 if not all(finite(v) for v in (sx, sy, px, py)) or min(sx, sy) <= 0:
20 raise ValueError("BAD_PREPROCESS_TRANSFORM")
21 x, y, w, h = box
22 x0, y0 = (x-px)/sx, (y-py)/sy
23 x1, y1 = (x+w-px)/sx, (y+h-py)/sy
24 width, height = image_size
25 clipped = x0 < 0 or y0 < 0 or x1 > width or y1 > height
26 x0, y0, x1, y1 = max(0.0, x0), max(0.0, y0), min(float(width), x1), min(float(height), y1)
27 if x1 <= x0 or y1 <= y0:
28 raise ValueError("DETECTION_IN_PADDING_ONLY")
29 return (x0, y0, x1-x0, y1-y0), clipped

32class YoloDetector:
33 """Backend load(metadata), infer(raw) and close() are supplied with the model.
34
35 infer returns {space, detections, transform}. space is raw_pixels or
36 network_pixels; each box is xywh. Backend owns preprocessing AND NMS.
37 transform is mandatory for network pixels, and describes the operation
38 actually performed, not an idealized width/height ratio.
39 """
40 def __init__(self, metadata, backend=None, allow_test_backend=False,
41 labels=None, id_field="color_id", source_name="yolo"):
42 self.metadata, self.backend = metadata, backend
43 self.labels = tuple(BOMB_LABELS if labels is None else labels)
44 if not self.labels or len(self.labels) > 32 or len(set(self.labels)) != len(self.labels) or any(not isinstance(v, str) or not v for v in self.labels):
45 raise ValueError("INVALID_MODEL_LABELS")
46 if id_field not in ("color_id", "shape_id") or not isinstance(source_name, str) or not source_name:
47 raise ValueError("INVALID_DETECTOR_SEMANTICS")
48 self.id_field, self.source_name = id_field, source_name
49 self.ready, self.reason = False, "MODEL_NOT_READY"
50 self.allow_test_backend = allow_test_backend
51
52 def load(self):
53 self.close()
54 path = self.metadata.get("model_path")
55 if not path:
56 self.reason = "MODEL_NOT_READY"
57 return _failure(self.reason)
58 try:
59 os.stat(path)
60 except OSError:
61 self.reason = "MODEL_NOT_READY"
62 return _failure(self.reason)
63 required = ("family", "input_size", "input_dtype", "tensor_layout", "color_order",
64 "normalization", "preprocess", "output_contract", "nncase_version")
65 if any(self.metadata.get(k) is None for k in required) or self.metadata.get("contract_verified") is not True:
66 self.reason = "MODEL_METADATA_NOT_VERIFIED"
67 return _failure(self.reason)
68 if tuple(self.metadata.get("labels", ())) != self.labels:
69 self.reason = "MODEL_CLASS_ORDER_MISMATCH"
70 return _failure(self.reason)
71 if self.backend is None:
72 self.reason = "MODEL_BACKEND_NOT_CONFIGURED"
73 return _failure(self.reason)
74 if getattr(self.backend, "is_test_backend", False) and not self.allow_test_backend:
75 self.reason = "TEST_BACKEND_DISABLED"
76 return _failure(self.reason)
77 try:
78 result = self.backend.load(self.metadata)
79 if not result or result.get("ok") is not True:
80 self.reason = "MODEL_LOAD_FAILED"
81 self.backend.close()
82 return _failure(self.reason)
83 except Exception:
84 self.reason = "MODEL_LOAD_FAILED"
85 self.backend.close()
86 return _failure(self.reason)
87 self.ready, self.reason = True, "OK"
88 return {"ok": True, "reason": "OK"}

90 def detect(self, raw):
91 if not self.ready:
92 return _failure(self.reason)
93 try:
94 size = (raw.width(), raw.height())
95 if size != tuple(self.metadata["raw_size"]):
96 return _failure("IMAGE_SIZE_MISMATCH")
97 output = self.backend.infer(raw)
98 items = output["detections"]
99 if len(items) > MAX_DETECTIONS:
100 return _failure("TOO_MANY_DETECTIONS")
101 detections, rejected = [], []
102 for item in items:
103 cls, score = item["class_id"], item["score"]
104 if type(cls) is not int or not 0 <= cls < len(self.labels) or not finite(score) or not 0 <= score <= 1:
105 raise ValueError("INVALID_DETECTION_CLASS_OR_SCORE")
106 if score < DETECTION_SCORE_MIN:
107 continue
108 try:
109 if output["space"] == "network_pixels":
110 box, clipped = restore_bbox_xywh(item["bbox_xywh"], output["transform"], size)
111 elif output["space"] == "raw_pixels":
112 identity = {"raw_size": size, "scale_x": 1.0, "scale_y": 1.0, "pad_x": 0.0, "pad_y": 0.0}
113 box, clipped = restore_bbox_xywh(item["bbox_xywh"], identity, size)
114 else:
115 raise ValueError("UNKNOWN_DETECTION_COORDINATES")
116 except ValueError as exc:
117 if str(exc)!="DETECTION_IN_PADDING_ONLY":raise
118 # A false positive wholly in letterbox padding is not an
119 # observed object and does not invalidate other detections.
120 rejected.append({"reason":str(exc),"class_id":cls})
121 continue
122 detections.append({"class_id": cls, "label": self.labels[cls], self.id_field: cls+1,
123 "score": score, "bbox_raw_px": box, "bbox_clipped": clipped})
124 return {"ok": True, "reason": "OK", "detections": detections,"rejected":rejected,
125 "source": "test_backend" if getattr(self.backend, "is_test_backend", False) else self.source_name}
126 except Exception as exc:
127 return _failure("DETECTION_ERROR:"+str(exc))

129 def close(self):
130 if self.ready and self.backend is not None:
131 self.backend.close()
132 self.ready = False
133 self.reason = "MODEL_NOT_READY"
