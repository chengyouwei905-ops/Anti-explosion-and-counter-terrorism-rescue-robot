1"""Four-target stationary field test, compatible with the existing MCU QR callback.
2
3All test packets deliberately use task_cmd=0x10, slot_id=0 as a DEBUG channel.
4XYZ is the actual selected target reference in calibrated E, never replaced
5by the expected value. This is not the formal competition command semantics.
6The MCU's qr_position_body then means CURRENT TARGET position in B.
7Run only one field_*_tcp_test.py at a time, separate from competition_main.
8"""

26class FieldTCPStream(QRStream):
27 def __init__(self,job,link,kind,target_id=1,sample_count=3,send_interval_ms=1000,
28 sample_window_ms=3000,arm_pose=None,body_reference=None,
29 clock=ticks_ms,diff=ticks_diff):
30 if kind not in PROFILES:raise ValueError("INVALID_FIELD_TEST_KIND")
31 if type(target_id) is not int or not 1<=target_id<=3:raise ValueError("INVALID_TARGET_ID")
32 if type(send_interval_ms) is not int or send_interval_ms<1000:
33 raise ValueError("FIELD_SEND_INTERVAL_MUST_BE_AT_LEAST_1000_MS")
34 super().__init__(job,link,clock,diff)
35 self.kind,self.target_id=kind,target_id
36 self.required_samples,self.send_interval_ms=sample_count,send_interval_ms
37 self.arm_pose=arm_pose
38 self.body_reference=PROFILES[kind]["body_reference"] if body_reference is None else body_reference
39 self.average=(QRMeasurementAverage(sample_count,8.,sample_window_ms,clock,diff)
40 if kind=="qr" and sample_window_ms>0 else MeasurementAverage(sample_count,8.))
41 self.debug_view=None
42 self.debug_pose_valid=None
43 enable=getattr(job,"enable_debug",None)
44 if enable is not None:enable()

46 def _clear_debug(self):
47 self.debug_view=None
48 self.debug_pose_valid=None
49
50 def _sync_debug(self):
51 snapshot=getattr(self.task,"debug_snapshot",None)
52 self.debug_view=snapshot() if snapshot is not None else None

54 def _preview_only(self,frame):
55 preview=getattr(self.task,"preview",None)
56 if preview is not None:preview(frame)

60 self.task.reset_window()

62 def resume(self,input_seq=None):
63 self._reset_job()
64 self._clear_debug()
65 self.last_observation=None
66 super().resume(input_seq)

68 def pause(self):

73 def process_frame(self,frame):
74 self._clear_debug()
75 if frame["epoch"]!=self.epoch:
76 self.epoch=frame["epoch"]
77 self.average.reset()
78 self._reset_job()
79 self.started_ms=self.clock()
80 if not self.active:return None
81 now=self.clock()
82 if self.diff(now,self.started_ms)<250:
83 self._preview_only(frame)
84 self.last_reason="SETTLING";return None
85 if self.last_send_ms is not None and self.diff(now,self.last_send_ms)<self.send_interval_ms:
86 self._preview_only(frame)
87 self.last_reason="WAIT_NEXT_BATCH";return None
88 observed=self.task.poll(frame)
89 self._sync_debug()
90 self.debug_pose_valid=observed.get("ok") is True
91 observed.setdefault("reference_point",PROFILES[self.kind]["reference_point"])
92 if isinstance(self.average,QRMeasurementAverage):
93 averaged,self.last_reason=self.average.add(observed,frame.get("capture_return_ms"))
94 else:
95 averaged,self.last_reason=self.average.add(observed)
96 if not observed.get("ok") and self.kind!="qr":self._reset_job()
97 if averaged is not None:
98 self.average.reset()
99 self._reset_job()
100 return averaged

102 def send_observation(self,observation):
103 if not self.active:return False
104 # Wire-only alias: preserve real task metadata in logs and samples.
105 wire=dict(observation)
106 wire["slot_id"]=0
107 wire.pop("target_slot",None)
108 wire.pop("normal_tool",None)
109 wire.pop("grasp_position_tool_mm",None)
110 wire.pop("quaternion_tool_from_qr",None)
111 ready=wire.get("control_ready") is True
112 status=p.SUCCESS if ready else p.OBSERVATION_ONLY
113 reason="OK" if ready else wire.get("reason","HANDEYE_CANDIDATE_ONLY")
114 request={"cmd":p.QR_START,"input_seq":self.input_seq,"slot_id":0}
115 payload=p.encode_result(request,self.run_id,status,reason,wire)
116 self.link.send(p.RESULT,payload)
117 self.last_send_ms=self.clock()
118 self.last_observation=observation
119 self.sent_count+=1
120 self.last_reason="SENT" if ready else "OBSERVATION_ONLY"
121 point=observation["position_tool_mm"]
122 print("FIELD_TCP_TX",self.kind,"target_id",self.target_id,"packet",self.sent_count,
123 "source_cmd",PROFILES[self.kind]["source_cmd"],"wire_cmd",p.QR_START,
124 "p_E_mm",tuple(point),"samples",observation["sample_count"],
125 "scatter_mm",observation["scatter_mm"])
126 if self.arm_pose is not None:
127 body=tcp_to_body(point,self.arm_pose)
128 print("BODY_USING_CONFIGURED_POSE_MM",tuple(body))
129 if self.body_reference is not None:
130 print("BODY_MINUS_CONFIGURED_REFERENCE_MM",
131 tuple(body[i]-self.body_reference[i] for i in range(3)))
132 return True

135def pump_controls(link,stream,button=None):
136 if not link.healthy():raise OSError("FIELD_LINK_ERROR:"+str(link.diagnostics()))
137 if button is not None:
138 if not button.healthy():raise OSError("FIELD_BUTTON_ERROR:"+str(button.error))
139 while button.pressed():
140 if stream.active:stream.pause()
141 else:stream.resume()
142 print("FIELD_STREAM",stream.kind,"RUNNING" if stream.active else "PAUSED")
143 stop=link.pop_priority()
144 if stop is not None:
145 if not stop.payload:stream.pause()
146 else:print("FIELD_BAD_STOP_PAYLOAD")
147 return
148 if link.take_overflow() is not None:raise OSError("FIELD_COMMAND_QUEUE_OVERFLOW")
149 for _ in range(cfg.COMMANDS_PER_POLL):
150 command=link.pop_command()
151 if command is None:break
152 if command.cmd==p.QR_START and not command.payload:
153 stream.resume(command.seq)
154 print("FIELD_MCU_REQUEST",command.seq,"reset samples",stream.kind)
155 else:print("FIELD_IGNORED_CMD",command.cmd)

158def render_preview(camera,stream,image=None,frame_id=None):
159 view=getattr(stream,"debug_view",None)
160 current=view is not None and frame_id is not None and view["frame_id"]==frame_id
161 detections=view["detections"] if current else []
162 selected=sum(item.get("selected",False) for item in detections)
163 pose_valid=getattr(stream,"debug_pose_valid",None) if current else None
164 lines=[PROFILES[stream.kind]["label"]+"测试","目标:"+target_label(stream.kind,stream.target_id),
165 "运行中" if stream.active else "已暂停",
166 "边界限制:"+("开启" if boundary_checks() else "关闭"),
167 "检测:%d 目标:%d" % (len(detections),selected),
168 "定位:"+("有效" if pose_valid is True else ("未通过" if pose_valid is False else "待采样")),
169 status_text(stream.last_reason),
170 "采样:%d/%d" % (len(stream.average.samples),stream.required_samples),
171 "已发送:%d" % stream.sent_count]
172 if current and view.get("error"):lines.append("检测反馈异常")
173 if stream.last_observation is not None:
174 age=stream.diff(stream.clock(),stream.last_send_ms)
175 lines.extend(["已发TCP(mm)","%dms前" % age])
176 lines.extend("%s:%+.1fmm" % (axis,value) for axis,value in
177 zip("XYZ",stream.last_observation["position_tool_mm"]))
178 status={"lines":lines,"detections":detections,
179 "source_size":view["source_size"] if current else (640,480),
180 "pose_valid":pose_valid,"clear_preview":image is None}
181 if stream.kind=="bomb":
182 from config import BOMB_DETECTION_ROI_RAW_PX
183 status["scan_gate"]=BOMB_DETECTION_ROI_RAW_PX
184 elif stream.kind=="aim":
185 from aim_config import TARGET_DETECTION_ROI_RAW_PX
186 status["scan_gate"]=TARGET_DETECTION_ROI_RAW_PX
187 camera.render_debug(image,status)

190def run_loop(link,stream,camera,button=None,max_loops=None,exitpoint=None,sleep_ms=_sleep_ms):
191 loops=0
192 last_preview=last_log=stream.clock()
193 while max_loops is None or loops<max_loops:
194 loops+=1
195 if exitpoint is not None:exitpoint()
196 pump_controls(link,stream,button)
197 camera.set_qr_capture(stream.active and stream.kind=="qr")
198 generation=stream.generation
199 frame=camera.get_frame()
200 if camera.state=="ERROR":raise OSError(camera.error or "CAMERA_ERROR")
201 pump_controls(link,stream,button)
202 if frame is None:
203 stream.average.reset()
204 stream._reset_job()
205 stream._clear_debug()
206 stream.last_reason="CAMERA_"+camera.state if stream.active else "PAUSED"
207 elif generation==stream.generation:
208 observed=stream.process_frame(frame)
209 pump_controls(link,stream,button)
210 if observed is not None and generation==stream.generation and stream.active:
211 stream.send_observation(observed)
212 now=stream.clock()
213 if stream.diff(now,last_preview)>=200:
214 render_preview(camera,stream,frame["image"] if frame is not None else None,
215 frame["frame_id"] if frame is not None else None)
216 last_preview=now
217 if stream.diff(now,last_log)>=1000:
218 view=stream.debug_view or {}
219 detections=view.get("detections",[])
220 print("FIELD_TCP_STATE",stream.kind,stream.last_reason,
221 "detected",len(detections),"target",sum(d.get("selected",False) for d in detections),
222 "samples",len(stream.average.samples),"sent",stream.sent_count)
223 last_log=now
224 frame=None
225 gc.collect()
226 sleep_ms(cfg.LOOP_SLEEP_MS)

229def main(kind="qr",target_id=1,arm_pose=None,body_reference=None,
230 sample_count=3,send_interval_ms=1000,sample_window_ms=3000,
231 auto_start=True,enable_user_button=True,max_loops=None,show_to_ide=True):
232 if kind not in PROFILES:raise ValueError("INVALID_FIELD_TEST_KIND")
233 if type(target_id) is not int or not 1<=target_id<=3:raise ValueError("INVALID_TARGET_ID")
234 link=camera=factory=job=button=stream=None
235 try:
236 calibration=load_calibration(cfg.PROJECT_ROOT,handeye_accepted_override=cfg.HANDEYE_ACCEPTED_BY_USER)
237 if calibration.T_tool_from_camera is None:raise ValueError("HANDEYE_NOT_CONFIGURED")
238 print("FIELD_TCP_DEBUG_ALIAS: all targets use wire task_cmd=0x10, slot_id=0")
239 print("No TOTALSTART; fixed chassis; pause, move arm, settle, resume.")
240 if arm_pose is None:
241 print("NO_ACTUAL_ARM_POSE_CONFIGURED: following TCP reference is a preset EXAMPLE only")
242 print("Set ARM_POSE_FOR_REFERENCE to the actual MCU vision_arm_pose for an applicable comparison")
243 print_reference(kind,arm_pose,body_reference,calibration)
244 factory=VisionTaskFactory(calibration,cfg.PROJECT_ROOT)
245 request={"cmd":PROFILES[kind]["source_cmd"],"input_seq":0,
246 "target_id":None if kind=="qr" else target_id,
247 "slot_id":None if kind=="qr" else 1}
248 job=factory.start(request)
249 link=VisionLink(uart_id=cfg.UART_ID,baudrate=cfg.UART_BAUDRATE,
250 tx_pin=cfg.UART_TX_PIN,rx_pin=cfg.UART_RX_PIN,
251 queue_size=cfg.COMMAND_QUEUE_SIZE,priority_cmds=(p.STOP,),read_chunk=cfg.RX_READ_BYTES)
252 camera=CameraService(qr_enabled=kind=="qr" and cfg.QR_HIGH_RES_ENABLED,display_to_ide=show_to_ide)
253 if enable_user_button:
254 from user_button import UserButton
255 button=UserButton()
256 stream=FieldTCPStream(job,link,kind,target_id,sample_count,send_interval_ms,
257 sample_window_ms,arm_pose,body_reference)
258 if auto_start:stream.resume()
259 print("FIELD_TCP_READY",kind,"target_id",target_id,"samples",sample_count,
260 "min_interval_ms",send_interval_ms)
261 run_loop(link,stream,camera,button,max_loops,getattr(os,"exitpoint",None))
262 except KeyboardInterrupt:
263 print("FIELD_TCP_STOPPED",kind)
264 finally:
265 try:
266 if stream is not None:stream.pause()
267 if job is not None:job.cancel()
268 finally:
269 try:
270 if button is not None and not button.close():print("BUTTON_SHUTDOWN_TIMEOUT")
271 finally:
272 try:
273 if factory is not None:factory.close()
274 finally:
275 try:
276 if camera is not None:camera.close()
277 finally:
278 if link is not None and not link.deinit():print("UART_SHUTDOWN_TIMEOUT")
279

281if __name__=="__main__":main()
