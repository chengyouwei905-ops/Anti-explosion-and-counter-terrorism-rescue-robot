6def letterbox_transform(raw_size,input_size,center=True):
7 w,h=raw_size;ow,oh=input_size
8 scale=min(ow/float(w),oh/float(h))
9 rw,rh=int(round(w*scale)),int(round(h*scale))
10 left=int(round((ow-rw)/2.0-0.1)) if center else 0
11 top=int(round((oh-rh)/2.0-0.1)) if center else 0
12 return {"raw_size":list(raw_size),"scale_x":rw/float(w),"scale_y":rh/float(h),
13 "pad_x":left,"pad_y":top,"padding":[top,oh-rh-top,left,ow-rw-left]}

16def _iou(a,b):
17 x=max(a[0],b[0]);y=max(a[1],b[1])
18 w=max(0.0,min(a[0]+a[2],b[0]+b[2])-x)
19 h=max(0.0,min(a[1]+a[3],b[1]+b[3])-y)
20 inter=w*h
21 return inter/max(a[2]*a[3]+b[2]*b[3]-inter,1e-9)

24def _prob(value):
25 value=float(value)
26 if not finite(value):raise ValueError("MODEL_NONFINITE_OUTPUT")
27 # Small quantization overshoots are permitted; logits require another decoder.
28 if value < -0.08 or value > 1.08:raise ValueError("MODEL_PROBABILITY_CONTRACT_MISMATCH")
29 return min(1.0,max(0.0,float(value)))

32def decode_outputs(outputs,metadata):
33 contract=metadata["output_contract"]
34 threshold=float(metadata.get("confidence_threshold",0.55))
35 limit=int(metadata.get("max_candidates",96));candidates=[]
36 classes=len(metadata["labels"])
37 def add(cls,score,cx,cy,w,h):
38 if score<threshold:return
39 values=(cx,cy,w,h)
40 if not all(finite(v) for v in values) or min(w,h)<=0:raise ValueError("MODEL_INVALID_BOX")
41 item={"class_id":int(cls),"score":float(score),"bbox_xywh":[float(cx-w/2),float(cy-h/2),float(w),float(h)]}
42 candidates.append(item)
43 if len(candidates)>limit:
44 candidates.sort(key=lambda r:r["score"],reverse=True)
45 del candidates[limit:]
46 if contract["kind"]=="yolov8_xywh_class_probabilities":
47 if len(outputs)!=1 or tuple(outputs[0].shape)!=(1,4+classes,contract["boxes"]):
48 raise ValueError("MODEL_OUTPUT_SHAPE_MISMATCH")
49 array=outputs[0][0]
50 for i in range(contract["boxes"]):
51 scores=[_prob(array[4+c,i]) for c in range(classes)]
52 cls=max(range(classes),key=lambda c:scores[c])
53 if scores[cls]>=threshold:add(cls,scores[cls],*[float(array[c,i]) for c in range(4)])
54 else:raise ValueError("MODEL_UNKNOWN_OUTPUT_CONTRACT")
55 candidates.sort(key=lambda r:r["score"],reverse=True)
56 selected=[]
57 for item in candidates:
58 if any(item["class_id"]==old["class_id"] and _iou(item["bbox_xywh"],old["bbox_xywh"])>metadata.get("nms_iou",0.45) for old in selected):continue
59 selected.append(item)
60 if len(selected)>=metadata.get("max_detections",12):break
61 return selected
