22def _finite(x):
23 # 不依赖某些固件缺少的 math.isfinite。
24 return x == x and abs(x) < 1e30

27def _dot(a, b):
28 return sum(a[i] * b[i] for i in range(3))

31def _cross(a, b):
32 return [a[1]*b[2]-a[2]*b[1], a[2]*b[0]-a[0]*b[2], a[0]*b[1]-a[1]*b[0]]

35def _unit(a):
36 n = math.sqrt(_dot(a, a))
37 if not _finite(n) or n < 1e-10:
38 raise ValueError("degenerate_vector")
39 return [x / n for x in a]

42def _mm(a, b):
43 return [[sum(a[i][k]*b[k][j] for k in range(3)) for j in range(3)] for i in range(3)]

46def _orthogonal(r):
47 # Gram-Schmidt 加右手叉积，使 R^T R=I、det(R)=+1。
48 a = _unit([r[i][0] for i in range(3)])
49 b = [r[i][1] for i in range(3)]
50 ab = _dot(a, b)
51 b = _unit([b[i]-ab*a[i] for i in range(3)])
52 c = _cross(a, b)
53 return [[a[i], b[i], c[i]] for i in range(3)]

56def _exp(w):
57 # SO(3) 指数映射，w 是弧度旋转增量，绝不是欧拉角。
58 a2 = _dot(w, w)
59 if a2 < 1e-8:
60 s, c = 1.0-a2/6.0, 0.5-a2/24.0
61 else:
62 a = math.sqrt(a2)
63 s, c = math.sin(a)/a, (1.0-math.cos(a))/a2
64 x, y, z = w
65 k = [[0.0, -z, y], [z, 0.0, -x], [-y, x, 0.0]]
66 return [[(1.0 if i == j else 0.0) + s*k[i][j] +
67 c*(w[i]*w[j]-(a2 if i == j else 0.0)) for j in range(3)] for i in range(3)]

70def _solve(a, b):
71 # 小型线性方程：逐行缩放、部分主元消去；不依赖 numpy/ulab。
72 n = len(b)
73 m = []
74 for i in range(n):
75 scale = max(abs(x) for x in a[i])
76 if not _finite(scale) or scale < 1e-20:
77 raise ValueError("singular_system")
78 m.append([x/scale for x in a[i]] + [b[i]/scale])
79 for k in range(n):
80 p = max(range(k, n), key=lambda i: abs(m[i][k]))
81 if abs(m[p][k]) < 1e-7:
82 raise ValueError("ill_conditioned_system")
83 m[k], m[p] = m[p], m[k]
84 for i in range(k+1, n):
85 f = m[i][k]/m[k][k]
86 for j in range(k+1, n+1):
87 m[i][j] -= f*m[k][j]
88 m[i][k] = 0.0
89 x = [0.0]*n
90 for i in range(n-1, -1, -1):
91 x[i] = (m[i][n]-sum(m[i][j]*x[j] for j in range(i+1, n)))/m[i][i]
92 if not all(_finite(v) for v in x):
93 raise ValueError("nonfinite_solution")
94 return x

97class QRPoseEstimator:
98 def __init__(self, qr_size_mm, camera_matrix=CAMERA_MATRIX,
99 dist_coeffs=DIST_COEFFS, image_size=IMAGE_SIZE,
100 detector_order=None, corner_adapter=None,
101 max_reprojection_error_px=2.0, corner_noise_px=0.5,
102 max_center_disagreement_mm=8.0, max_position_uncertainty_mm=25.0,
103 min_edge_px=8.0, max_iterations=35):
104 """qr_size_mm 必填且 >0；测量编码正方形边界，不含外围白色静区。
105
106 detector_order 默认 None：detect() 不信任未经核实的检测器方向。
107 核实为 CanMV quirc 网格角点顺序后，设为 'canmv_quirc'。
108 其他固件可提供 corner_adapter(img, code)，返回自身 TL,TR,BR,BL
109 原图角点；不能可靠确定方向时必须返回 None。
110 corner_noise_px 为每个像素坐标的预期标准差，用于保守歧义门限。
111 """
112 try:
113 if isinstance(qr_size_mm, bool):
114 raise ValueError()
115 self.size = float(qr_size_mm)
116 if not _finite(self.size) or self.size <= 0.0:
117 raise ValueError()
118 except (TypeError, ValueError, OverflowError):
119 raise ValueError("qr_size_mm must be a finite positive measured size in mm")
120 try:
121 k = [[float(x) for x in row] for row in camera_matrix]
122 d = [float(x) for x in dist_coeffs]
123 if len(k) != 3 or any(len(row) != 3 for row in k) or len(d) != 5:
124 raise ValueError()
125 if not all(_finite(x) for row in k for x in row) or not all(_finite(x) for x in d):
126 raise ValueError()
127 if k[0][0] <= 0 or k[1][1] <= 0 or any(abs(x) > 1e-8 for x in
128 (k[0][1], k[1][0], k[2][0], k[2][1], k[2][2]-1.0)):
129 raise ValueError()
130 self.image_size = tuple(int(x) for x in image_size)
131 if len(self.image_size) != 2 or min(self.image_size) <= 0 or tuple(image_size) != self.image_size:
132 raise ValueError()
133 except (TypeError, ValueError, IndexError, OverflowError):
134 raise ValueError("invalid K (zero skew required), five-term D, or image_size")
135 self.fx, self.fy, self.cx, self.cy = k[0][0], k[1][1], k[0][2], k[1][2]
136 self.d = d
137 if detector_order not in (None, "canmv_quirc"):
138 raise ValueError("unknown detector_order; use corner_adapter for other detectors")
139 if corner_adapter is not None and not callable(corner_adapter):

142 values = tuple(float(x) for x in (max_reprojection_error_px, corner_noise_px,
143 max_center_disagreement_mm, max_position_uncertainty_mm, min_edge_px))
144 if not all(_finite(x) and x > 0 for x in values):
145 raise ValueError("thresholds and corner_noise_px must be finite and positive")
146 self.max_error, self.noise, self.max_center_difference, self.max_position_uncertainty, self.min_edge = values
147 self.iterations = int(max_iterations)
148 if self.iterations < 5 or self.iterations > 100:
149 raise ValueError("max_iterations must be in 5..100")

151 def _distort(self, x, y):
152 # Brown-Conrady / OpenCV 五参数模型，同时返回解析导数。
153 k1, k2, p1, p2, k3 = self.d
154 r2 = x*x+y*y
155 radial = 1.0+r2*(k1+r2*(k2+r2*k3))
156 g = k1+r2*(2.0*k2+3.0*k3*r2)
157 rx, ry = 2.0*x*g, 2.0*y*g
158 xd = x*radial+2.0*p1*x*y+p2*(r2+2.0*x*x)
159 yd = y*radial+p1*(r2+2.0*y*y)+2.0*p2*x*y
160 return (xd, yd, radial+x*rx+2*p1*y+6*p2*x,
161 x*ry+2*p1*x+2*p2*y, y*rx+2*p1*x+2*p2*y,
162 radial+y*ry+6*p1*y+2*p2*x)

164 def _undistort(self, u, v):
165 # 输出无畸变归一化坐标 (X/Z,Y/Z)，不是像素，后续单应不再用 K。
166 xd, yd = (u-self.cx)/self.fx, (v-self.cy)/self.fy
167 x, y = xd, yd
168 for _ in range(15):
169 a, b, j00, j01, j10, j11 = self._distort(x, y)
170 ex, ey = a-xd, b-yd
171 if max(abs(ex)*self.fx, abs(ey)*self.fy) < 0.0002:
172 return (x, y)
173 det = j00*j11-j01*j10
174 if not _finite(det) or det < 1e-8:
175 raise ValueError("undistortion_singular")
176 dx, dy = (j11*ex-j01*ey)/det, (-j10*ex+j00*ey)/det
177 # 回溯，避免 Newton 在不适用的标定范围内跑向远处的根。
178 accepted = False
179 for _ in range(8):
180 nx, ny = x-dx, y-dy
181 if _finite(nx) and _finite(ny) and max(abs(nx), abs(ny)) < 3:
182 a, b = self._distort(nx, ny)[:2]
183 if (a-xd)**2+(b-yd)**2 < ex*ex+ey*ey:
184 x, y, accepted = nx, ny, True
185 break
186 dx, dy = dx*0.5, dy*0.5
187 if not accepted:
188 break
189 a, b = self._distort(x, y)[:2]
190 if max(abs(a-xd)*self.fx, abs(b-yd)*self.fy) > 0.002:
191 raise ValueError("undistortion_did_not_converge")
192 return (x, y)

194 def _homography_pose(self, uv):
195 # 物体坐标暂以边长为 1；同时居中/缩放像点，改善 float32 条件数。
196 mx = sum(p[0] for p in uv)/4.0
197 my = sum(p[1] for p in uv)/4.0
198 scale = math.sqrt(sum((p[0]-mx)**2+(p[1]-my)**2 for p in uv)/4.0)
199 if scale < 1e-8:
200 raise ValueError("degenerate_normalized_corners")
201 a, b = [], []
202 for (x, y), (u, v) in zip(_SQUARE, uv):
203 u, v = (u-mx)/scale, (v-my)/scale
204 a.append([x, y, 1.0, 0.0, 0.0, 0.0, -u*x, -u*y]); b.append(u)
205 a.append([0.0, 0.0, 0.0, x, y, 1.0, -v*x, -v*y]); b.append(v)
206 h = _solve(a, b) + [1.0]
207 # 撤销像点归一化，H: (X/L,Y/L,1) -> (x_normal,y_normal,1)。
208 h = [scale*h[j]+mx*h[6+j] for j in range(3)] + [
209 scale*h[3+j]+my*h[6+j] for j in range(3)] + h[6:9]
210 c1, c2 = [h[0], h[3], h[6]], [h[1], h[4], h[7]]
211 lam = 2.0/(math.sqrt(_dot(c1, c1))+math.sqrt(_dot(c2, c2)))
212 # h33=1，故 t_z 为正；另一整体符号会把全部角点放到相机后方。
213 r = _orthogonal([[c1[i]*lam, c2[i]*lam, 0.0] for i in range(3)])
214 return r, [h[2]*lam, h[5]*lam, lam]

216 def _evaluate(self, r, t, corners, jacobian=False):
217 residual, rows = [], []
218 for (xq, yq), (u, v) in zip(_SQUARE, corners):
219 q = [r[i][0]*xq+r[i][1]*yq for i in range(3)]
220 p = [q[i]+t[i] for i in range(3)]
221 if not all(_finite(x) for x in p) or p[2] <= 1e-6:
222 raise ValueError("nonpositive_corner_depth")
223 x, y = p[0]/p[2], p[1]/p[2]
224 if max(abs(x), abs(y)) > 3.0:
225 raise ValueError("projection_out_of_model_range")
226 xd, yd, a, b, c, d = self._distort(x, y)
227 residual.extend((u-(self.fx*xd+self.cx), v-(self.fy*yd+self.cy)))
228 if jacobian:
229 for fx, dx, dy in ((self.fx, a, b), (self.fy, c, d)):
230 g = [fx*dx/p[2], fx*dy/p[2], -fx*(dx*x+dy*y)/p[2]]
231 # R <- exp(dw) R，t 单独更新；旋转导数为 q × g。
232 rows.append(_cross(q, g)+g)
233 if not all(_finite(x) for x in residual):
234 raise ValueError("nonfinite_projection")
235 return sum(x*x for x in residual), residual, rows

237 def _normal(self, rows, residual):
238 scales = [math.sqrt(sum(row[j]*row[j] for row in rows)) for j in range(6)]
239 if min(scales) < 1e-8:
240 raise ValueError("unobservable_pose")
241 j = [[row[k]/scales[k] for k in range(6)] for row in rows]
242 a = [[sum(row[i]*row[k] for row in j) for k in range(6)] for i in range(6)]
243 b = [sum(row[i]*e for row, e in zip(j, residual)) for i in range(6)]
244 return a, b, scales

246 def _refine(self, r, t, corners):
247 damping = 0.001
248 cost, residual, rows = self._evaluate(r, t, corners, True)
249 converged = False
250 for _ in range(self.iterations):
251 a, b, scales = self._normal(rows, residual)
252 if max(abs(v) for v in b) < 0.00002 or cost < 1e-8:
253 converged = True
254 break
255 accepted = False
256 for _ in range(9):
257 damped = [[a[i][j]+(damping if i == j else 0.0) for j in range(6)] for i in range(6)]
258 step = _solve(damped, b)
259 step = [step[i]/scales[i] for i in range(6)]
260 # 防止单步跨过正深度边界或大的旋转分支。
261 f = max(1.0, math.sqrt(_dot(step[:3], step[:3]))/0.35,
262 math.sqrt(_dot(step[3:], step[3:]))/max(0.1, 0.4*t[2]))
263 step = [v/f for v in step]
264 nr = _orthogonal(_mm(_exp(step[:3]), r))
265 nt = [t[i]+step[i+3] for i in range(3)]
266 try:
267 nc, ne, nj = self._evaluate(nr, nt, corners, True)
268 except ValueError:
269 nc = 1e30
270 if nc < cost:
271 improvement = cost-nc
272 r, t, cost, residual, rows = nr, nt, nc, ne, nj
273 damping = max(1e-7, damping*0.3)
274 accepted = True
275 if improvement < max(1e-8, cost*2e-6):
276 converged = True
277 break
278 damping *= 10.0
279 if converged:
280 break
281 if not accepted:
282 # float32 在极值附近可能无法再减小残差；最后仍做梯度检查。
283 converged = max(abs(v) for v in b) < 0.002
284 break
285 if not converged:
286 _, b, _ = self._normal(rows, residual)
287 converged = max(abs(v) for v in b) < 0.002
288 if not converged:
289 raise ValueError("optimizer_not_converged")
290 return cost, r, t

292 def _seeds(self, r, t):
293 yield r, t
294 # 关于中心视线反射法向，构造平面双解的另一侧初值。
295 n = [r[i][2] for i in range(3)]
296 v = _unit(t)
297 n2 = [2.0*_dot(n, v)*v[i]-n[i] for i in range(3)]
298 axis = _cross(n, n2)
299 sn, cs = math.sqrt(_dot(axis, axis)), _dot(n, n2)
300 if sn > 1e-6:
301 angle = math.atan2(sn, cs)
302 mirror = _mm(_exp([a*angle/sn for a in axis]), r)
303 yield mirror, t
304 # 从多个倾角寻找不同局部极小；有限多起点不构成全局唯一性证明。
305 for angle in (-0.55, 0.55):
306 yield _mm(_exp([angle, 0.0, 0.0]), r), t
307 yield _mm(_exp([0.0, angle, 0.0]), r), t

309 def _position_uncertainty(self, r, t, corners):
310 """Translation covariance with model tilt as nuisance parameters, mm.
311
312 No rotation confidence or angular acceptance is computed. The three
313 translation columns of the inverse normal matrix retain coupling to
314 tilt; treating a fitted plane as perfectly known would understate error.
315 """
316 _, residual, rows = self._evaluate(r, t, corners, True)
317 a, _, scales = self._normal(rows, residual)
318 variance = 0.0
319 for i in range(3, 6):
320 b = [0.0]*6; b[i] = 1.0
321 diagonal = _solve(a, b)[i]/(scales[i]*scales[i])
322 if diagonal <= 0 or not _finite(diagonal):
323 raise ValueError("position_unobservable")
324 variance += diagonal*self.noise*self.noise*self.size*self.size
325 return 3.0*math.sqrt(variance)

332 def estimate_from_corners(self, corners, actual_image_size=None):
333 """输入原始畸变图的物理 TL,TR,BR,BL；不得输入 rect() 外接框。
334
335 调用者负责确认视场/裁剪/翻转/缩放与标定完全一致。
336 actual_image_size 可用于额外检查，未知的几何变化不能仅靠图像尺寸检测。
337 返回字典；失败不返回伪造的 R/t/T。正面 QR、非镜像图是本模块前提。
338 """
339 points = None
340 try:
341 if actual_image_size is not None and tuple(actual_image_size) != self.image_size:
342 raise ValueError("image_size_mismatch")
343 if len(corners) != 4 or any(len(p) != 2 for p in corners):
344 raise ValueError("need_four_2d_corners")
345 points = [(float(p[0]), float(p[1])) for p in corners]
346 width, height = self.image_size
347 if not all(_finite(x) for p in points for x in p):
348 raise ValueError("nonfinite_corners")
349 if not all(0 <= u < width and 0 <= v < height for u, v in points):
350 raise ValueError("corners_outside_calibrated_image")
351 crosses = []
352 for i in range(4):
353 a, b, c = points[i], points[(i+1)%4], points[(i+2)%4]
354 ex, ey, fx, fy = b[0]-a[0], b[1]-a[1], c[0]-b[0], c[1]-b[1]
355 if ex*ex+ey*ey < (self.min_edge*self.min_edge if boundary_checks() else 1e-8):
356 raise ValueError("qr_too_small_or_duplicate_corners")
357 crosses.append(ex*fy-ey*fx)
358 if min(crosses) <= 1e-4:
359 raise ValueError("nonconvex_mirrored_or_wrong_corner_winding")
360 area = 0.5*sum(points[i][0]*points[(i+1)%4][1]-points[i][1]*points[(i+1)%4][0] for i in range(4))
361 span2 = max((a[0]-b[0])**2+(a[1]-b[1])**2 for a in points for b in points)
362 if area < 32.0 or area/span2 < 0.015 or min(crosses)/span2 < 0.002:
363 raise ValueError("degenerate_or_edge_on_quadrilateral")
364 r, t = self._homography_pose([self._undistort(u, v) for u, v in points])
365 candidates, failed_starts = [], 0
366 for sr, st in self._seeds(r, t):
367 try:
368 candidate = self._refine(sr, st, points)
369 cr, ct = candidate[1:]
370 # Deduplicate identical numerical models only. No heading
371 # difference is computed or used to accept a position.
372 if not any(sum((cr[i][j]-x[1][i][j])**2 for i in range(3) for j in range(3)) < 1e-10 and
373 sum((ct[i]-x[2][i])**2 for i in range(3)) < 1e-10 for x in candidates):
374 candidates.append(candidate)
375 except (ValueError, OverflowError, ZeroDivisionError):
376 failed_starts += 1
377 if not candidates:
378 raise ValueError("no_converged_position_model")
379 candidates.sort(key=lambda x: x[0])
380 cost, r, t = candidates[0]
381 error = math.sqrt(cost/4.0) # 四角二维欧氏残差平方的均方根，单位像素。
382 if boundary_checks() and error > self.max_error:
383 return self._failure("reprojection_error_too_large", points,
384 reprojection_error_px=error)
385 # Only center disagreement is relevant to this application. Two
386 # plane models may face differently while locating the same center.
387 plausible=[item for item in candidates if item[0]-cost <= 9.0*self.noise*self.noise]
388 center_difference=max(math.sqrt(sum(((item[2][i]-t[i])*self.size)**2 for i in range(3)))
389 for item in plausible)
390 if center_difference > self.max_center_difference:
391 return self._failure("qr_center_ambiguous", points,
392 center_disagreement_mm=center_difference)
393 uncertainty=max(self._position_uncertainty(rr,tt,points) for _,rr,tt in plausible)
394 if uncertainty > self.max_position_uncertainty:
395 return self._failure("qr_position_uncertain", points,
396 position_3sigma_mm=uncertainty)
397 tm = [v*self.size for v in t]
398 if not all(_finite(v) for v in tm) or not _finite(_dot(tm, tm)):
399 raise ValueError("nonfinite_metric_translation")
400 transform = [r[i][:]+[tm[i]] for i in range(3)]+[[0.0, 0.0, 0.0, 1.0]]
401 return {"ok": True, "reason": "ok", "payload": None, "corners_px": points,
402 "R_camera_from_qr": r, "t_camera_from_qr_mm": tm,
403 "T_camera_from_qr": transform, "depth_mm": tm[2],
404 "distance_mm": math.sqrt(_dot(tm, tm)),
405 "reprojection_error_px": error,
406 "boundary_checks_enabled":boundary_checks(),
407 "boundary_warning":error>self.max_error,
408 "position_camera_mm": tm, "position_3sigma_mm": uncertainty,
409 "center_disagreement_mm": center_difference,
410 "_candidate_models": [(rr,tt) for _,rr,tt in plausible],
411 "candidate_count": len(candidates),
412 "failed_starts": failed_starts}
413 except (ValueError, TypeError, IndexError, OverflowError, ZeroDivisionError) as exc:
414 return self._failure(str(exc), points)

416 def detect(self, img, target_payload=None, roi=None, geometry_confirmed=False):
417 """全图或原图 ROI 检测；返回列表，未检测到目标则 []。
418
419 geometry_confirmed=True 表示调用者确认这是标定对应的原始图像：
420 同一传感器模式、裁剪、缩放、翻转、视场与焦距，未事先去畸变。
421 仅检查 width/height 无法识别同尺寸不同视场，所以需显式声明。
422 CanMV quirc 的 ROI 输出已经是原图坐标，本方法不重复加 ROI 偏移。
